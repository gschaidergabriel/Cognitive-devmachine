def _make_handler():
    class Handler(BaseHTTPRequestHandler):
        def log_message(self, fmt, *args):
            pass  # keine Zugriffs-Logs auf Disk

        def _json(self, obj, code=200):
            b = json.dumps(obj, ensure_ascii=False).encode("utf-8")
            self.send_response(code)
            self.send_header("Content-Type", "application/json; charset=utf-8")
            self.send_header("Content-Length", str(len(b)))
            self.send_header("Cache-Control", "no-store")
            self.end_headers()
            self.wfile.write(b)

        def _bild_ausliefern(self, voller_pfad):
            # Bedingtes GET statt Stunden-Cache. LEHRE (2026-09-11): vorher
            # "Cache-Control: max-age=3600" auf allen Bildern -- der Browser
            # hat eine Stunde lang NIE nachgefragt, egal was auf der Platte
            # neu lag. Jetzt: immer neu anfragen (no-cache = revalidieren,
            # nicht "nie cachen"), Server antwortet mit 304 wenn unveraendert
            # -> genauso billig wie vorher, aber Aenderungen wirken sofort.
            try:
                mtime = os.path.getmtime(voller_pfad)
            except OSError:
                self.send_response(404); self.end_headers(); return
            last_mod = time.strftime("%a, %d %b %Y %H:%M:%S GMT", time.gmtime(mtime))
            ims = self.headers.get("If-Modified-Since")
            if ims and ims == last_mod:
                self.send_response(304)
                self.send_header("Cache-Control", "no-cache")
                self.end_headers()
                return
            with open(voller_pfad, "rb") as f:
                daten = f.read()
            self.send_response(200)
            self.send_header("Content-Type", "image/png")
            self.send_header("Content-Length", str(len(daten)))
            self.send_header("Cache-Control", "no-cache")
            self.send_header("Last-Modified", last_mod)
            self.end_headers()
            self.wfile.write(daten)

        def do_GET(self):

            path, qs = u.path, urllib.parse.parse_qs(u.query)

                self.send_header("Content-Type", "text/html; charset=utf-8")
                self.send_header("Content-Length", str(len(aktuelle_seite)))
                self.send_header("Cache-Control", "no-store")
                self.end_headers()
                self.wfile.write(aktuelle_seite)
            elif path == "/filmlook.js":

                # Farbabstimmung, Vignette) liegt in EINER eigenen Datei, nicht
                # in buehne.html. Grund: die Seite gehoert waehrend des Umbaus
                # dem Animations-Agenten -- zwei Agenten in derselben Datei
                # erzeugen Konflikte. So braucht buehne.html nur EINE Zeile
                # (<script src="/filmlook.js">), und der Look kann unabhaengig
                # weitergebaut oder komplett abgeschaltet werden (Datei leeren).
