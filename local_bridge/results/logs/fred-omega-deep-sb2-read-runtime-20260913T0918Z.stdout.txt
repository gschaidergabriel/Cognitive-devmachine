def _repo_path(rel: str) -> Path:

def _sha256(path: Path) -> str | None:
    try:
        h = hashlib.sha256()
        with path.open("rb") as fh:
            for chunk in iter(lambda: fh.read(1024 * 1024), b""):
                h.update(chunk)
        return h.hexdigest()
    except OSError:
        return None

def _check(name: str, ok: bool, detail: str) -> dict[str, Any]:
    return {"name": name, "ok": bool(ok), "detail": detail}


def _launcher_structure(source: str) -> tuple[bool, bool, str]:
    """Return (has_sqlite_import, has_resident_loop, parse_detail)."""
    try:
        tree = ast.parse(source)
    except SyntaxError as exc:
        return True, True, f"parse failed: {exc}"

    has_db_import = False
    has_resident_loop = False
    for node in ast.walk(tree):
        if isinstance(node, ast.Import) and any(alias.name == "sqlite3" for alias in node.names):
            has_db_import = True
        elif isinstance(node, ast.ImportFrom) and node.module == "sqlite3":
            has_db_import = True
        elif isinstance(node, ast.While) and isinstance(node.test, ast.Constant) and node.test.value is True:
            has_resident_loop = True
        elif isinstance(node, ast.Call) and isinstance(node.func, ast.Attribute) and node.func.attr == "serve_forever":
            has_resident_loop = True
    return has_db_import, has_resident_loop, "AST parsed"


def _sql_mutation_literals(source: str) -> list[str]:
    """Find literal SQL mutation statements without flagging SELECT-only DB access."""
    try:
        tree = ast.parse(source)
    except SyntaxError:
        return ["<parse-failed>"]
    prefixes = ("INSERT ", "UPDATE ", "DELETE ", "CREATE ", "DROP ", "ALTER ", "REPLACE ")
    found: set[str] = set()
    for node in ast.walk(tree):
        if isinstance(node, ast.Constant) and isinstance(node.value, str):
            text = " ".join(node.value.upper().split())
            for prefix in prefixes:
                if text.startswith(prefix):
                    found.add(prefix.strip())
    return sorted(found)

def _exec_script(path: Path, argv: list[str]) -> int:
    if not path.is_file():
        print(f"OMEGA: organ missing: {path}", file=sys.stderr)
        return 2
    os.execv(sys.executable, [sys.executable, str(path), *argv])
    return 127
