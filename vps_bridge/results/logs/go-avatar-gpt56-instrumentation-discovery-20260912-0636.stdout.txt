  function wennBereit(fn) {
    if (document.readyState === "loading") {

    }
  }

    var buehne = document.getElementById("buehne");
    var kamera = document.getElementById("kamera");
    if (!buehne || !kamera) {
      console.warn(
        "[filmlook] #buehne/#kamera nicht gefunden -- Seitenstruktur anders " +

      );

    }
