function regenZeichnenEbene(c, cv, liste, alpha, breite){
  c.clearRect(0,0,cv.width, cv.height);
  c.lineWidth = breite;
  for(const t of liste){
    c.strokeStyle = `rgba(${t.ton},${alpha})`;
    c.beginPath();
    c.moveTo(t.x, t.y);
    c.lineTo(t.x, t.y + t.len);
    c.stroke();
    t.y += t.v * 3.2;
  }
}

}
