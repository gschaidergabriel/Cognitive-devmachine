[Unit]
Description=FranKenstein Koerperpflege -- er kuemmert sich selbst um seinen Leib (RAM/Swap/Platte/Temperatur)
After=graphical-session.target
# StartLimitBurst/IntervalSec stehen im [Unit]-Abschnitt (systemd-Vorgabe),
# nicht im [Service]-Abschnitt. Bremse gegen einen Neustart-Loop, wie am
# 2026-08-07 bei synteq-frank-wacht (Restart=always ohne Bremse blockierte
# die Tastatur, 44 Units mussten hart abgeschaltet werden). Hier: hoechstens
# 5 Neustarts in 5 Minuten, danach bleibt die Unit failed liegen statt
# endlos neu zu starten.
StartLimitIntervalSec=300
StartLimitBurst=5

[Service]
Type=simple
WorkingDirectory=/opt/entityos/voice
Environment=VOICE_DIR=/opt/entityos/voice
Environment=FRANKENSTEIN_UNIFIED_DB=%h/.local/share/agentzero/unified.db
# Takt 60s -- eng genug, um ein volllaufendes Swap noch vor dem naechsten
# Vorfall zu bemerken, weit genug, um selbst keine messbare Last zu sein.
Environment=KOERPERPFLEGE_TAKT_S=60
ExecStart=/opt/entityos/voice/venv/bin/python3 /opt/entityos/voice/koerperpflege.py
# Bremse gegen Neustart-Loop (siehe StartLimit* oben): on-failure statt
# always, plus 30s Abstand -- kein Prozess, der sich selbst in eine
# Endlosschleife aus Abstuerzen und Sofort-Neustarts katapultieren kann.
Restart=on-failure
RestartSec=30
Nice=15

[Install]
WantedBy=default.target
