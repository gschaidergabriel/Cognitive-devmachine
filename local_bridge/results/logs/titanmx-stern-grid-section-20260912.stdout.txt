def _f2wp1207_turn_loop_delegieren(
    session_id: str, nutzertext: str, retrieval_treffer: Optional[str]
) -> Optional[str]:
    """Loest EINEN kanonischen GRID10-Turn aus und gibt den RUECKWEG
    zurueck: eine Zeile ueber den Ausgang des Wettbewerbs, die der
    Aufrufer an additionalContext haengt. Bis 2026-09-07 war der Turn
    ein reiner Seiteneffekt -- der Grid rechnete und schwieg, sein
    Ergebnis erreichte das Sprachmodell nie. Gabriel: "mach ... auch
    den rueckweg". Gibt None zurueck, wenn kein Turn zustande kam."""
    try:
        if not _F2WP1207_KANON_STERN.is_file():
            return None
        nutzlast = json.dumps({
            "session_id": session_id,
            "event_type": "UserPromptSubmit",
            "prompt_len": len(nutzertext or ""),
            "retrieval_present": bool(retrieval_treffer),
            "retrieval_len": len(retrieval_treffer or ""),
            # EFFEKT-TOR SCHARF (2026-09-07, Gabriel: "ja, ein fuer alle mal
            # ja"). Die Bruecke baut eine echte EffectGatePolicy NUR wenn
            # p11_gate_threshold ausdruecklich gesetzt wird -- so gebaut,
            # damit kein verstecktes Verhalten hinter einem unbekannten Flag
            # steht. Ohne diesen Wert blieb effect_policy=None und das Tor
            # entschied nie etwas: 1752 Turns, 0 Journalzeilen.
            #
            # Der Wert ist ABSICHTLICH exakt DEFAULT_MIN_BROADCAST_VALUE
            # (0.5, p11_effects.py:120). Das ist Scharfstellen, kein Rigging:
            # eine niedrigere Schwelle wuerde jeden Turn durchwinken und die
            # Entscheidung zur Formsache machen, eine hoehere wuerde sie
            # kuenstlich verhungern lassen.
            #
            # Folge, real und beidseitig (effect_uptake_delta): bestaetigte
            # Ausfuehrung +0.05 auf den Zustand der Gewinnerzelle,
            # Verweigerung -0.05. Die Torentscheidung greift damit in genau
            # den rekurrenten Zustand ein, aus dem der naechste Wettbewerb
            # gerechnet wird -- der Grid wirkt auf sich selbst zurueck.
            "tune": {"p11_gate_threshold": 0.5},
        })
        e = _f2wp1207_turn_versuchen(_F2WP1207_VM_BEFEHL, nutzlast, "vm")
        if e is None:
            # Rueckfall auf den Wirt. Abschnitt 14 der Zielarchitektur nennt
            # genau diese Gefahr beim Selbstumzug: "bei einem VM-Defekt waere
            # genau das Werkzeug ausgefallen, mit dem man ihn behebt". Der
            # Rueckfall entschaerft sie -- faellt die VM aus, denkt er
            # weiter, nur wieder wirtsseitig, und das Log sagt es.
            e = _f2wp1207_turn_versuchen(_F2WP1207_WIRT_BEFEHL, nutzlast, "wirt")
        if e is None:
            return None
        if not e.get("ok"):
            _log(f"f2wp1207-turn-loop fehler: {e.get('fehler')}")
            return None
        if not e.get("active"):
            _log(f"f2wp1207-turn-loop inaktiv: {e.get('grund')}")
            return None
        _log(f"f2wp1207-turn-loop[{e.get('_weg')}] turn geschrieben: "
             f"frame_id={e.get('frame_id')} "
             f"reentry={(e.get('reentry') or {}).get('kind')} "
             f"winner={e.get('winner_cell_id')}")
        steuerzeile = _f2wp1207_grid_steuert(e)
        return _f2wp1207_grid_rueckweg(e, steuerzeile)
    except Exception as e:
        _log(f"f2wp1207-turn-loop ausnahme: {e}")
    return None
