# ── Token-Einwurf: Anbieter aus einem rohen API-Schluessel erkennen ────────
# Praefixe, die WIRKLICH im Tresor/keys.env dieser Maschine beobachtet wurden
# (vault-liste 2026-08-27, siehe Bauprotokoll) + ein paar allgemein dokumen-
# tierte Praefixe fuer Anbieter, die noch keinen Eintrag haben (Together,
# Anthropic, OpenAI, HuggingFace) -- letztere NICHT live gegen diese Maschine
# verifiziert, deshalb "sicher": False fuer die, die nicht selbst beobachtet
# wurden. Mistral-Schluessel haben KEIN erkennbares Praefix (32 Zeichen,
# reines alnum) -- dafuer gibt es keine zuverlaessige Praefixerkennung, nur
# eine Laengen-Heuristik mit sicher=False.

]


def provider_erkennen(secret: str) -> dict:
    """Ordnet einen rohen API-Schluessel per Praefix einem Anbieter zu.
    Rueckgabe: {"anbieter", "url", "konto_vorschlag", "sicher"}. Wenn kein
    Praefix passt: Laengen-Heuristik fuer Mistral (32 Zeichen reines alnum,
    das einzige bekannte praefixlose Format in diesem Pool), sonst
    "anbieter": None -- der Aufrufer (Dashboard) muss dann den Anbieter per
    Dropdown von Hand waehlen, es wird NICHT geraten und als sicher
    ausgegeben."""
    s = (secret or "").strip()
    for praefix, anbieter, url, konto, sicher in PRAEFIX_ERKENNUNG:
        if s.startswith(praefix):
            return {"anbieter": anbieter, "url": url, "konto_vorschlag": konto, "sicher": sicher}
    if len(s) == 32 and s.isalnum():
        return {"anbieter": "Mistral AI (Vermutung -- kein Praefix, nur Laengen-Heuristik)",
                "url": "https://console.mistral.ai", "konto_vorschlag": "MISTRAL_API_KEY",
                "sicher": False}
    return {"anbieter": None, "url": None, "konto_vorschlag": None, "sicher": False}
