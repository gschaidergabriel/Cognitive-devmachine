class Blickfuehrung:
    """Ein Prozess-weites Singleton (siehe instanz()). starten()/stoppen()
    sind idempotent und duerfen von der Fensterverwaltung direkt an
    oeffnen()/schliessen() gehaengt werden."""

    def __init__(self):
        self._cap = None
        self._kaskade = None
        self._thread = None
        self._lauf = False
        self._lock = threading.Lock()
        self._ziel = None          # (cx, cy, anteil) normiert 0..1
        self._kandidat_seit = 0
        self._letzte_erkennung_ts = 0.0
        self._letzte_emotion = None       # letztes face_emotion.classify_faces()-dict
        self._letzte_emotion_ts = 0.0
        self.verfuegbar = cv2 is not None

    def _kamera_oeffnen(self):
        for idx in (KAMERA_INDEX, 0, 1, 2, 3, 4):
            try:
                cap = cv2.VideoCapture(idx, cv2.CAP_V4L2)
            except Exception:
                continue
            if cap.isOpened():
                cap.set(cv2.CAP_PROP_FRAME_WIDTH, 640)
                cap.set(cv2.CAP_PROP_FRAME_HEIGHT, 360)
                return cap
            cap.release()
        return None

    def starten(self) -> bool:
        """Kameraexklusivitaet: stoppt frankenstein-sehen, oeffnet selbst,
        startet den Erkennungs-Thread. Gibt False zurueck (und macht den
        Sichtdienst sofort wieder an), wenn kein Kamerazugriff moeglich war
        -- kein Vortaeuschen einer Blickfuehrung ohne echtes Bild."""
        if not self.verfuegbar:
            return False
        with self._lock:
            if self._lauf:
                return True
            subprocess.run(["systemctl", "--user", "stop", "frankenstein-sehen"],
                            stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
            time.sleep(0.6)  # Geraet muss wirklich frei werden, siehe Docstring
            try:
                self._kaskade = cv2.CascadeClassifier(
                    cv2.data.haarcascades + "haarcascade_frontalface_default.xml")
            except Exception:
                self._kaskade = None
            self._cap = self._kamera_oeffnen() if self._kaskade is not None else None
            if self._cap is None:
                # Kein sicherer Weg -- Sichtdienst SOFORT wieder hochfahren,
                # ehrlich aufgeben statt eine kaputte Blickfuehrung zu bauen.
                subprocess.run(["systemctl", "--user", "start", "frankenstein-sehen"],
                                stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
                return False
            try:
                with open(LOCK_PFAD, "w") as f:
                    f.write(str(os.getpid()))
            except Exception:
                pass
            self._ziel = None
            self._kandidat_seit = 0
            self._lauf = True
            self._thread = threading.Thread(target=self._schleife, daemon=True)
            self._thread.start()
            print("[blick] Kamera uebernommen, frankenstein-sehen angehalten")
            return True

    def stoppen(self):
        """IMMER (auch im Fehlerfall) wird frankenstein-sehen am Ende neu
        gestartet -- das ist Gabriels harte Bedingung, siehe Docstring."""
        with self._lock:
            war_aktiv = self._lauf
            self._lauf = False
        if self._thread:
            self._thread.join(timeout=2.0)
            self._thread = None
        if self._cap is not None:
            try:
                self._cap.release()
            except Exception:
                pass
            self._cap = None
        try:
            os.remove(LOCK_PFAD)
        except Exception:
            pass
        subprocess.run(["systemctl", "--user", "start", "frankenstein-sehen"],
                        stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
        if war_aktiv:
            print("[blick] Kamera freigegeben, frankenstein-sehen wieder gestartet")
