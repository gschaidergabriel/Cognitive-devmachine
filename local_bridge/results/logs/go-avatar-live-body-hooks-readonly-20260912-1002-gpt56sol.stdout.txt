// ⭐ 2026-09-11 (Gabriel woertlich: "als wuerde er sich selbst echt mit dem
// Handy filmen, am besten sogar ganz leicht Cam-Shake, aber realistisch
// passend zur Bewegung. Volle Immersion..."). EIN gemeinsamer Transform auf
// #kamera fuer die GESAMTE Szene (Hintergrund+Koerper+Kopf+Mund+Augen) --
// nie pro Ebene einzeln, sonst zerfaellt die Illusion sofort ("kein
// echtes Handy waere so exakt synchron mit sich selbst"). Drei
// ueberlagerte, bewusst NICHT-zufaellige Komponenten (Gabriels Vorgabe:
// gleichfoermiges Zufallsrauschen sieht nicht wie eine echte Hand aus):
//  1) langsame Drift (Atmung/Ermuedung der haltenden Hand, ~4-5s Periode,
//     wenige px) -- Fortsetzung der bisherigen "nie ganz still"-Drift,
//     jetzt auf die ganze Kamera statt nur auf #gesicht verschoben.
//  2) schnelles Mikro-Zittern (~8-12Hz, <1px) -- der Muskelzittern-Anteil
//     einer echten Hand, zwei knapp benachbarte Sinusfrequenzen statt einer
//     einzelnen, damit es nicht mechanisch/taktfest wirkt.
//  3) minimale Rotation (Bruchteile eines Grads) -- eine Hand kippt nie rein
//     linear.
// Hintergrund bewegt sich absichtlich einen Hauch WENIGER als der Rest
// (0.85x) -- winziger Parallaxe-Tiefenhinweis, siehe Koordinator ("optional:
// Hintergrund bewegt sich minimal weniger als der Vordergrund").
// ⭐⭐ 2026-09-11 RETUNE nach Gabriels Live-Urteil ("er zittert, schaut
// relativ shit aus"): die 8-10Hz-Mikrozitter-Komponente unten war zu schnell
// UND zu stark -- auf einem kleinen Panel (~296px breit), mit dem neuen
// engeren Kopf-Crop, UND auf einem Chrome-Fenster ohne GPU-Kompositing
// (--disable-gpu, siehe Fensterstart) liest sich eine 8-10Hz-Oszillation
// nicht als sanftes Handzittern, sondern als sichtbares Vibrieren/Ruckeln --
// das Auge ist fuer zeitliches Flackern empfindlicher als fuer die reine
// Pixelamplitude. Fix: Frequenz UND Amplitude aller drei Komponenten deutlich
// gesenkt (Tremor von ~10/8Hz auf ~3/2.3Hz, Drift/Rotation-Amplitude
// grob halbiert). Ziel ist jetzt spuerbar ruhiger -- lieber zu wenig als
// wieder "zittert".
let driftPhaseX = Math.random() * Math.PI * 2;
let driftPhaseY = Math.random() * Math.PI * 2;
let tremorPhaseA = Math.random() * Math.PI * 2;
let tremorPhaseB = Math.random() * Math.PI * 2;
let rotPhase = Math.random() * Math.PI * 2;
function kameraWackeln(){
  const t = Math.max(0.4, Math.min(1.6, tempo)); // "passend zur Bewegung": etwas mehr Schub bei hoeherem Tempo
  driftPhaseX += 0.0131; // ~8.0s Periode bei 60fps (war ~4.0s)
  driftPhaseY += 0.0105; // ~9.6s Periode -- absichtlich versetzt zu X (war ~4.9s)
  tremorPhaseA += 0.314;  // ~3Hz (war ~10Hz)
  tremorPhaseB += 0.251;  // ~2.4Hz, knapp daneben statt exakt gleich (war ~8Hz)
  rotPhase += 0.0063;     // ~16s Periode, noch langsamer als die Drift (war ~8.3s)
  const driftX = Math.sin(driftPhaseX) * 0.7 * t;
  const driftY = Math.cos(driftPhaseY) * 0.55 * t;
  const tremorX = (Math.sin(tremorPhaseA) * 0.14 + Math.sin(tremorPhaseB) * 0.09) * t;
  const tremorY = (Math.cos(tremorPhaseA * 0.97) * 0.12 + Math.cos(tremorPhaseB * 1.05) * 0.07) * t;
  const rotDeg = Math.sin(rotPhase) * 0.06 * t;
  const x = driftX + tremorX, y = driftY + tremorY;
  $("kamera").style.transform = `translate(${x.toFixed(2)}px, ${y.toFixed(2)}px) rotate(${rotDeg.toFixed(3)}deg)`;
  // Hintergrund bekommt denselben Wackel, aber 0.85x gedaempft -- winzige
  // Parallaxe, das Bild wirkt nicht wie eine starre Fotomontage auf dem
  // Hintergrund geklebt. #hg-frame ist Kind von #kamera -- dieser Transform
  // addiert sich zu dem der Kamera, deshalb hier bewusst der GEGEN-Anteil
  // (0.85-1 = -0.15x) statt des vollen Werts.
  $("hg-frame").style.transform = `translate(${(x*-0.15).toFixed(2)}px, ${(y*-0.15).toFixed(2)}px)`;
}
