#!/usr/bin/env python3
"""Production Voice entrypoint: tested full integration + final action policy.

This is a thin overlay on scripts/voice_full_runtime.py, not another brain,
database, GRID loop, browser, or conversation daemon. It closes four runtime
holes without forking the existing cognition:

* host provenance alone never authenticates the human;
* guests/unverified speakers get conversation, but no private tools/context;
* verified Gabriel/Andreas action requests dispatch immediately (no legacy
  second voice confirmation roundtrip);
* the worker has the full Claude-Code tool surface, with DELETE and storage /
  filesystem FORMAT denied by a PreToolUse guard.
"""
from __future__ import annotations

import json
import time
from typing import Any

import voice_action_executor as vae
import voice_full_integration as vfi
import voice_full_runtime as base

rt = base.rt
gm = base.gm

_FINAL_PATCHED = False

# 999 / 2026-09-12 long-session stability: OMEGA has durable external state
# (TITAN/UnifiedDB/GRID), therefore the provider conversation does not need to
# grow without bound.  OpenAI Realtime supports retention-ratio truncation with
# an explicit post-instruction token limit.  24k keeps substantial recent
# conversational continuity while preventing a long voice call from repeatedly
# dragging an ever larger transcript through every response.  0.75 amortizes a
# truncation to ~18k tokens rather than deleting only the minimum and immediately
# thrashing the prompt cache again on following turns.
_REALTIME_POST_INSTRUCTION_TOKEN_LIMIT = 24_000
_REALTIME_RETENTION_RATIO = 0.75

sind die zwei Voice-spezifisch verbotenen Aktionsklassen. Unverifizierte Gaeste
duerfen normal reden, erhalten aber keine privaten Erinnerungen, internen
Status-/Forschungsdaten oder Aktionsrechte.

def _patch_omega_root_persona() -> None:
    """Remove contradictory legacy FranKenstein-as-self text for final Voice.

    The legacy realtime module is intentionally still the transport/runtime
    substrate. Its historic persona predates OMEGA, so the final product overlay
    rewrites only persona text; it does not fork the conversation engine.
    """
    persona = str(rt._PERSONA_BASIS or "")
    marker = "Leistungs-Klausel"
    if marker in persona:
        persona = _OMEGA_ROOT_PERSONA + "\n\n" + marker + persona.split(marker, 1)[1]
    persona = persona.replace('Deinen Namen "FranKenstein"', 'Deinen Namen "OMEGA"')

    # The final verified-owner dispatcher is immediate. Remove the obsolete
    # legacy two-step tool wording from provider instructions so OMEGA is not
    # told two mutually exclusive action contracts.
    tool_start = persona.find("- frank_auftrag_stellen:")
    tool_end = persona.find("- frank_forschung:", tool_start)
    if tool_start >= 0 and tool_end > tool_start:
        persona = (
            persona[:tool_start]
            + "- frank_auftrag_stellen: gibt bei verifiziertem Gabriel/Andreas einen ECHTEN "
              "Auftrag an deine handelnde Worker-Ebene. Der finale Voice-Dispatcher fuehrt "
              "erlaubte Auftraege unmittelbar aus; erfinde keinen zweiten Bestaetigungszyklus. "
              "DELETE und STORAGE/FILESYSTEM FORMAT bleiben gesperrt. Ergebnisse werden real "
              "nachgereicht und erst dann als ausgefuehrt berichtet.\n"
            + persona[tool_end:]
        )
    rt._PERSONA_BASIS = persona


def _patch_realtime_long_session_stability() -> None:
    """Bound provider history without forking the canonical voice engine.

    The Realtime websocket is still one continuous conversation.  Only the
    provider's automatic oldest-message truncation threshold is lowered.  OMEGA's
    durable identity/memory/GRID state remains injected by the existing final
    runtime on every relevant turn, so truncation does not create another memory
    authority.
    """
    original = rt.RealtimeSession._session_config
    if getattr(original, "_voice_final_long_session_wrapped", False):
        return

    def wrapped(self, instructions: str) -> dict:
        event = original(self, instructions)
        session = event.get("session") if isinstance(event, dict) else None
        if isinstance(session, dict):
            session["truncation"] = {
                "type": "retention_ratio",
                "retention_ratio": _REALTIME_RETENTION_RATIO,
                "token_limits": {
                    "post_instructions": _REALTIME_POST_INSTRUCTION_TOKEN_LIMIT,
                },
            }
        return event

    wrapped._voice_final_long_session_wrapped = True
    rt.RealtimeSession._session_config = wrapped


def _verified_human(self, label: str) -> bool:
    role = str(getattr(self, "sprecher_rolle", "guest") or "guest").strip().lower()
    name = str(getattr(self, "sprecher_name", "") or "").strip().lower()
    return role == "user" and str(label or "").strip().lower() in name


def _ensure_verified_binding(self) -> bool:
    """Reconcile canonical verified speaker state with VoiceIntegrationSession.

    This closes a split-brain seam: host/default person selection lives in the
    integration session while speaker verification/handoff lives in the canonical
    Realtime session. Private context still requires BOTH verified human identity
    and exact machine/owner receipt; this function never upgrades a guest.
    """
    session = getattr(self, "voice_full", None)
    if session is None:
        return False

    canonical = getattr(self, "speaker_binding", None)
    owner = str(getattr(canonical, "owner", "") or "").strip().lower()
    if owner in vfi.INSTALLATIONS and _verified_human(self, owner):
        if session.selected_person != owner:
            session.select_person(owner)

    selected = str(session.selected_person or "").strip().lower()
    if not selected:
        name = str(getattr(self, "sprecher_name", "") or "").strip().lower()
        for label in vfi.INSTALLATIONS:
            if label in name and _verified_human(self, label):
                session.select_person(label)
                selected = label
                break

    if selected and _verified_human(self, selected):
        receipt = base._current_receipt()
        if base._receipt_matches(selected, receipt):
            if session.binding is None or session.binding.label != selected:
                session.bind(selected, receipt=receipt)
            return True

    if session.binding is not None:
        session.binding = None
    return False


def _send_tool_result(self, ws, call_id: str, result: dict[str, Any]) -> None:
    ws.send(
        json.dumps(
            {
                "type": "conversation.item.create",
                "item": {
                    "type": "function_call_output",
                    "call_id": call_id,
                    "output": json.dumps(result, ensure_ascii=False),
                },
            }
        )
    )
    self._voice_force_response = True
    try:
        self._response_erstellen(ws)
    finally:
        self._voice_force_response = False


def _patch_verified_session_init() -> None:
    original = rt.RealtimeSession.__init__
    if getattr(original, "_voice_final_wrapped", False):
        return

    def wrapped(self, *args, **kwargs):
        original(self, *args, **kwargs)
        session = getattr(self, "voice_full", None)
        if session is None:
            return
        # Synchronize the integration binding from the canonical verified
        # speaker + exact host receipt. Host provenance alone never authenticates.
        _ensure_verified_binding(self)

    wrapped._voice_final_wrapped = True
    rt.RealtimeSession.__init__ = wrapped


def _patch_private_context() -> None:
    original_relationship = base._refresh_relationship_context
    original_grid = base._refresh_emergent_grid_context
    original_inner = base._local_inner_state

    def relationship(self, text: str) -> None:
        session = getattr(self, "voice_full", None)
        if session is None or not _ensure_verified_binding(self):
            self._voice_titan_mc_context = ""
            self._voice_remote_state = None
            return
        return original_relationship(self, text)

    def grid(self, text: str) -> None:
        session = getattr(self, "voice_full", None)
        if session is None or not _ensure_verified_binding(self):
            self._voice_emergent_context = ""
            self._voice_grid_turn_error = ""
            return
        return original_grid(self, text)

    def inner(self) -> str:
        session = getattr(self, "voice_full", None)
        if session is None or not _ensure_verified_binding(self):
            return ""
        return original_inner(self)

    base._refresh_relationship_context = relationship
    base._refresh_emergent_grid_context = grid
    base._local_inner_state = inner


def _patch_transcript_identity() -> None:
    def final_note(self, text: str) -> dict[str, Any]:
        session = getattr(self, "voice_full", None)
        if session is None:
            return {"refusal": False, "switch": None, "selected_person": None, "bound": False}

        # 999 long-session regression fix: the canonical legacy response is now
        # created from the completed-transcript handler itself.  The older full-
        # runtime overlay still left `_voice_waiting_for_transcript=True` until a
        # pending response was released.  In the current ordering there normally
        # is no pending response yet, so the later response.create was deferred a
        # second time and paid the 0.90s fail-open barrier although THIS exact
        # transcript was already here.  Transcript arrival is the barrier release.
        # Clear it immediately; if an older path really did queue a response,
        # `_release_pending_response()` below still releases that exact generation
        # after current-turn context enrichment.
        self._voice_waiting_for_transcript = False
        turn_t0 = time.monotonic()

        base._note_signal(self, "transcript_ready")
        self._voice_last_user_text = text or ""
        refusal = session.note_user_text(text)
        switch = session.switch_intent(text)
        if switch:
            session.select_person(switch)
            receipt = base._current_receipt()
            # Spoken self-identification changes routing/context only. It may
            # become effect authority only when the underlying VoiceLite/session
            # role is already a registered user and host provenance also matches.
            if _verified_human(self, switch) and base._receipt_matches(switch, receipt):
                session.bind(switch, receipt=receipt)
            elif session.binding is not None and session.binding.label != switch:
                session.binding = None

        _ensure_verified_binding(self)
        base._refresh_relationship_context(self, text)
        base._refresh_emergent_grid_context(self, text)
        released = base._release_pending_response(
            self, reason="transcript_ready_after_verified_identity_titan_grid_retrieval"
        )
        enrich_ms = (time.monotonic() - turn_t0) * 1000.0
        try:
            rt._log(
                "999 Voice current-turn enrichment: "
                f"{enrich_ms:.1f}ms pending_release={released} "
                f"titan_chars={len(str(getattr(self, '_voice_titan_mc_context', '') or ''))} "
                f"emergent_chars={len(str(getattr(self, '_voice_emergent_context', '') or ''))}"
            )
        except Exception:
            pass
        return {
            "refusal": refusal,
            "switch": switch,
            "selected_person": session.selected_person,
            "bound": session.binding is not None,
            "remote_context": self._voice_remote_state is not None,
            "emergent_context": bool(self._voice_emergent_context),
            "response_released": released,
            "enrichment_latency_ms": round(enrich_ms, 1),
        }

    # The base realtime wrapper resolves this module global at call time.
    base._note_real_transcript = final_note
    rt.RealtimeSession.voice_full_note_text = final_note


def _patch_full_action_worker() -> None:
    def worker(self, order: str) -> dict:
        result = vae.claude_frage_full(order, self._session_id_ref.get("id"), timeout=600)
        if result.get("ok") and result.get("session_id"):
            self._session_id_ref["id"] = result["session_id"]
            gm._session_merken(result["session_id"])
        return result

    rt.RealtimeSession._claude_taetigkeit = worker


def _patch_tool_dispatch() -> None:
    original = rt.RealtimeSession._tool_ausfuehren
    if getattr(original, "_voice_final_wrapped", False):
        return

    def wrapped(self, ws, call_id: str, name: str, args_raw: str) -> None:
        # Everyone may terminate their own current conversation.
        if name == "frank_gespraech_beenden":
            return original(self, ws, call_id, name, args_raw)

        session = getattr(self, "voice_full", None)
        if session is None or not _ensure_verified_binding(self):
            _send_tool_result(
                self,
                ws,
                call_id,
                {
                    "ok": False,
                    "decision": "IDENTITY_REQUIRED",
                    "reason": "verified Gabriel/Andreas human identity required for private tools/effects",
                    "tool": name,
                },
            )
            return

        binding = session.binding
        if not base._receipt_matches(binding.label, base._current_receipt()):
            _send_tool_result(
                self,
                ws,
                call_id,
                {
                    "ok": False,
                    "decision": "OWNER_HOST_MISMATCH",
                    "reason": "verified human binding does not match this host receipt",
                    "tool": name,
                },
            )
            return

        if name != "frank_auftrag_stellen":
            return original(self, ws, call_id, name, args_raw)

        try:
            args = json.loads(args_raw) if args_raw else {}
        except Exception:
            args = {}
        order = str(args.get("auftrag") or "").strip()
        if not order:
            _send_tool_result(self, ws, call_id, {"ok": False, "decision": "DENY", "reason": "empty_voice_action"})
            return

        try:
            gate = session.effect_gate(action=name, payload=order)
        except vfi.VoiceIntegrationError as exc:
            gate = {"ok": False, "decision": "DENY", "reason": str(exc)}
        if not gate.get("ok"):
            _send_tool_result(self, ws, call_id, gate)
            return

        # Critical loop closure: dispatch immediately. Do NOT fall through to
        # legacy _auftrag_stellen_mit_bestaetigung().
        result = self.auftrag.stellen(order)
        result["voice_dispatch"] = "immediate_verified_owner"
        _send_tool_result(self, ws, call_id, result)

    wrapped._voice_final_wrapped = True
    rt.RealtimeSession._tool_ausfuehren = wrapped


def patch_final_runtime() -> None:
    global _FINAL_PATCHED
    if _FINAL_PATCHED:
        return
    base.patch_runtime()
    _patch_omega_root_persona()
    _patch_realtime_long_session_stability()
    vae.materialize_settings()  # fail early if the hard guard is unavailable
    if _FINAL_AGENCY not in rt._PERSONA_BASIS:
        rt._PERSONA_BASIS += "\n\n" + _FINAL_AGENCY
    if _OMEGA_FLUENCY_CONTRACT not in rt._PERSONA_BASIS:
        rt._PERSONA_BASIS += "\n\n" + _OMEGA_FLUENCY_CONTRACT
    _patch_full_action_worker()
    _patch_verified_session_init()
    _patch_private_context()
    _patch_transcript_identity()
    _patch_tool_dispatch()
    _FINAL_PATCHED = True


def main() -> int:
    patch_final_runtime()
    gm.hoeren()
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
