def _patch_realtime_long_session_stability() -> None:
    """Bound provider history without forking the canonical voice engine.

    The Realtime websocket is still one continuous conversation.  Only the
    provider's automatic oldest-message truncation threshold is lowered.  OMEGA's
    durable identity/memory/GRID state remains injected by the existing final
    runtime on every relevant turn, so truncation does not create another memory
    authority.
    """
    original = rt.RealtimeSession._session_config
    if getattr(original, "_voice_final_long_session_wrapped", False):
        return

    def wrapped(self, instructions: str) -> dict:
        event = original(self, instructions)
        session = event.get("session") if isinstance(event, dict) else None
        if isinstance(session, dict):
            session["truncation"] = {
                "type": "retention_ratio",
                "retention_ratio": _REALTIME_RETENTION_RATIO,
                "token_limits": {
                    "post_instructions": _REALTIME_POST_INSTRUCTION_TOKEN_LIMIT,
                },
            }
        return event

    wrapped._voice_final_long_session_wrapped = True
    rt.RealtimeSession._session_config = wrapped


def _verified_human(self, label: str) -> bool:
    role = str(getattr(self, "sprecher_rolle", "guest") or "guest").strip().lower()
    name = str(getattr(self, "sprecher_name", "") or "").strip().lower()
    return role == "user" and str(label or "").strip().lower() in name


def _ensure_verified_binding(self) -> bool:
    """Reconcile canonical verified speaker state with VoiceIntegrationSession.

    This closes a split-brain seam: host/default person selection lives in the
    integration session while speaker verification/handoff lives in the canonical
    Realtime session. Private context still requires BOTH verified human identity
    and exact machine/owner receipt; this function never upgrades a guest.
    """
    session = getattr(self, "voice_full", None)
    if session is None:
        return False

    canonical = getattr(self, "speaker_binding", None)
    owner = str(getattr(canonical, "owner", "") or "").strip().lower()
    if owner in vfi.INSTALLATIONS and _verified_human(self, owner):
        if session.selected_person != owner:
            session.select_person(owner)

    selected = str(session.selected_person or "").strip().lower()
    if not selected:
        name = str(getattr(self, "sprecher_name", "") or "").strip().lower()
        for label in vfi.INSTALLATIONS:
            if label in name and _verified_human(self, label):
                session.select_person(label)
                selected = label
                break

    if selected and _verified_human(self, selected):
        receipt = base._current_receipt()
        if base._receipt_matches(selected, receipt):
            if session.binding is None or session.binding.label != selected:
                session.bind(selected, receipt=receipt)
            return True

    if session.binding is not None:
        session.binding = None
    return False

def _patch_verified_session_init() -> None:
    original = rt.RealtimeSession.__init__
    if getattr(original, "_voice_final_wrapped", False):
        return

    def wrapped(self, *args, **kwargs):
        original(self, *args, **kwargs)
        session = getattr(self, "voice_full", None)
        if session is None:
            return
        # Synchronize the integration binding from the canonical verified
        # speaker + exact host receipt. Host provenance alone never authenticates.
        _ensure_verified_binding(self)

    wrapped._voice_final_wrapped = True
    rt.RealtimeSession.__init__ = wrapped


def _patch_private_context() -> None:
    original_relationship = base._refresh_relationship_context
    original_grid = base._refresh_emergent_grid_context
    original_inner = base._local_inner_state

    def relationship(self, text: str) -> None:
        session = getattr(self, "voice_full", None)
        if session is None or not _ensure_verified_binding(self):
            self._voice_titan_mc_context = ""
            self._voice_remote_state = None
            return
        return original_relationship(self, text)

        if session is None or not _ensure_verified_binding(self):
            self._voice_emergent_context = ""
            self._voice_grid_turn_error = ""
            return
        return original_grid(self, text)

        session = getattr(self, "voice_full", None)
        if session is None or not _ensure_verified_binding(self):
            return ""
        return original_inner(self)

    base._refresh_relationship_context = relationship
    base._refresh_emergent_grid_context = grid
    base._local_inner_state = inner
