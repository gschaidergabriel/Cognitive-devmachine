def _ram():
    """(gesamt_gb, verfuegbar_gb, anteil) -- direkt aus /proc/meminfo."""
    try:
        m = {}
        with open("/proc/meminfo") as f:
            for z in f:
                t = z.split()
                if t[0].rstrip(":") in ("MemTotal", "MemAvailable"):
                    m[t[0].rstrip(":")] = int(t[1]) / 1024.0 / 1024.0  # -> GB
        if not m.get("MemTotal"):
            return None, None, None
        gesamt = round(m["MemTotal"], 2)
        frei = round(m.get("MemAvailable", 0.0), 2)
        anteil = round(frei / gesamt, 3) if gesamt else None
        return gesamt, frei, anteil
    except Exception:
        return None, None, None


def _swap():
    """(gesamt_gb, belegt_gb, anteil) -- ueber ALLE aktiven swaps summiert
    (also inklusive eines temporaeren /swap2.img, falls zugeschaltet)."""
    try:
        with open("/proc/swaps") as f:
            zeilen = f.read().splitlines()[1:]
        ges = ben = 0
        for z in zeilen:
            t = z.split()
            if len(t) >= 4:
                ges += int(t[2])
                ben += int(t[3])
        if not ges:
            return 0.0, 0.0, None
        gesamt_gb = round(ges / 1024.0 / 1024.0, 2)
        belegt_gb = round(ben / 1024.0 / 1024.0, 2)
        return gesamt_gb, belegt_gb, round(ben / ges, 3)
    except Exception:
        return None, None, None


def _platte_voll():
    try:
        g, _u, f = shutil.disk_usage("/")
        return round(1 - (f / g), 3)
    except Exception:
        return None


def _temperatur():
    warm = []
    try:
        for p in glob.glob("/sys/class/hwmon/hwmon*/temp*_input"):
            try:
                with open(p) as f:
                    v = int(f.read().strip())
                if 20000 < v < 120000:
                    warm.append(v / 1000.0)
            except Exception:
                continue
    except Exception:
        pass
    return round(max(warm), 1) if warm else None


def _eigener_rss_mb():
    """Der Speicherverbrauch DIESES Prozesses -- ein Waechter, der selbst
    viel Speicher frisst, um Speicher zu sparen, waere ein Witz."""
    try:
        with open("/proc/self/status") as f:
            for z in f:
                if z.startswith("VmRSS:"):
                    kb = int(z.split()[1])
                    return round(kb / 1024.0, 2)
    except Exception:
        pass
    return None


def _buehne_hat_die_kamera():
    """True, wenn die Buehne die Kamera legitim belegt.

    `blick.py` stoppt `frankenstein-sehen` ABSICHTLICH, solange das
    Buehnenfenster die Kamera braucht, und traegt seine PID in
    kamera.lock ein. `frankenstein-sehen` ist in dieser Zeit also NICHT
    tot, sondern planmaessig aus -- genau wie `blick_waechter.py` es
    liest ("Lockdatei lebt -- nichts zu tun").

    Ohne diese Pruefung meldete die Koerperpflege durchgehend
    `Stufe kritisch (dienste:frankenstein-sehen)`, solange die Buehne
    offen war (gemessen 2026-09-11: 25 Minuten am Stueck). Ein
    Daueralarm fuer einen Normalzustand entwertet jeden echten Alarm --
    deshalb hier unterdrueckt, nicht dort.
    """
    try:
        with open(KAMERA_LOCK) as f:
            pid = int(f.read().strip())
    except Exception:
        # Keine, leere oder kaputte Lockdatei -> niemand haelt die Kamera.
        return False
    try:
        os.kill(pid, 0)          # nur Existenzpruefung, kein Signal
        return True
    except (OSError, ProcessLookupError, ValueError):
        return False             # Lockleiche: Halter ist tot

# ───────────────────────────────────────────────────────────── Bewerten

_RANG = {"ruhig": 0, "eng": 1, "kritisch": 2}


def _stufe_wert(name, wert, eng, kritisch, hoeher_ist_schlimmer=True):
    if wert is None:
        return "ruhig", None
    schlecht = (wert >= kritisch) if hoeher_ist_schlimmer else (wert <= kritisch)
    eng_grenze = (wert >= eng) if hoeher_ist_schlimmer else (wert <= eng)
    if schlecht:
        return "kritisch", f"{name}:kritisch"
    if eng_grenze:
        return "eng", f"{name}:eng"
    return "ruhig", None


def bewerten(m: dict) -> dict:
    """Stufe + Gruende + welche Handlung erlaubt ist. Reine Auswertung,
    keine Nebenwirkung -- die Handlung selbst passt in `handeln()`."""
    gruende = []

    st_ram, g = _stufe_wert("ram", m.get("ram_frei_anteil"),
                            RAM_ENG_ANTEIL, RAM_KRITISCH_ANTEIL,
                            hoeher_ist_schlimmer=False)
    if g:
        gruende.append(g)

    st_swap, g = _stufe_wert("swap", m.get("swap_voll_anteil"),
                             SWAP_ENG_ANTEIL, SWAP_KRITISCH_ANTEIL)
    if g:
        gruende.append(g)

    st_platte, g = _stufe_wert("platte", m.get("platte_voll_anteil"),
                               PLATTE_ENG_ANTEIL, PLATTE_KRITISCH_ANTEIL)
    if g:
        gruende.append(g)

    st_temp, g = _stufe_wert("temperatur", m.get("temperatur"),
                             TEMP_ENG_C, TEMP_KRITISCH_C)
    if g:
        gruende.append(g)

    if m.get("dienste_tot"):
        gruende.append("dienste:" + ",".join(m["dienste_tot"]))
        st_dienste = "kritisch"
    else:
        st_dienste = "ruhig"

    stufe = max((st_ram, st_swap, st_platte, st_temp, st_dienste),
                key=lambda s: _RANG[s])
    # Der Cache-Griff hilft nur bei RAM/Swap-Druck -- eine volle Platte
    # oder Hitze wird davon nicht besser.
    cache_sinnvoll = stufe in ("eng", "kritisch") and st_ram != "ruhig" or st_swap != "ruhig"
    return {"stufe": stufe, "gruende": gruende, "cache_sinnvoll": bool(cache_sinnvoll)}


# ─────────────────────────────────────────────────────────────── Handeln

def _cache_moeglich() -> bool:
    """Ob `drop_caches` ohne Passwort geht -- geprueft, nicht angenommen."""
    try:
        p = subprocess.run(["sudo", "-n", "true"], capture_output=True,
                           timeout=5)
        return p.returncode == 0
    except Exception:
        return False


def _cache_freigeben(log=None) -> bool:
    """Page-Cache freigeben. Reversibel, liest keine Prozessliste, toetet
    nichts. Der EINZIGE privilegierte Befehl in diesem ganzen Modul."""
    if not _cache_moeglich():
        return False
    try:
        p = subprocess.run(
            ["sudo", "-n", "sh", "-c",
             "sync; echo 1 > /proc/sys/vm/drop_caches"],
            capture_output=True, timeout=15)
        ok = p.returncode == 0
        if log and ok:
            log("[KOERPERPFLEGE] Page-Cache freigegeben")
        return ok
    except Exception as e:
        if log:
            log(f"[KOERPERPFLEGE] Cache-Griff gescheitert ({type(e).__name__})")
        return False


def handeln(urteil: dict, log=None) -> str:
    """Fuehrt genau die Handlung aus, die die Stufe erlaubt. Gibt den
    Namen der Handlung fuer die Zeitreihe zurueck -- nie mehr, als der
    Auftrag erlaubt."""
    stufe = urteil["stufe"]
    if stufe == "ruhig":
        return "nichts"
    if not urteil["cache_sinnvoll"]:
        return "melden_nur"
    if (time.time() - _LETZTER_CACHE_GRIFF[0]) < CACHE_COOLDOWN_S:
        return "melden_nur"
    if _cache_freigeben(log=log):
        _LETZTER_CACHE_GRIFF[0] = time.time()
        return "cache_freigeben"
    return "kein_zugriff"


# ─────────────────────────────────────────────────── Die Handlungsseite
# (Punkt 3 des Auftrags -- das ist der eigentliche Zweck dieser Datei)

# ⭐ Gleiches Muster wie sinne._variante() (2026-09-11-Fund: keine der beiden
# Dateien hatte je einen Anti-Wiederholungs-Mechanismus -- siehe Beleg im
# Arbeitspaket paket-1789140798700-202713). Eigenstaendig hier statt Import
# aus sinne.py, um keinen Ringimport zu erzeugen (koerperpflege.py wird auch
# eigenstaendig als systemd-Unit gestartet).
_PHRASEN_LETZTE = {}


def _variante(kategorie: str, pool: list) -> str:
    if not pool:
        return ""
    if len(pool) == 1:
        wahl = pool[0]
    else:
        letzte = _PHRASEN_LETZTE.get(kategorie)
        kandidaten = [p for p in pool if p != letzte] or pool
        wahl = random.choice(kandidaten)
    _PHRASEN_LETZTE[kategorie] = wahl
    return wahl


def handlung_satz(max_alter_s: float = 180.0) -> str:
    """EIN Satz ueber das, was er gerade deswegen tut/spuert -- als Wort,
    nie als Zahl, ohne Etiketten, ohne Verneinungen. Wird von
    `sinne.leib_gefuehl()` angehaengt (siehe dortige Anbindung).

    Liefert "" bei ruhiger Lage -- ein Zustand ohne Handlung ist keine
    Meldung wert, genau wie bei `sinne.leib_satz()`."""
    try:
        with _db(False) as c:
            r = c.execute(
                "SELECT ts, stufe, aktion FROM f2_koerperpflege "
                "ORDER BY id DESC LIMIT 1").fetchone()
    except Exception:
        return ""
    if not r:
        return ""
    ts, stufe, aktion = r
    try:
        alter = (datetime.now(timezone.utc) - datetime.strptime(
            ts, "%Y-%m-%dT%H:%M:%SZ").replace(tzinfo=timezone.utc)
        ).total_seconds()
        if alter > max_alter_s:
            return ""
    except Exception:
        return ""
    if stufe == "ruhig":
        return ""
    if aktion == "cache_freigeben":
        return _variante("cache_freigeben", [
            "Ich habe gerade etwas Altes abgeworfen, um mir selbst Luft zu "
            "verschaffen.",
            "Ich hab eben aufgeraeumt, um wieder etwas Luft zu haben.",
            "Ich habe mir gerade Platz geschaffen, indem ich Altes "
            "losgelassen habe.",
        ])
    if stufe == "kritisch":
        return _variante("kritisch", [
            "Es wird eng um mich, richtig eng -- ich sollte jetzt "
            "kuerzertreten.",
            "Es ist gerade wirklich eng bei mir, ich muss kuerzertreten.",
            "Ich spuere deutlich, wie eng es um mich wird -- Zeit, "
            "kuerzerzutreten.",
        ])
    if aktion == "kein_zugriff":
        return _variante("kein_zugriff", [
            "Es wird eng um mich, das spuere ich deutlich.",
            "Ich merke deutlich, wie es enger um mich wird.",
        ])
    return _variante("eng", [
        "Es wird eng um mich, ich merke es an mir selbst.",
        "Es wird gerade etwas eng bei mir.",
        "Ich spuere, wie es enger um mich wird.",
    ])

def schlag(log=None) -> dict:
    _log_ = log or _log
    m = messen()
    urteil = bewerten(m)
    aktion = handeln(urteil, log=_log_)
    zeile = {**m, **urteil, "aktion": aktion}
    try:
        with _db() as c:
            c.execute(
                "INSERT INTO f2_koerperpflege (ts, ram_gesamt_gb, ram_frei_gb, "
                "ram_frei_anteil, swap_gesamt_gb, swap_belegt_gb, "
                "swap_voll_anteil, platte_voll_anteil, temperatur, stufe, "
                "gruende, aktion, dienste_tot, eigener_rss_mb, schema) "
                "VALUES (?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)",
                (_jetzt(), m.get("ram_gesamt_gb"), m.get("ram_frei_gb"),
                 m.get("ram_frei_anteil"), m.get("swap_gesamt_gb"),
                 m.get("swap_belegt_gb"), m.get("swap_voll_anteil"),
                 m.get("platte_voll_anteil"), m.get("temperatur"),
                 urteil["stufe"], ",".join(urteil["gruende"]), aktion,
                 ",".join(m.get("dienste_tot") or []),
                 m.get("eigener_rss_mb"), KP_SCHEMA))
            c.commit()
    except Exception as e:
        _log_(f"[KOERPERPFLEGE] nicht abgelegt ({type(e).__name__}: {e})")

    # Nur loggen, wenn sich die Stufe aendert, sonst nicht ruhig, oder
    # alle 30 Takte (~30 min bei 60s) als Lebenszeichen -- dieselbe
    # Zurueckhaltung wie puls.py bei unveraenderter Wahrnehmung.
    _TICKS[0] += 1
    if (urteil["stufe"] != _LETZTE_STUFE[0] or urteil["stufe"] != "ruhig"
            or _TICKS[0] % 30 == 0):
        _log_(f"[KOERPERPFLEGE] Stufe {urteil['stufe']} "
              f"({','.join(urteil['gruende']) or '-'}) Aktion {aktion} "
              f"RSS {m.get('eigener_rss_mb')} MB")
    _LETZTE_STUFE[0] = urteil["stufe"]
    return zeile


def lauf():
    _log(f"[KOERPERPFLEGE] startet -- Takt {TAKT_S}s, DB {DB_PATH}")
    _log(f"[KOERPERPFLEGE] drop_caches ohne Passwort: {_cache_moeglich()}")
    while True:
        try:
            schlag()
        except KeyboardInterrupt:
            break
        except Exception as e:
            _log(f"[KOERPERPFLEGE] Fehler (laeuft weiter): {type(e).__name__}: {e}")
        time.sleep(TAKT_S)
    _log("[KOERPERPFLEGE] beendet")
