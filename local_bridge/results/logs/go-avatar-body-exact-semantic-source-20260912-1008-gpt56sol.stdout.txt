function koerperAktionStarten(fertig){
  if(koerperAktion || _sonderLaeuft){ if(fertig) fertig(); return; }
  const pool = KOERPER_AKTIONEN.filter(a => a.name !== _letzteKoerperAktion);
  const wahl = pool[Math.floor(Math.random()*pool.length)];
  _letzteKoerperAktion = wahl.name;

}
function koerperAktionWeiterPlanen(){
  // 22-50s Ruhe zwischen zwei Koerperaktionen -- selten, unregelmaessig,

  setTimeout(() => koerperAktionStarten(() => koerperAktionWeiterPlanen()),

function koerperSchritt(jetzt){
  if(koerperAktion){
    const a = koerperAktion;
    if(jetzt - a.letzterT < a.ms) return;
    const eintrag = bildLaden(a.bilder[a.idx]);
    if(!eintrag.ok) return; // Frame noch nicht dekodiert -- warten statt Flash
    a.letzterT = jetzt;
    koerperCrossfade(a.bilder[a.idx]);
    a.idx++;
    if(a.idx >= a.bilder.length){
      koerperAktion = null;
      const f = a.fertig; a.fertig = null;
      if(f) f();
    }
    return;
  }
  if(jetzt - atmenLetzterWechsel < ATMEN_MS_PRO_BILD) return;
  atmenLetzterWechsel = jetzt;
  atmenIndex = (atmenIndex + 1) % ATMEN_BILDER.length;
  koerperCrossfade(ATMEN_BILDER[atmenIndex]);
}
