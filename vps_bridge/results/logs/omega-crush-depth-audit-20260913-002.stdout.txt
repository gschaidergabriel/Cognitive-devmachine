The product boundary is the Crush runtime, not a VM. Crush remains the stable portable shell for UI, CLI, tools, MCP, permissions, providers and session presentation. OMEGA lives below that surface and above model/tool execution.

Phase 1 intentionally keeps the working Frankenstein 2.0 execution topology: ChatGPT through the existing ChatGPT bridge, Claude Code through the existing Claude/CLI path, Free Pool through the existing Crush provider harness, plus the currently proven Local/VPS bridge and receipt mechanisms.

Portable cognitive core: GRID10 competition and broadcast, TITAN MX state dynamics, Unified persistent state, LCME/dynamic cognitive compilation, GWT/reentry, identity, goals/hypotheses/world-model state, module competition and effect gating.

Host-specific bridges and launchers are adapters around the core. The core must not hard-code Gabriel- or Andreas-specific paths, machine IDs, display devices, or one provider.

Current integration seam: internal/omega.Runtime.Compile() executes before model inference and produces a bounded deterministic cognitive frame; Reenter() receives the terminal outcome before Crush publishes run completion. A nil runtime maps to NoopRuntime, preserving upstream behavior.

Migration law: keep existing bridge/provider/state-sync behavior authoritative until an equivalent native internal/omega path has executable tests and readback/receipt evidence. Replace mechanisms incrementally, never by creating parallel authorities.

Development law: on every explicit CRUSH trigger refresh current OMEGA/EntityOS architecture, Gabriel's working Crush harness, current ChatGPT/Claude/Free-Pool bridge/provider state, and this project before porting the strongest nonduplicate semantic delta.

// Package omega defines the portable cognitive runtime seam used by EntityOS.
//
// It intentionally depends only on the Go standard library. The Crush shell
// remains responsible for UI, providers, tools, MCP, permissions, and session
// presentation. OMEGA owns cognitive state formation and reentry.
package omega

import (
	"context"
	"fmt"
	"sort"
	"strings"
)

// Version identifies the first portable EntityOS/OMEGA frame contract.
const Version = "ENTITYOS_OMEGA_FRAME/v1"

// Candidate is one contribution competing for inclusion in the current
// cognitive frame. Scores are deliberately explicit rather than hidden in
// prompt prose so GRID can later replace the baseline ordering without
// changing the Crush-facing seam.
type Candidate struct {
	ID         string
	Kind       string
	Source     string
	Payload    string
	Salience   float64
	Confidence float64
	Urgency    float64
}

// TurnInput is the minimal observation OMEGA receives from the harness.
// It contains no host-specific paths, services, or provider assumptions.
type TurnInput struct {
	SessionID string
	RunID     string
	Prompt    string
}

// Frame is the bounded cognitive state presented to the current model organ.
// StateDigest binds the frame to the persistent state it was compiled from.
type Frame struct {
	Version       string
	TurnID        string
	StateDigest   string
	Candidates    []Candidate
	SystemContext string
}

// TurnOutcome is the terminal observation fed back into OMEGA for reentry.
type TurnOutcome struct {
	SessionID string
	RunID     string
	Text      string
	Err       string
	Cancelled bool
}

// Runtime is the only cognitive contract the Crush agent loop needs.
//
// Compile runs before model inference. Reenter runs once at terminal turn
// completion. Implementations may persist state, run GRID competition, TITAN
// consolidation, LCME transforms, or future cognitive modules behind this
// interface without leaking those concerns into Crush UI/provider/tool code.
type Runtime interface {
	Compile(context.Context, TurnInput) (Frame, error)
	Reenter(context.Context, Frame, TurnOutcome) error
}

// NoopRuntime preserves upstream Crush behavior exactly when no EntityOS
// cognitive runtime has been configured.
type NoopRuntime struct{}

func (NoopRuntime) Compile(context.Context, TurnInput) (Frame, error) {
	return Frame{Version: Version}, nil
}

func (NoopRuntime) Reenter(context.Context, Frame, TurnOutcome) error {
	return nil
}

// RenderSystemContext deterministically serializes the bounded winning
// coalition for model injection. It never serializes the entire persistent
// state. Stable sorting makes identical state/input produce identical frame
// text, independent of candidate production order.
func (f Frame) RenderSystemContext() string {
	if strings.TrimSpace(f.SystemContext) == "" && len(f.Candidates) == 0 {
		return ""
	}

	candidates := append([]Candidate(nil), f.Candidates...)
	sort.SliceStable(candidates, func(i, j int) bool {
		li := candidateUtility(candidates[i])
		lj := candidateUtility(candidates[j])
		if li != lj {
			return li > lj
		}
		return candidates[i].ID < candidates[j].ID
	})

	var b strings.Builder
	b.WriteString("<omega-cognitive-frame version=\"")
	if f.Version == "" {
		b.WriteString(Version)
	} else {
		b.WriteString(escapeAttr(f.Version))
	}
	b.WriteString("\"")
	if f.TurnID != "" {
		fmt.Fprintf(&b, " turn=\"%s\"", escapeAttr(f.TurnID))
	}
	if f.StateDigest != "" {
		fmt.Fprintf(&b, " state=\"%s\"", escapeAttr(f.StateDigest))
	}
	b.WriteString(">\n")

	if s := strings.TrimSpace(f.SystemContext); s != "" {
		b.WriteString("<broadcast>\n")
		b.WriteString(s)
		b.WriteString("\n</broadcast>\n")
	}
	for _, c := range candidates {
		if strings.TrimSpace(c.Payload) == "" {
			continue
		}
		fmt.Fprintf(
			&b,
			"<candidate id=\"%s\" kind=\"%s\" source=\"%s\" salience=\"%.6f\" confidence=\"%.6f\" urgency=\"%.6f\">%s</candidate>\n",
			escapeAttr(c.ID),
			escapeAttr(c.Kind),
			escapeAttr(c.Source),
			c.Salience,
			c.Confidence,
			c.Urgency,
			strings.TrimSpace(c.Payload),
		)
	}
	b.WriteString("</omega-cognitive-frame>")
	return b.String()
}

func candidateUtility(c Candidate) float64 {
	// Baseline only. GRID will own competition in the real runtime.
	return c.Salience * c.Confidence * (1 + c.Urgency)
}

func escapeAttr(s string) string {
	r := strings.NewReplacer(
		"&", "&amp;",
		"\"", "&quot;",
		"<", "&lt;",
		">", "&gt;",
	)
	return r.Replace(s)
}

package omega

import (
	"context"
	"strings"
	"testing"
)

func TestNoopRuntimePreservesUpstreamPrompt(t *testing.T) {
	rt := NoopRuntime{}
	frame, err := rt.Compile(context.Background(), TurnInput{
		SessionID: "s1",
		RunID:     "r1",
		Prompt:    "hello",
	})
	if err != nil {
		t.Fatal(err)
	}
	if got := frame.RenderSystemContext(); got != "" {
		t.Fatalf("noop runtime injected context: %q", got)
	}
	if err := rt.Reenter(context.Background(), frame, TurnOutcome{}); err != nil {
		t.Fatal(err)
	}
}

func TestFrameRenderIsDeterministicAcrossProductionOrder(t *testing.T) {
	a := Candidate{ID: "a", Kind: "hypothesis", Source: "grid", Payload: "A", Salience: .8, Confidence: .9, Urgency: .1}
	b := Candidate{ID: "b", Kind: "goal", Source: "titan", Payload: "B", Salience: .4, Confidence: .9, Urgency: .2}

	f1 := Frame{Version: Version, TurnID: "t1", StateDigest: "deadbeef", Candidates: []Candidate{a, b}}
	f2 := Frame{Version: Version, TurnID: "t1", StateDigest: "deadbeef", Candidates: []Candidate{b, a}}

	got1 := f1.RenderSystemContext()
	got2 := f2.RenderSystemContext()
	if got1 != got2 {
		t.Fatalf("render depends on candidate order:\n%s\n---\n%s", got1, got2)
	}
	if strings.Index(got1, `id="a"`) > strings.Index(got1, `id="b"`) {
		t.Fatalf("higher-utility candidate did not win ordering:\n%s", got1)
	}
}

func TestFrameDoesNotDumpAbsentState(t *testing.T) {
	got := (Frame{
		Version:       Version,
		SystemContext: "only-current-broadcast",
	}).RenderSystemContext()
	if !strings.Contains(got, "only-current-broadcast") {
		t.Fatalf("missing current broadcast: %s", got)
	}
	if strings.Contains(got, "memory") || strings.Contains(got, "history") {
		t.Fatalf("unexpected implicit memory/history dump: %s", got)
	}
}
