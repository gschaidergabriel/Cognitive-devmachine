def _konfig_lesen(schluessel, default=None):
    """Liest EINEN star_konfig-Wert aus der kanonischen unified.db. Fail-closed:
    jeder Fehler (DB gesperrt, Tabelle fehlt) gibt den uebergebenen default
    zurueck, wirft nie."""
    try:
        with contextlib.closing(sqlite3.connect(memory_abruf.DB_PATH, timeout=5.0)) as c:
            row = c.execute(
                "SELECT wert FROM star_konfig WHERE schluessel = ?", (schluessel,)
            ).fetchone()
            return row[0] if row is not None else default
    except Exception as e:
        _log(f"[KONFIG] FEHLER beim Lesen von {schluessel}: {type(e).__name__}: {e}")
        return default


def _konfig_schreiben(schluessel, wert, quelle):
    """Schreibt EINEN star_konfig-Wert in dieselbe kanonische unified.db (kein
    zweiter Zustandspfad). Nie ein Fehler nach aussen -- nur geloggt."""
    try:
        with contextlib.closing(sqlite3.connect(memory_abruf.DB_PATH, timeout=8.0)) as c:

                "INSERT INTO star_konfig (schluessel, wert, typ, quelle, geaendert) "
                "VALUES (?, ?, 'str', ?, ?) "
                "ON CONFLICT(schluessel) DO UPDATE SET wert=excluded.wert, "
                "quelle=excluded.quelle, geaendert=excluded.geaendert",
                (schluessel, str(wert), quelle, time.time()),
