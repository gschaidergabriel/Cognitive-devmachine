// ⭐⭐ 2026-09-11 #kopf-canvas exakt auf die reale object-fit:cover-Geometrie
// bringen, die #koerper-schicht/#kopf-schicht vorher PER CSS bekamen (cover,
// object-position:50% 15%). Cover-Formel: scale = max(containerW/bildW,
// containerH/bildH); die Leinwand wird auf renderedW x renderedH gesetzt und
// so weit verschoben, dass an der object-position-Stelle (50%/15%) kein
// Rand entsteht -- exakt die Formel, die der Browser intern fuer
// object-fit:cover verwendet. #gesicht-rahmen beschneidet den Ueberstand per
// overflow:hidden -- optisch identisch zum alten object-fit:cover, aber
// jetzt kennt jedes Kind (koerper/kopf/mund) dieselbe 1536x1024-Leinwand und
// einfache Prozentzahlen (siehe #mund-effekt) treffen den Manifest-Anker
// exakt. offsetWidth/offsetHeight statt getBoundingClientRect -- die sind
// transform-unabhaengig, der Cam-Shake auf #kamera soll hier NICHT
// mitgemessen werden (nur einmal bei Start/Resize aufgerufen, nicht pro
// Frame -- die Leinwand selbst ist statisch, der Shake wirkt automatisch
// weiter oben auf #kamera und bewegt die ganze Szene inklusive Leinwand mit).
const BILD_W = 1536, BILD_H = 1024, OBJ_POS_X = 0.50, OBJ_POS_Y = 0.15;
let _kopfCanvasScale = 1; // aktuelle Skala Leinwand-px -> CSS-px, siehe kopfCanvasAnpassen()
function kopfCanvasAnpassen(){
  const rahmen = $("gesicht-rahmen");
  const cw = rahmen.offsetWidth, ch = rahmen.offsetHeight;
  if(!cw || !ch) return; // noch nicht gelayoutet -- naechster Aufruf greift
  const scale = Math.max(cw / BILD_W, ch / BILD_H);
  _kopfCanvasScale = scale;
  const rw = BILD_W * scale, rh = BILD_H * scale;
  const ox = (cw - rw) * OBJ_POS_X;
  const oy = (ch - rh) * OBJ_POS_Y;
  const kc = $("kopf-canvas");
  kc.style.left = ox.toFixed(2) + "px";
  kc.style.top = oy.toFixed(2) + "px";
  kc.style.width = rw.toFixed(2) + "px";
  kc.style.height = rh.toFixed(2) + "px";
}
window.addEventListener("resize", kopfCanvasAnpassen);
kopfCanvasAnpassen();
