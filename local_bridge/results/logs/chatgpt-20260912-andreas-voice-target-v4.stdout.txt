# ⚠️ NACHSICHTIGE IMPORTE fuer alles, was spaeter dazukam. Am 2026-09-09
# starb Andreas' kompletter Sprachmodus an einem ModuleNotFoundError, weil
# ich sehen.py bei ihm nicht eingespielt hatte -- ein Import, der eine
# ZUSATZfaehigkeit holt, darf nie das Sprechen verhindern. Fehlt eines
# dieser Module, faellt genau seine Faehigkeit weg, sonst nichts.
class _Fehlt:
    """Platzhalter fuer ein fehlendes Zusatzmodul.

    ⚠️ DIE RUECKFALLWERTE MUESSEN ZUR SCHNITTSTELLE PASSEN, sonst tauscht man
    einen Absturz gegen einen anderen: `konfabulation.pruefen()` gibt ein
    PAAR zurueck, und ein blindes None waere beim Entpacken sofort ein
    TypeError. Und die Richtung des Rueckfalls ist eine Entscheidung: der
    fehlende Wachhund laesst DURCH (True), die fehlende Zuwendung laesst
    ALLES herein. Ein ausgefallener Filter darf nie strenger sein als der
    vorhandene -- sonst verstummt er, und stumm war heute schon einmal das
    schlechtere Ergebnis."""

    _WERTE = {
        "pruefen": lambda *a, **k: (True, None),      # Wachhund: durchlassen
        "zuwenden": lambda quellen=None, *a, **k: list(quellen or []),
        "faehigkeiten": lambda *a, **k: "",
        "sichtblock": lambda *a, **k: "",
        "ist_meinungsfrage": lambda *a, **k: None,
        "reflektieren": lambda *a, **k: "",
        "buehne": lambda *a, **k: {},
        "sprechbar": lambda t="", *a, **k: t,
        "erkennen_reflex": lambda *a, **k: None,
        "als_befehl": lambda satz="", laeuft=False, *a, **k: (
            __import__("abspieler").absicht(satz, laeuft=laeuft)),
        "erleben": lambda *a, **k: {},
        "block": lambda *a, **k: "",
        "juengste_gedanken": lambda *a, **k: "",
        "ERSATZ": "",
    }

    def __getattr__(self, name):
        wert = self._WERTE.get(name)
        if wert is not None:
            return wert
        return lambda *a, **k: None
