const BILD_W = 1536, BILD_H = 1024, OBJ_POS_X = 0.50, OBJ_POS_Y = 0.15;
let _kopfCanvasScale = 1; // aktuelle Skala Leinwand-px -> CSS-px, siehe kopfCanvasAnpassen()
function kopfCanvasAnpassen(){
  const rahmen = $("gesicht-rahmen");
  const cw = rahmen.offsetWidth, ch = rahmen.offsetHeight;
  if(!cw || !ch) return; // noch nicht gelayoutet -- naechster Aufruf greift
  const scale = Math.max(cw / BILD_W, ch / BILD_H);
  _kopfCanvasScale = scale;
  const rw = BILD_W * scale, rh = BILD_H * scale;
  const ox = (cw - rw) * OBJ_POS_X;
  const oy = (ch - rh) * OBJ_POS_Y;
  const kc = $("kopf-canvas");
  kc.style.left = ox.toFixed(2) + "px";
  kc.style.top = oy.toFixed(2) + "px";
  kc.style.width = rw.toFixed(2) + "px";
  kc.style.height = rh.toFixed(2) + "px";
}
window.addEventListener("resize", kopfCanvasAnpassen);
kopfCanvasAnpassen();

    x: Math.random() * canvasHinten.width, y: -10,

  });
}
function regenZeichnenEbene(c, cv, liste, alpha, breite){
  c.clearRect(0,0,cv.width, cv.height);
  c.lineWidth = breite;
  for(const t of liste){
    c.strokeStyle = `rgba(${t.ton},${alpha})`;
    c.beginPath();
    c.moveTo(t.x, t.y);
    c.lineTo(t.x, t.y + t.len);
    c.stroke();
    t.y += t.v * 3.2;
  }
}
