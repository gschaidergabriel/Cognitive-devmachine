  const wechsel = (neu, mitSchwung) => {
    const sichtbar = ausdruckAktiv === 1 ? $("ausdruck-schicht") : $("ausdruck-schicht2");

    // Ebene braucht den brandaktuellen Wert deshalb hier einmalig.

    versteckt.style.filter = _atmenFilter;
    versteckt.style.opacity = "1";
    sichtbar.style.opacity = "0";
    ausdruckAktiv = ausdruckAktiv === 1 ? 2 : 1;
    if(mitSchwung){
      const rahmen = $("gesicht-rahmen");
      rahmen.classList.remove("pose-wechsel");
      requestAnimationFrame(() => rahmen.classList.add("pose-wechsel"));
      setTimeout(() => rahmen.classList.remove("pose-wechsel"), 420);
    }
  };
  const sauber = () => {
    $("ausdruck-schicht").style.opacity = "0";
    $("ausdruck-schicht2").style.opacity = "0";
    $("mund-effekt").style.opacity = "1";
    if(!mundAktivFrames) $("mund-effekt").src = `/mund_keyframes/${MUND_RUHE}.png`;
    _sonderLaeuft = false;
    gesichtNeuZeichnen(true);
  };
  let stufe = 0;
  wechsel(stufeSrc(0), true);
  const hoch = () => {
    stufe++;
    wechsel(stufeSrc(stufe), false);

}
