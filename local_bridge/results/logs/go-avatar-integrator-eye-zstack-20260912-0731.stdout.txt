function gazeZielBestimmen(){
  if(!gazeBasisSichtbar()) return {x:0.5, y:0.5, opazitaet:0};
  const wegAktiv = blickPauseAktiv || gespraechsFokus === "thema";
  if(wegAktiv) return {x:_wegZiel.x, y:_wegZiel.y, opazitaet:0.4};
  return {x:serverBlick.cx, y:serverBlick.cy, opazitaet:1};
}
function gazeSchritt(){
  const ziel = gazeZielBestimmen();
  zielGazeX = ziel.x; zielGazeY = ziel.y;
  aktGazeX += (zielGazeX - aktGazeX) * 0.07;
  aktGazeY += (zielGazeY - aktGazeY) * 0.07;
  const maxPx = 8.5; // Bewegungsspielraum bewusst klein -- ein Glimmer, kein Springen
  const dx = (aktGazeX - 0.5) * 2 * maxPx + mikroSakX;
  const dy = (aktGazeY - 0.5) * 2 * maxPx + BLICK_ABWAERTS_BIAS_PX + mikroSakY;
  // ⭐⭐ 2026-09-11 NACHTRAG 4: rotate() dranhaengen (Rest-Ausgleich zur echten
