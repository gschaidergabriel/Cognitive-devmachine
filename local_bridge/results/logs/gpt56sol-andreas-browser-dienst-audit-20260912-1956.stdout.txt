# ── Token ────────────────────────────────────────────────────────────────

def token_laden_oder_erzeugen() -> str:
    """Token aus KONFIG_DIR lesen, sonst neu erzeugen. Datei bleibt 0600."""
    if TOKEN_DATEI.is_file():
        wert = TOKEN_DATEI.read_text(encoding="utf-8").strip()
        if wert:
            return wert
    KONFIG_DIR.mkdir(parents=True, exist_ok=True)
    wert = secrets.token_urlsafe(32)
    TOKEN_DATEI.write_text(wert + "\n", encoding="utf-8")
    os.chmod(TOKEN_DATEI, 0o600)
    return wert


def token_pruefen(erwartet: str, anfrage) -> bool:
    """Zeitkonstanter Vergleich -- siehe Moduldocstring, Punkt 2.

    ⚠️ GEMESSEN 2026-08-21 (beide Rechner, Gabriel UND Andreas): der Aufrufer
    ist clayverse/browser_fern.py -- das schickt den Token als rohen Header
    `X-Zero-Token: <token>`, NICHT als `Authorization: Bearer <token>`. Der
    urspruengliche Entwurf hier pruefte NUR Authorization -- das lief also nie
    scharf, auf keiner der beiden Maschinen (401 bei jedem echten Aufruf ueber
    die Bruecke, nicht nur bei curl-Handtests ohne Header). clayverse/ ist
    NICHT mein Bereich (gehoert dem Koordinator) -- also wird hier akzeptiert,
    was die Bruecke tatsaechlich schickt, plus Authorization:Bearer bleibt
    zusaetzlich gueltig fuer manuelle Tests/curl.
    """
    kandidat = anfrage.headers.get("X-Zero-Token")
    if not kandidat:
        auth = anfrage.headers.get("Authorization")
        if auth and auth.startswith("Bearer "):
            kandidat = auth[len("Bearer "):]
    if not kandidat:
        return False
    return hmac.compare_digest(kandidat, erwartet)


# ── Bind-Adresse gegen Tailscale pruefen ────────────────────────────────

def tailscale_ips() -> list[str]:
    try:
        cp = subprocess.run(["tailscale", "ip"], capture_output=True, text=True, timeout=5.0)
    except (OSError, subprocess.TimeoutExpired):
        return []
    if cp.returncode != 0:
        return []
    return [z.strip() for z in cp.stdout.splitlines() if z.strip()]


def bind_adresse_pruefen(adresse: str) -> None:
    verboten = {"0.0.0.0", "::", "*"}
    if adresse in verboten:
        raise SicherheitsFehler(
            f"Bind an {adresse!r} verweigert: das waere jede Schnittstelle, "
            "nicht nur Tailscale."
        )
    ips = tailscale_ips()
    if not ips:
        raise SicherheitsFehler(
            "`tailscale ip` lieferte keine Adresse -- ohne Beleg, dass "
            f"{adresse!r} eine Tailscale-IP dieser Maschine ist, wird nicht gebunden."
        )
    if adresse not in ips:
        raise SicherheitsFehler(
            f"Bind an {adresse!r} verweigert: keine der Tailscale-IPs dieser "
            f"Maschine ({ips}). Nur Tailscale-Erreichbarkeit ist erlaubt."
        )


def _cdp_erreichbar(port: int) -> bool:
    import urllib.request
    try:
        with urllib.request.urlopen(f"http://127.0.0.1:{port}/json/version", timeout=1.5):
            return True
    except Exception:
        return False

# ── EIN gemeinsamer Playwright-Treiber fuer den ganzen Dienst ────────────
# ⚠️ GEMESSEN 2026-08-21: mehrere UNABHAENGIGE `sync_playwright().start()`-
# Aufrufe (einer je Sitzung) im selben Prozess/Thread scheitern mit

# fuer alle drei Browser (chromium() UND firefox() haengen beide am selben

_PW_LOCK = threading.Lock()
_PW = None


def _pw_instance():
    global _PW
    with _PW_LOCK:
        if _PW is None:
            from playwright.sync_api import sync_playwright
            _PW = sync_playwright().start()
        return _PW


# ── Chromium-Sitzung (Chrome ODER Brave -- identischer CDP-Weg) ──────────

class ChromiumSitzung:
    def __init__(self, schluessel: str):
        assert schluessel in ("chrome", "brave")
        self.schluessel = schluessel
        self.profil = _profil_pfad(schluessel)
        self.cdp_port = CDP_PORTS[schluessel]
        self._lock = threading.RLock()
        self._pw = None
        self._browser = None
        self._page = None
        self._prozess = None  # siehe schliessen(): Playwright.close() reicht bei
                               # einer CDP-ANGEHEFTETEN Sitzung nicht aus.

    def _cdp_erreichbar(self) -> bool:
        return _cdp_erreichbar(self.cdp_port)

    def _binary_ermitteln(self) -> str:
        override = os.environ.get(f"BROWSER_DIENST_BINARY_{self.schluessel.upper()}", "").strip()
        if override:
            return override
        pfad = browser_erkennung.binary_finden(self.schluessel)
        if not pfad:
            raise SicherheitsFehler(
                f"Kein {self.schluessel}-Binary gefunden (bekannte Namen: "
                f"{browser_erkennung._KANDIDATEN.get(self.schluessel)}). Menschliche "
                f"Entscheidung noetig -- installieren, oder "
                f"BROWSER_DIENST_BINARY_{self.schluessel.upper()}=<pfad> setzen."
            )
        return pfad

    def _singleton_haelt_wer(self) -> str | None:
        """PID-Hinweis aus dem SingletonLock des Zielprofils -- nur wenn die
        PID auch wirklich noch lebt UND zu diesem Browser gehoert (nicht z.B.
        ein laufendes Brave, das faelschlich als Chrome-Halter gilt).

        ⚠️ GEMESSEN 2026-08-21: der Lock zeigte einmal auf eine laengst tote
        PID (Browser nicht sauber beendet). Chrome selbst ignoriert das beim
        naechsten Start -- diese Pruefung muss dasselbe tun.
        """
        lock = Path(self.profil) / "SingletonLock"
        if not lock.is_symlink():
            return None
        try:
            ziel = os.readlink(lock)
            pid = ziel.rsplit("-", 1)[-1]
        except OSError:
            return None
        if not pid.isdigit():
            return None
        cmdline_pfad = Path(f"/proc/{pid}/cmdline")
        if not cmdline_pfad.is_file():
            return None  # PID lebt nicht mehr -- stale lock
        try:
            cmdline = cmdline_pfad.read_bytes()
        except OSError:
            return None
        if _LOCK_MUSTER[self.schluessel] not in cmdline.lower():
            return None  # PID wiederverwendet, gehoert nicht mehr diesem Browser
        return pid
