import (

// omega_status.go renders the landing screen's live OMEGA status line: organ
// health and GRID10 state revision/reentry lineage, read directly from the
// on-disk substrates the real runtime writes -- never a hardcoded or mocked
// number.
//
// It intentionally does NOT go through the Workspace abstraction
// (internal/workspace/workspace.go) the way the rest of this package's
// LSP/MCP/skills panels do: that interface has no organ/state plumbing, and
// adding it would mean threading a new RPC through the client/server HTTP
// boundary (internal/server/endpoints.go, internal/client/proto.go) for a
// splash-screen status line. Instead it opens fresh, read-only
// *omega.PersistentRuntime / *omega.TitanBodyOrgan handles onto the exact
// same on-disk paths omega.DefaultStatePath / omega.DefaultTitanBodyPath
// resolve for the coordinator's own live runtime
// (internal/agent/coordinator.go's NewCoordinator, which is constructed
// before the TUI is ever shown). That works in both local and client/server
// deployments as long as they share a filesystem, which is true of OMEGA's
// local deployment. It is honest, not a shortcut: reads only, no caching,
// no assumed process-local object graph.

// omegaStatusLine renders a compact, single-line OMEGA status summary for
// the landing screen: organ health summary plus GRID10 revision/reentry
// lineage. Returns "" if no data directory is configured (nothing to probe
// yet, e.g. during early onboarding).
func (m *UI) omegaStatusLine(width int) string {
	if m.com == nil {
		return ""
	}
	t := m.com.Styles
	cfg := m.com.Config()
	if cfg == nil || cfg.Options == nil {
		return ""
	}
	dataDir := strings.TrimSpace(cfg.Options.DataDirectory)
	if dataDir == "" {
		return ""
	}

	organIcon, organText := omegaOrganSummary(t, dataDir)
	stateText := omegaStateSummary(dataDir)

	line := fmt.Sprintf("%s %s%s%s",
		organIcon,
		t.Resource.CapabilityCount.Render(organText),
		"  ",
		t.Resource.AdditionalText.Render(stateText),
	)
	return lipgloss.NewStyle().Width(width).Render(line)
}

// omegaOrganSummary reads TITAN-Body's on-disk organ topology (the same
// substrate omega_titan_body.go's TitanBodyOrgan.RegisterOrgan/ReportHealth
// write to during boot -- internal/agent/coordinator.go's NewCoordinator
// calls omega.RegisterDefaultOrgans then organRuntime.Start synchronously
// before the coordinator is returned, so by the time the TUI can be showing
// anything, every default organ has at least a registered topology record,
// even if its self-reported health hasn't landed yet) and summarizes
// registered-organ health as "N/M organs healthy".
//
// omega.HealthOK is 0 in the HealthStage enum (omega_monitor.go:
// HealthOK=0 < HealthTight=1 < HealthCritical=2) -- the lowest/healthy
// value, not a zero-value placeholder. Organs that registered but never
// self-reported (HealthKnown=false) are counted separately as "unknown"
// rather than silently folded into either healthy or unhealthy -- an honest
// "still booting" signal instead of a fabricated number.
func omegaOrganSummary(t *styles.Styles, dataDir string) (icon, text string) {
	path := omega.DefaultTitanBodyPath(dataDir)
	if path == "" {
		return t.Resource.OfflineIcon.String(), "organs: n/a"
	}
	body, err := omega.NewTitanBodyOrgan(path)
	if err != nil {
		return t.Resource.OfflineIcon.String(), "organs: n/a"
	}
	topo, err := body.Topology()
	if err != nil {
		return t.Resource.OfflineIcon.String(), "organs: unavailable"
	}
	if len(topo) == 0 {
		return t.Resource.BusyIcon.String(), "organs: booting…"
	}

	var healthy, unhealthy, unknown int
	for _, o := range topo {
		switch {
		case !o.HealthKnown:
			unknown++
		case o.Health == omega.HealthOK:
			healthy++
		default:
			unhealthy++
		}
	}

	total := len(topo)
	summary := fmt.Sprintf("%d/%d organs healthy", healthy, total)
	if unknown > 0 {
		summary += fmt.Sprintf(" (%d unknown)", unknown)
	}

	switch {
	case unhealthy > 0:
		icon = t.Resource.ErrorIcon.String()
	case unknown > 0:
		icon = t.Resource.BusyIcon.String()
	default:
		icon = t.Resource.OnlineIcon.String()
	}
	return icon, summary
}

// omegaStateSummary reads PersistentRuntime's on-disk state (the same
// state.json omega_turn_loop.go's Reenter durably advances Revision on) via
// Peek -- never Snapshot, which would auto-vivify a throwaway entity file
// as a side effect of merely drawing the splash screen (see Peek's doc
// comment in state.go) -- and reports the GRID10 state revision plus
// whether this is a reentry into prior canonical lineage (Revision > 0: at
// least one prior Reenter transition has been committed to this entity) or
// a fresh entity that has never completed a turn (no state file yet, or
// Revision == 0 with no LastTurnID).
func omegaStateSummary(dataDir string) string {
	path := omega.DefaultStatePath(dataDir)
	if path == "" {
		return "state: n/a"
	}
	rt, err := omega.NewPersistentRuntime(path)
	if err != nil {
		return "state: n/a"
	}
	state, ok, err := rt.Peek()
	if err != nil {
		return "state: unavailable"
	}
	if !ok {
		return "fresh entity (no reentry yet)"
	}

	if state.Revision == 0 && strings.TrimSpace(state.LastTurnID) == "" {
		return fmt.Sprintf("rev 0 · fresh entity %s", shortOmegaID(state.EntityID))
	}

	return fmt.Sprintf("rev %d · reentry (last turn %s)", state.Revision, shortOmegaID(state.LastTurnID))
}

// shortOmegaID truncates a hex id (entity id or turn id digest) to an
// 8-character prefix for compact display, leaving short strings untouched.
func shortOmegaID(id string) string {
	id = strings.TrimSpace(id)
	if len(id) <= 8 {
		return id
	}
	return id[:8]
}

crush update-providers /path/to/local-providers.json

	},

// Default is the default [Client].

// Info contains information about an available update.
type Info struct {
	Current string
	Latest  string
	URL     string
}

// Matches a version string like:
// v0.0.0-0.20251231235959-06c807842604
var goInstallRegexp = regexp.MustCompile(`^v?\d+\.\d+\.\d+-\d+\.\d{14}-[0-9a-f]{12}$`)

func (i Info) IsDevelopment() bool {
	return i.Current == "devel" || i.Current == "unknown" || strings.Contains(i.Current, "dirty") || goInstallRegexp.MatchString(i.Current)
}

// Available returns true if there's an update available.
//
// If both current and latest are stable versions, returns true if versions are
// different.
// If current is a pre-release and latest isn't, returns true.
// If latest is a pre-release and current isn't, returns false.
func (i Info) Available() bool {
	cpr := strings.Contains(i.Current, "-")
	lpr := strings.Contains(i.Latest, "-")
	// current is pre release && latest isn't a prerelease
	if cpr && !lpr {
		return true
	}
	// latest is pre release && current isn't a prerelease
	if lpr && !cpr {
		return false
	}
	return i.Current != i.Latest
}

// Check checks if a new version is available.
func Check(ctx context.Context, current string, client Client) (Info, error) {
	info := Info{
		Current: current,
		Latest:  current,
	}

	release, err := client.Latest(ctx)
	if err != nil {
		return info, fmt.Errorf("failed to fetch latest release: %w", err)
	}

	info.Latest = strings.TrimPrefix(release.TagName, "v")
	info.Current = strings.TrimPrefix(info.Current, "v")
	info.URL = release.HTMLURL
	return info, nil
}

// Release represents a GitHub release.
type Release struct {
	TagName string `json:"tag_name"`
	HTMLURL string `json:"html_url"`
}

// Client is a client that can get the latest release.
type Client interface {
	Latest(ctx context.Context) (*Release, error)
}

type github struct{}

// Latest implements [Client].
func (c *github) Latest(ctx context.Context) (*Release, error) {
	client := &http.Client{
		Timeout: 30 * time.Second,
	}

	req, err := http.NewRequestWithContext(ctx, "GET", githubApiUrl, nil)
	if err != nil {
		return nil, err
	}
	req.Header.Set("User-Agent", userAgent)
	req.Header.Set("Accept", "application/vnd.github.v3+json")

	resp, err := client.Do(req)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()

	if resp.StatusCode != http.StatusOK {
		body, _ := io.ReadAll(resp.Body)
		return nil, fmt.Errorf("GitHub API returned status %d: %s", resp.StatusCode, string(body))
	}

	var release Release
	if err := json.NewDecoder(resp.Body).Decode(&release); err != nil {
		return nil, err
	}

	return &release, nil
}
