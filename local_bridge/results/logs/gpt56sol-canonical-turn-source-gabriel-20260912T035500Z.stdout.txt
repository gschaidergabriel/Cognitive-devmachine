def cmd_f2wp1207_turn_loop_turn(force_new_epoch: bool = False,
                                declared_reentry: Optional[str] = None,
                                evidence_path: Optional[str] = None) -> int:
    """CLI-Einstiegspunkt `stern.py f2wp1207-turn-loop-turn`. Liest EIN
    JSON von stdin (TypedEntry-Felder, siehe Moduldocstring oben), schreibt
    EIN JSON nach stdout. Fail-closed komplett, Exit-Code bleibt immer 0 --
    gleiches Muster wie cmd_f2wp1207_p15_canary_turn oben."""
    try:
        raw = sys.stdin.read()
        data = json.loads(raw) if raw.strip() else {}
        if not isinstance(data, dict):
            data = {}
    except Exception:
        data = {}
    data.setdefault("session_id", "unknown")
    if force_new_epoch:
        data["force_new_epoch"] = True
    if declared_reentry:
        data["declared_reentry"] = declared_reentry
    if evidence_path:
        data["evidence_path"] = evidence_path
    try:
        ergebnis = _f2wp1207_turn_loop_turn_ausfuehren(data)
    except Exception as e:  # doppelte Absicherung, wie bei P15
        ergebnis = {"ok": False, "active": False, "fehler": str(e)}
    print(json.dumps(ergebnis, ensure_ascii=False))
    return 0
