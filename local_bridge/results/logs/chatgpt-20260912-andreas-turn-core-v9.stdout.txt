def sag_kurz(tts_voice, text):
    """Einen einzelnen Satz sprechen, ohne das Sprachmodell zu bemuehen.

    BEWUSST OHNE MODELL: Anlass ist, dass gerade NICHTS Verlaessliches
    verstanden wurde. Genau dann eine Formulierung generieren zu lassen,
    hiesse, das Modell auf Rauschen antworten zu lassen -- also exakt die
    Konfabulation, die hier verhindert werden soll."""
    try:
        # Waehrend ER spricht, muss die Musik zurueck -- sonst geht seine
        # Stimme darin unter. Danach sofort wieder hoch.
        if _ABSPIELER[0] is not None:
            _ABSPIELER[0].ducken(True)
        q = queue.Queue()
        halter = {}
        t = threading.Thread(target=tts_speaker_worker,
                             args=(tts_voice, q, halter))
        t.start()
        q.put(("sentence", text))
        q.put(("done", text, 0.0))
        t.join(timeout=30)
    except Exception as e:
        _log(f"[GEHOER] konnte Nachfrage nicht sprechen: {type(e).__name__}: {e}")
    finally:
        if _ABSPIELER[0] is not None:
            try:
                _ABSPIELER[0].ducken(False)
            except Exception:
                pass


def self_introduction(tts_voice, session_id):
    """Einmalige gesprochene Selbstvorstellung beim ALLERERSTEN Einschalten des
    Sprachmodus (Auftrag §1 / HANDOVER_SPRACHMODUS.md Punkt 2d). Wird vom LLM
    selbst live erzeugt (kein vorformulierter Satz), gated durch einen
    persistenten star_konfig-Marker, damit sie nie ein zweites Mal laeuft.
    Schlaegt speech fehl, bleibt der erzeugte Text in star_konfig stehen --
    das Dashboard kann ihn als Text-Rueckfallebene zeigen (Auftrag: Text ist
    die Rueckfallebene, gesprochen ist das Ziel)."""
    schon = _konfig_lesen("gespraech.begruessung_erfolgt", "0")
    if str(schon).strip() in ("1", "true", "True"):
        _log("[INTRO] Begruessung wurde bereits einmal gehalten -- ueberspringe.")
        return
    _log("[INTRO] allererste Aktivierung erkannt -- generiere die Selbstvorstellung live.")

    if not erzeugung_verfuegbar():
        _log("[INTRO] FEHLER: llama-server nicht erreichbar -- Begruessung wird beim "
             "naechsten Einschalten erneut versucht.")
        return

    fakt = _eine_erinnerung_fuer_begruessung(session_id)
    system_content = INTRO_SYSTEM_PROMPT
    if fakt:
        system_content += f"\n\nEcht live abgerufene Tatsache: {fakt}"
    else:
        system_content += "\n\n(Kein Treffer im Gedaechtnis fuer diesen Versuch.)"
    messages = [
        {"role": "system", "content": system_content},
        {"role": "user", "content": "Stell dich jetzt vor, wie oben beschrieben."},
    ]

    sentence_q = queue.Queue()
    result_holder = {}
    llm_thread = threading.Thread(
        target=stream_llm_to_sentences, args=(messages, sentence_q), kwargs={"max_tokens": 260}
    )
    speaker_thread = threading.Thread(target=tts_speaker_worker, args=(tts_voice, sentence_q, result_holder))
    llm_thread.start()
    speaker_thread.start()
    llm_thread.join(timeout=150)
    speaker_thread.join(timeout=120)

    if result_holder.get("status") == "done" and result_holder.get("full_text"):
        text = result_holder["full_text"]
        _log(f"[INTRO] gesprochen ({result_holder.get('sentence_count')} Saetze, "
             f"{result_holder.get('llm_s')}s LLM, {result_holder.get('tts_total_s')}s TTS): {text!r}")
        _konfig_schreiben("gespraech.begruessung_text", text, "gespraech_loop-intro")
        _konfig_schreiben("gespraech.begruessung_erfolgt", "1", "gespraech_loop-intro")
    else:
        _log(f"[INTRO] FEHLER: Selbstvorstellung konnte nicht erzeugt/gesprochen werden "
             f"(status={result_holder.get('status')}) -- wird beim naechsten Einschalten "
             "erneut versucht, NICHT als erfolgt markiert.")
