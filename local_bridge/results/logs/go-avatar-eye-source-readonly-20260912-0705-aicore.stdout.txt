def masse(pfad):
    arr = np.array(Image.open(pfad).convert("RGBA"))
    al = arr[..., 3]
    grau = cv2.cvtColor(arr[..., :3], cv2.COLOR_RGB2GRAY)
    ys, xs = np.where(al > 128)
    if len(xs) == 0:
        return None
    x0, x1, y0, y1 = xs.min(), xs.max(), ys.min(), ys.max()
    g = grau[y0:y1 + 1, x0:x1 + 1].astype(np.float32)
    dx = np.abs(np.diff(g, axis=1)).mean()
    dy = np.abs(np.diff(g, axis=0)).mean()
    return dict(alpha=al.tobytes(), bbox=(int(x0), int(y0), int(x1), int(y1)),
                kontrast=round(float(g.std()), 2), kante=round(float((dx + dy) / 2), 3),
                helligkeit=round(float(g.mean()), 2))
