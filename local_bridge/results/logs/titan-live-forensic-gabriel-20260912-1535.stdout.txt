#!/usr/bin/env python3
"""TITAN-MX HyperMirror Claude UserPromptSubmit hook.

Separate from stern.py by design: MemoryX is cache/context, never truth or
effect authority. The hook only emits additionalContext and always fails
silent with exit 0.

Activation is fail-closed. It is ON only when either:
- TITAN_MX_INJECT is explicitly truthy, or
- the gate file exists (default ~/.local/share/agentzero/titan_mx_inject.enabled).

The continuous TITAN-MX daemon owns consolidation. This hook only reads the
latest bounded mirror from titan_mx.db; missing/broken/stale state emits
nothing.
"""
from __future__ import annotations

import json
import os
import sys
from pathlib import Path

from f2wp1207_titan_mx_inject import injection_text

TRUTHY = {"1", "true", "yes", "on"}
FALSY = {"0", "false", "no", "off"}
DEFAULT_STATE_DB = Path.home() / ".local/share/agentzero/titan_mx.db"
DEFAULT_GATE_FILE = Path.home() / ".local/share/agentzero/titan_mx_inject.enabled"
DEFAULT_MAX_CHARS = 1400
DEFAULT_MAX_AGE_S = 900.0


def _enabled() -> bool:
    raw = os.environ.get("TITAN_MX_INJECT")
    if raw is not None:
        v = raw.strip().lower()
        if v in TRUTHY:
            return True
        if v in FALSY:
            return False
        return False
    gate = Path(os.environ.get("TITAN_MX_GATE_FILE", str(DEFAULT_GATE_FILE))).expanduser()
    return gate.is_file()


def _int_env(name: str, default: int, lo: int, hi: int) -> int:
    try:
        return max(lo, min(hi, int(os.environ.get(name, str(default)))))
    except Exception:
        return default


def _float_env(name: str, default: float, lo: float, hi: float) -> float:
    try:
        return max(lo, min(hi, float(os.environ.get(name, str(default)))))
    except Exception:
        return default


def run(payload: dict) -> dict | None:
    if payload.get("hook_event_name") != "UserPromptSubmit":
        return None
    if not _enabled():
        return None

    state_db = Path(os.environ.get("TITAN_MX_STATE_DB", str(DEFAULT_STATE_DB))).expanduser()
    max_chars = _int_env("TITAN_MX_MAX_CHARS", DEFAULT_MAX_CHARS, 128, 2000)
    max_age_s = _float_env("TITAN_MX_MAX_AGE_S", DEFAULT_MAX_AGE_S, 1.0, 86400.0)
    text = injection_text(state_db, max_chars=max_chars, max_age_s=max_age_s)
    if not text:
        return None
    return {
        "hookSpecificOutput": {
            "hookEventName": "UserPromptSubmit",
            "additionalContext": text,
        }
    }


def main() -> int:
    try:
        raw = sys.stdin.read()
        payload = json.loads(raw) if raw.strip() else {}
        out = run(payload if isinstance(payload, dict) else {})
        if out:
            print(json.dumps(out, ensure_ascii=False, separators=(",", ":")))
    except Exception:
        # Harte Grenze: this context organ must never break the host hook chain.
        pass
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
