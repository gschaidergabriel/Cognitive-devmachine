function kopfRollSequenzZu(ziel, fertig){
  const schritte = [];
  const richtung = ziel > kopfRollWert ? 5 : -5;
  for(let w = kopfRollWert + richtung; ; w += richtung){
    schritte.push({w, src: kopfRollDatei(w)});
    if(w === ziel) break;
  }
  kopfRollSchritte = schritte; kopfRollIdx = 0; kopfRollLetzterT = 0;
  kopfRollFertigFn = fertig || null;
}
function kopfRollTick(jetzt){
  if(!kopfRollSchritte) return;
  if(jetzt - kopfRollLetzterT < KOPF_ROLL_MS) return;
  const schritt = kopfRollSchritte[kopfRollIdx];
  const eintrag = bildLaden(schritt.src);
  if(!eintrag.ok) return; // Frame noch nicht dekodiert -- warten statt halbes Bild zeigen
  kopfRollLetzterT = jetzt;
  const ein = kopfRollAktiv ? "kopf-schicht" : "kopf-schicht2";
  const aus = kopfRollAktiv ? "kopf-schicht2" : "kopf-schicht";
  $(ein).src = schritt.src;
  // ⭐⭐ PERF-FIX 2026-09-11: atmen() aktualisiert seit diesem Fix nur noch
  // die aktuell sichtbare Kopf-Ebene (siehe atmen() oben) -- die Ebene, die
  // hier gerade sichtbar wird, muss deshalb JETZT einmalig den brandaktuellen
  // Atem-Wert bekommen, sonst zeigt sie beim ersten sichtbaren Frame noch den
  // Wert von ihrem letzten Auftritt (Pop). _atmenTransform/_atmenFilter sind
  // von atmen() diesen Frame schon gesetzt (schleife() ruft atmen() vor
  // kopfRollTick() auf).
  $(ein).style.transform = _atmenTransform;
  $(ein).style.filter = _atmenFilter;
  $(ein).style.opacity = "1";
  $(aus).style.opacity = "0";
  kopfRollAktiv = !kopfRollAktiv;
  kopfRollWert = schritt.w;
  kopfRollIdx++;
  if(kopfRollIdx >= kopfRollSchritte.length){
    const f = kopfRollFertigFn;
    kopfRollSchritte = null; kopfRollFertigFn = null;
    if(f) f();
  }
}

const KOPF_ROLL_AUGEN_OFFSET = { // native 1536x1024-Leinwand-Pixel, exakt
  "-25": {dx:-32.5, dy: 22.0}, "-20": {dx:-30.0, dy: 12.0},
  "-15": {dx:-24.0, dy:  5.0}, "-10": {dx:-16.5, dy:  0.0},

  "25":  {dx: 62.5, dy:-112.0},
};
let augenOffsetDx = 0, augenOffsetDy = 0; // geglaettete ANZEIGE, siehe kopfRollAusgleichDeg
function federSchritt(ist, v, ziel, steife, daempfung){
  v += (ziel - ist) * steife - v * daempfung;
  ist += v;
  return [ist, v];
}
function kopfKoerperDrehSchritt(){
  // Federziel aus dem ECHTEN Rollwert (12% davon, auf ±3° begrenzt) -- die
  // Bildpose macht die Neigung, die Feder nur das kleine Nachschwingen.
  const zielRot = Math.max(-3, Math.min(3, kopfRollSoll * 0.12));
  [kDrehRot, kDrehRotV] = federSchritt(kDrehRot, kDrehRotV, zielRot, 0.020, 0.20);
  [kDrehTy,  kDrehTyV ] = federSchritt(kDrehTy,  kDrehTyV,  0,      0.020, 0.20);
  // 34% des AKTUELLEN (nicht Ziel-)Kopfwerts + eigene weichere Feder =
  // Koerper folgt spaeter und schwaecher, ohne starren festen Versatz.
  [bDrehRot, bDrehRotV] = federSchritt(bDrehRot, bDrehRotV, kDrehRot*0.34, 0.010, 0.26);
  [bDrehTy,  bDrehTyV ] = federSchritt(bDrehTy,  bDrehTyV,  kDrehTy*0.34,  0.010, 0.26);
  $("kopf-dreh").style.transform = `rotate(${kDrehRot.toFixed(2)}deg) translateY(${kDrehTy.toFixed(1)}px)`;
  $("koerper-dreh").style.transform = `rotate(${bDrehRot.toFixed(2)}deg) translateY(${bDrehTy.toFixed(1)}px)`;
  // Rest-Ausgleich fuer #pupillen-glow/#augen-blink -- siehe Kommentar oben
  // (STIRN-LUECKE-FIX): Ziel ist jetzt kopfRollWert selbst (kDrehRot kuerzt
  // sich per CSS-Vererbung von selbst weg, siehe Kommentar), UND geglaettet
  // per Lerp statt hartem Sprung -- entkoppelt den Flicken vom 5°-Sprung
  // des Fotowechsels, ohne ihn wahrnehmbar hinter dem echten Winkel
  // zurueckfallen zu lassen (Faktor 0.4, faengt sich in 2-3 Frames).
  // Geometrische Registrierung ist atomar mit dem bereits in kopfRollTick()
  // sichtbaren Foto-Schritt. Nur der gemeinsame #kopf-dreh/#koerper-dreh-Wrapper
  // darf federn. Eine lokale Lerp-Verzoegerung erzeugt sonst 2-3 Frames lang
  // absichtliche Fehlregistrierung zwischen Foto und Augen-/Blink-Overlay.
  kopfRollAusgleichDeg = kopfRollWert;
  const zielOffset = KOPF_ROLL_AUGEN_OFFSET[String(kopfRollWert)] || {dx:0, dy:0};
  augenOffsetDx = zielOffset.dx;
  augenOffsetDy = zielOffset.dy;
  const augenDxCss = augenOffsetDx * _kopfCanvasScale;
  const augenDyCss = augenOffsetDy * _kopfCanvasScale;
  $("augen-blink").style.transform =
    `translate(${augenDxCss.toFixed(2)}px, ${augenDyCss.toFixed(2)}px) rotate(${kopfRollAusgleichDeg.toFixed(2)}deg)`;
}

// ⚠️ Galton-Watson-Regel UNVERAENDERT (siehe LECK-FIX oben): EINE

function blinzeln(zeitplanNeu = true){
  
  const schritte = [];
  for(let f = 0; f <= 8; f++) schritte.push([f, blinzelDauerSchliessen(), f < 7 ? "halb1" : "zu"]);
  schritte.push([8, blinzelDauerGeschlossenHalten(), "zu"]);
  for(let f = 9; f <= 11; f++) schritte.push([f, blinzelDauerOeffnen(), "halb2"]);
  blinzelAbspielen(schritte, 0, zeitplanNeu);
}
function blinzelAbspielen(schritte, idx, zeitplanNeu){
  if(idx >= schritte.length){
    $("augen-blink").src = BLINK_BILDER[0];
    $("augen-blink").style.opacity = "1";

    if(zeitplanNeu){
      if(Math.random() < 0.12){ setTimeout(() => blinzeln(false), 120 + Math.random()*80); }
      blinzelWeiterPlanen();

  }
  const [frame, dauer, phase] = schritte[idx];
  $("augen-blink").src = BLINK_BILDER[frame];
  $("augen-blink").style.opacity = "1";
  blinzelPhase = phase; gesichtNeuZeichnen(false);
  setTimeout(() => blinzelAbspielen(schritte, idx + 1, zeitplanNeu), dauer);
}

//                      Mikro-Pause aktiv -> Ziel = _wegZiel (sichtbar
//                      anders, aber gedaempft: "da, aber nicht fixiert")
//   0    Opazitaet  -- serverBlick nicht vorhanden/nicht nah (unveraendert,
//                      Hartregel: kein erfundener Blick ohne Erkennung)
function gazeZielBestimmen(){
  if(!gazeBasisSichtbar()) return {x:0.5, y:0.5, opazitaet:0};
  const wegAktiv = blickPauseAktiv || gespraechsFokus === "thema";
  if(wegAktiv) return {x:_wegZiel.x, y:_wegZiel.y, opazitaet:0.4};
  return {x:serverBlick.cx, y:serverBlick.cy, opazitaet:1};
}
function gazeSchritt(){
  const ziel = gazeZielBestimmen();
  zielGazeX = ziel.x; zielGazeY = ziel.y;
  aktGazeX += (zielGazeX - aktGazeX) * 0.07;
  aktGazeY += (zielGazeY - aktGazeY) * 0.07;
  const maxPx = 8.5; // Bewegungsspielraum bewusst klein -- ein Glimmer, kein Springen
  const dx = (aktGazeX - 0.5) * 2 * maxPx + mikroSakX;
  const dy = (aktGazeY - 0.5) * 2 * maxPx + BLICK_ABWAERTS_BIAS_PX + mikroSakY;
  // ⭐⭐ 2026-09-11 NACHTRAG 4: rotate() dranhaengen (Rest-Ausgleich zur echten
  // Rollpose, siehe kopfKoerperDrehSchritt()) -- sonst bleibt der Glimmer beim
  // Kopfkippen an der falschen Stelle stehen (Geisterkopf-Kante).
  // ⭐⭐⭐ 2026-09-12: dieselbe exakte Roll-Translation wie #augen-blink (siehe
  // KOPF_ROLL_AUGEN_OFFSET/kopfKoerperDrehSchritt oben) draufaddiert -- sonst
  // driftet der Glimmer beim Rollen vom Blinzel-Flicken weg, obwohl beide
  // denselben Augen-Anker verfolgen sollen.
  const dxGesamt = dx + augenOffsetDx * _kopfCanvasScale;
  const dyGesamt = dy + augenOffsetDy * _kopfCanvasScale;
  $("pupillen-glow").style.transform = `translate(${dxGesamt.toFixed(2)}px, ${dyGesamt.toFixed(2)}px) rotate(${kopfRollAusgleichDeg.toFixed(2)}deg)`;
  // ⭐ 2026-09-11: waehrend eines Sonderausdrucks aus -- dessen Frames tragen
  // ihre eigenen Augen, sonst laege der Erkennungs-Glimmer auf einem fremden
  // Gesicht (zweites Augenpaar statt zweiter Kopf).
  gazeSichtbar = ziel.opazitaet > 0; // Rueckwaertskompatibilitaet: anderswo evtl. gelesen
  $("pupillen-glow").style.opacity = blinzelPhase ? "0" : String(ziel.opazitaet);
}

const BILD_W = 1536, BILD_H = 1024, OBJ_POS_X = 0.50, OBJ_POS_Y = 0.15;

function kopfCanvasAnpassen(){

  if(!cw || !ch) return; // noch nicht gelayoutet -- naechster Aufruf greift

  _kopfCanvasScale = scale;

  const oy = (ch - rh) * OBJ_POS_Y;
  const kc = $("kopf-canvas");
  kc.style.left = ox.toFixed(2) + "px";
  kc.style.top = oy.toFixed(2) + "px";
  kc.style.width = rw.toFixed(2) + "px";

function regenZeichnenEbene(c, cv, liste, alpha, breite){
  c.clearRect(0,0,cv.width, cv.height);
  c.lineWidth = breite;
  for(const t of liste){
    c.strokeStyle = `rgba(${t.ton},${alpha})`;
    c.beginPath();
    c.moveTo(t.x, t.y);
    c.lineTo(t.x, t.y + t.len);
    c.stroke();
    t.y += t.v * 3.2;
  }
}

  _mikroUnruhe += (_mikroUnruheZiel - _mikroUnruhe) * 0.02; // sehr langsam, kein Sprung

  const transform = `scale(${skala})`;
  const filter = `saturate(1.05) brightness(${helligkeit})`;
  _atmenTransform = transform;
  _atmenFilter = filter;

  // Koerper bekommt DENSELBEN Transform+Filter wie der Kopf -- beide
  // skalieren/hellen im GLEICHEN Takt, die Halskante bleibt an derselben
  // Relativstelle. Der Koerper "atmet" dadurch zusaetzlich zu seinen 12
  // fotografierten Stufen leicht mit (max +1,2% Groesse) -- das ist die

// Handy filmen, am besten sogar ganz leicht Cam-Shake, aber realistisch

// ueberlagerte, bewusst NICHT-zufaellige Komponenten (Gabriels Vorgabe:
// gleichfoermiges Zufallsrauschen sieht nicht wie eine echte Hand aus):
//  1) langsame Drift (Atmung/Ermuedung der haltenden Hand, ~4-5s Periode,
//     wenige px) -- Fortsetzung der bisherigen "nie ganz still"-Drift,
//     jetzt auf die ganze Kamera statt nur auf #gesicht verschoben.
//  2) schnelles Mikro-Zittern (~8-12Hz, <1px) -- der Muskelzittern-Anteil
//     einer echten Hand, zwei knapp benachbarte Sinusfrequenzen statt einer
//     einzelnen, damit es nicht mechanisch/taktfest wirkt.
//  3) minimale Rotation (Bruchteile eines Grads) -- eine Hand kippt nie rein
//     linear.
// Hintergrund bewegt sich absichtlich einen Hauch WENIGER als der Rest
// (0.85x) -- winziger Parallaxe-Tiefenhinweis, siehe Koordinator ("optional:
// Hintergrund bewegt sich minimal weniger als der Vordergrund").
// ⭐⭐ 2026-09-11 RETUNE nach Gabriels Live-Urteil ("er zittert, schaut
// relativ shit aus"): die 8-10Hz-Mikrozitter-Komponente unten war zu schnell
// UND zu stark -- auf einem kleinen Panel (~296px breit), mit dem neuen
// engeren Kopf-Crop, UND auf einem Chrome-Fenster ohne GPU-Kompositing
// (--disable-gpu, siehe Fensterstart) liest sich eine 8-10Hz-Oszillation
// nicht als sanftes Handzittern, sondern als sichtbares Vibrieren/Ruckeln --
// das Auge ist fuer zeitliches Flackern empfindlicher als fuer die reine
// Pixelamplitude. Fix: Frequenz UND Amplitude aller drei Komponenten deutlich
// gesenkt (Tremor von ~10/8Hz auf ~3/2.3Hz, Drift/Rotation-Amplitude
// grob halbiert). Ziel ist jetzt spuerbar ruhiger -- lieber zu wenig als
// wieder "zittert".
let driftPhaseX = Math.random() * Math.PI * 2;
let driftPhaseY = Math.random() * Math.PI * 2;
let tremorPhaseA = Math.random() * Math.PI * 2;
let tremorPhaseB = Math.random() * Math.PI * 2;
let rotPhase = Math.random() * Math.PI * 2;
function kameraWackeln(){
  const t = Math.max(0.4, Math.min(1.6, tempo)); // "passend zur Bewegung": etwas mehr Schub bei hoeherem Tempo
  driftPhaseX += 0.0131; // ~8.0s Periode bei 60fps (war ~4.0s)
  driftPhaseY += 0.0105; // ~9.6s Periode -- absichtlich versetzt zu X (war ~4.9s)
  tremorPhaseA += 0.314;  // ~3Hz (war ~10Hz)
  tremorPhaseB += 0.251;  // ~2.4Hz, knapp daneben statt exakt gleich (war ~8Hz)
  rotPhase += 0.0063;     // ~16s Periode, noch langsamer als die Drift (war ~8.3s)
  const driftX = Math.sin(driftPhaseX) * 0.7 * t;
  const driftY = Math.cos(driftPhaseY) * 0.55 * t;
  const tremorX = (Math.sin(tremorPhaseA) * 0.14 + Math.sin(tremorPhaseB) * 0.09) * t;
  const tremorY = (Math.cos(tremorPhaseA * 0.97) * 0.12 + Math.cos(tremorPhaseB * 1.05) * 0.07) * t;
  const rotDeg = Math.sin(rotPhase) * 0.06 * t;
  const x = driftX + tremorX, y = driftY + tremorY;
  $("kamera").style.transform = `translate(${x.toFixed(2)}px, ${y.toFixed(2)}px) rotate(${rotDeg.toFixed(3)}deg)`;
  // Hintergrund bekommt denselben Wackel, aber 0.85x gedaempft -- winzige
  // Parallaxe, das Bild wirkt nicht wie eine starre Fotomontage auf dem
  // Hintergrund geklebt. #hg-frame ist Kind von #kamera -- dieser Transform
  // addiert sich zu dem der Kamera, deshalb hier bewusst der GEGEN-Anteil
  // (0.85-1 = -0.15x) statt des vollen Werts.
  $("hg-frame").style.transform = `translate(${(x*-0.15).toFixed(2)}px, ${(y*-0.15).toFixed(2)}px)`;
}
