# ── Hilfsfunktionen ──────────────────────────────────────────────────────
def _jetzt() -> float:
    return time.time()


def _kurz(wert, n: int = 220) -> str:
    if isinstance(wert, (dict, list)):
        wert = json.dumps(wert, ensure_ascii=False, separators=(",", ":"))
    s = str(wert if wert is not None else "").replace("\n", " ").strip()
    return s if len(s) <= n else s[: n - 1] + "…"


def _kurz_wortgrenze(wert, n: int = 90) -> str:
    """Wie _kurz(), aber kappt an einer Wortgrenze (textwrap.shorten) statt
    mitten im Wort -- Gabriel-Feedback 2026-08-26: reale Renders des
    Willkommensbanners zeigten mitten im Wort abgeschnittenen Text
    ('mechanisc…', 'passw…'). NUR fuer Anzeige-Zwecke (Banner-Stichworte) --
    _kurz() bleibt fuer effects/Protokoll-Zwecke unveraendert (dort zaehlt
    exakte Zeichenzahl als Kappungsgrenze, keine Wortgrenze). Fallback auf
    harte Zeichenkappung, falls textwrap.shorten() bei einem einzelnen
    ueberlangen "Wort" (z.B. Pfad ohne Leerzeichen) leer zurueckkaeme --
    nie eine leere Zeile ausgeben."""
    if isinstance(wert, (dict, list)):
        wert = json.dumps(wert, ensure_ascii=False, separators=(",", ":"))
    s = str(wert if wert is not None else "").replace("\n", " ").strip()
    if len(s) <= n:
        return s
    gekuerzt = textwrap.shorten(s, width=n, placeholder="…")
    if not gekuerzt.strip("… "):
        return s[: n - 1] + "…"
    return gekuerzt


def _uid(prefix: str) -> str:
    return f"{prefix}-{int(_jetzt() * 1000)}-{uuid.uuid4().hex[:6]}"


# ── Geheimnis-Redaktion (Sicherheits-Nachtrag zu PHASE 18, 2026-08-26) ──────
# Realer, demonstrierter Leak: EffectGate/Ereignisspur (zulassen()/merken(),
# unten) schreiben tool_input/argv/target (u.a. Bash-`command`, Write/Edit-
# `content`) UNGEFILTERT in effects/durable_memory. Jedes Geheimnis, das je
# ueber einen Bash-Argumenttext oder einen Datei-Inhalt laeuft, landete
# dauerhaft im Klartext in unified.db -- unabhaengig vom Tresor (PHASE 18),
# dessen eigene gpg-Aufrufe die Passphrase NACHWEISLICH per `--passphrase-fd`
# (eigener Pipe, kein argv, siehe _gpg_lauf()) uebergeben und daher selbst
# NICHT betroffen sind. Primaere Abwehr laut Tresor-Onboarding: Mensch tippt
# Geheimnisse im eigenen Terminal, nie per modellgesteuerten Bash-Aufruf.
# Diese Funktion ist die SEKUNDAERE, defense-in-depth Abwehr: geheimnisartige
# Teilstrings werden VOR dem Schreiben getilgt, unabhaengig davon, wie das
# Geheimnis in den Aufruf gelangt ist. Nur die Werte werden ersetzt (Marker
# "[REDACTED]"), der Rest der Zeile bleibt lesbar -- kein Blackout des
# gesamten Log-Eintrags. Reihenfolge wichtig: spezifische Muster (bekannte
# Praefixe, benannte Flags/Env-Zuweisungen) zuerst, generischer Hochentropie-
# Faenger zuletzt (er wuerde sonst z.B. den Marker selbst nicht treffen, aber
# koennte bereits redigierte Reste anders verzerren).
_GEHEIM_PRAEFIX_MUSTER = re.compile(
    r"\b(?:sk-[A-Za-z0-9_\-]{10,}|gsk_[A-Za-z0-9_\-]{10,}|gh[pousr]_[A-Za-z0-9_\-]{10,}"
    r"|xox[baprs]-[A-Za-z0-9\-]{10,}|AKIA[0-9A-Z]{16})\b"
)
_GEHEIM_BEARER_MUSTER = re.compile(r"\bBearer\s+[A-Za-z0-9._\-]{10,}")
# Benannte Flags (--password, --passphrase, --token, --api-key, --secret, ...):
# Flag-Name bleibt sichtbar, nur der Wert dahinter wird getilgt.
_GEHEIM_FLAG_MUSTER = re.compile(
    r"(?i)((?:--)?(?:passphrase|password|pass|api[-_]?key|apikey|token|secret)\s*[=: ]\s*)(\S+)"
)
# Env-Zuweisungsstil (GROSSSCHREIBUNG bewusst Pflicht -- reduziert Fehltreffer
# auf normalen Fliesstext): PASSWORD=..., API_KEY=..., TOKEN=..., SECRET=...
_GEHEIM_ENV_MUSTER = re.compile(
    r"\b([A-Z][A-Z0-9_]*(?:KEY|TOKEN|SECRET|PASSWORD|PASSPHRASE)[A-Z0-9_]*=)(\S+)"
)
# mysql-artiges "-p <wert>" -- NUR wenn Wert gemischt Buchstabe+Ziffer UND
# >=6 Zeichen ist, sonst zu viele Fehltreffer auf -p als Port/Parallelitaet.
_GEHEIM_MYSQLP_MUSTER = re.compile(r"(-p\s+)((?=\S*[0-9])(?=\S*[A-Za-z])\S{6,})")
# Generischer Hochentropie-Block: >=20 Zeichen [A-Za-z0-9_-], Buchstabe UND
# Ziffer gemischt (verwirft reine Wortfolgen und reine Zifferfolgen) --
# faengt Base64/Hex-Blobs und rohe Passwoerter ohne bekanntes Praefix, z.B.
# in `printf '...' | ...`-Pipes.
_GEHEIM_BLOB_MUSTER = re.compile(
    r"\b(?=[A-Za-z0-9_\-]{20,}\b)(?=[A-Za-z0-9_\-]*[0-9])(?=[A-Za-z0-9_\-]*[A-Za-z])[A-Za-z0-9_\-]{20,}\b"
)


def _geheimnis_redigieren(text: str) -> str:
    """Tilgt geheimnisartige Teilstrings aus `text`, ersetzt sie durch
    "[REDACTED]". Nicht-Strings/leere Werte werden unveraendert zurueckgegeben.
    Reihenfolge: bekannte Praefixe -> Bearer -> benannte Flags -> Env-Stil ->
    mysql-`-p` -> generischer Hochentropie-Faenger."""
    if not isinstance(text, str) or not text:
        return text
    s = text
    s = _GEHEIM_PRAEFIX_MUSTER.sub("[REDACTED]", s)
    s = _GEHEIM_BEARER_MUSTER.sub("Bearer [REDACTED]", s)
    s = _GEHEIM_FLAG_MUSTER.sub(r"\1[REDACTED]", s)
    s = _GEHEIM_ENV_MUSTER.sub(r"\1[REDACTED]", s)
    s = _GEHEIM_MYSQLP_MUSTER.sub(r"\1[REDACTED]", s)
    s = _GEHEIM_BLOB_MUSTER.sub("[REDACTED]", s)
    return s


def _kurz_redigiert(wert, n: int = 220) -> str:
    """Wie _kurz(), aber tilgt Geheimnisse VOR dem Kappen (ein Geheimnis, das
    genau an der Kappungsgrenze liegt, darf nicht als unerkannter Rest
    durchrutschen). Nur an den Stellen verwenden, die tool_input/argv/target/
    tool_response in effects/durable_memory schreiben -- _kurz() selbst bleibt
    unveraendert fuer alle anderen (nicht geheimnisverdaechtigen) Aufrufer."""
    if isinstance(wert, (dict, list)):
        wert = json.dumps(wert, ensure_ascii=False, separators=(",", ":"))
    s = str(wert if wert is not None else "").replace("\n", " ").strip()
    s = _geheimnis_redigieren(s)
    return s if len(s) <= n else s[: n - 1] + "…"


def _argv_normalisiert_redigiert(tool_input) -> str:
    """_argv_normalisiert() + Geheimnis-Tilgung VOR dem Kappen auf 500 Zeichen
    (siehe _kurz_redigiert()-Begruendung)."""

BUSY_MS = 30000
VERSUCHE = 4
BACKOFF_S = 0.15


def _verbinden(pfad: str = DB_PATH) -> sqlite3.Connection:
    """Nur-lesende Verbindung mit busy_timeout und Wiederholung."""
    letzte = None
    for versuch in range(VERSUCHE):
        try:
            c = sqlite3.connect("file:%s?mode=ro" % pfad, uri=True, timeout=BUSY_MS / 1000.0)
            c.execute("PRAGMA busy_timeout=%d" % BUSY_MS)
            c.execute("PRAGMA query_only=ON")
            return c
        except sqlite3.Error as e:
            letzte = e
            time.sleep(BACKOFF_S * (2 ** versuch))
    raise letzte  # type: ignore[misc]


def verbindung(pfad: Optional[str] = None) -> sqlite3.Connection:
    """Die nur-lesende Standardverbindung des Hauses. `pfad=None` (Vorgabe)
    liest das Modul-Global DB_PATH bei JEDEM Aufruf frisch nach, statt einen
    an Importzeit gebundenen Default zu benutzen -- PHASE 19: die
    Selbstheilung (_db_selbstheilung_sicherstellen()) kann DB_PATH innerhalb
    EINES Hook-Prozesslaufs neu setzen (Zeiger/Rueckfallsuche/Schnappschuss/
    frische DB), und alle Aufrufer weiter unten im selben Lauf muessen das
    sofort sehen, nicht den Pfad von vor der Heilung."""
    return _verbinden(pfad if pfad is not None else DB_PATH)


def _verbinden_schreibend(pfad: str = DB_PATH) -> sqlite3.Connection:
    letzte = None
    for versuch in range(VERSUCHE):
        try:
            c = sqlite3.connect(pfad, timeout=BUSY_MS / 1000.0)
            c.execute("PRAGMA busy_timeout=%d" % BUSY_MS)
            return c
        except sqlite3.Error as e:
            letzte = e
            time.sleep(BACKOFF_S * (2 ** versuch))
    raise letzte  # type: ignore[misc]


def schreib_verbindung(pfad: Optional[str] = None) -> sqlite3.Connection:
    """Die schreibende Standardverbindung des Hauses. Gleiche Begruendung
    fuer `pfad=None` wie bei verbindung() oben (PHASE 19-Selbstheilung)."""
    return _verbinden_schreibend(pfad if pfad is not None else DB_PATH)

_KONFIG_CACHE: Optional[dict] = None


def _konfig_parsen(wert: str, typ: str, rueckfall):
    try:
        if typ == "float":
            return float(wert)
        if typ == "int":
            return int(wert)
        if typ == "bool":
            return wert.strip().lower() in ("1", "true", "wahr", "ja")
        if typ == "json":
            return json.loads(wert)
        return wert  # "str"
    except Exception:
        return rueckfall


def _konfig() -> dict:
    """Ein Cache je Prozess (Modul-Global) -- kein Aufrufer soll sich um das
    Zwischenspeichern kuemmern muessen. Fehlende Zeilen: Rueckfallwert
    benutzen UND (best effort) nachtragen, damit die naechste Aenderung
    einen Ort in der DB findet statt im Code gesucht werden zu muessen."""
    global _KONFIG_CACHE
    if _KONFIG_CACHE is not None:
        return _KONFIG_CACHE
    ergebnis = {k: v[0] for k, v in _KONFIG_DEFAULTS.items()}
    fehlend = []
    try:
        with contextlib.closing(verbindung()) as c:
            vorhanden = {
                row[0]: (row[1], row[2])
                for row in c.execute("SELECT schluessel, wert, typ FROM star_konfig")
            }
        for schluessel, (rueckfall, typ) in _KONFIG_DEFAULTS.items():
            if schluessel in vorhanden:
                roh, gespeicherter_typ = vorhanden[schluessel]
                ergebnis[schluessel] = _konfig_parsen(roh, gespeicherter_typ or typ, rueckfall)
            else:
                fehlend.append(schluessel)
    except Exception as e:
        _log(f"konfig lesen fehler, bleibe bei Rueckfallwerten: {e}")
        fehlend = []

    if fehlend:
        try:
            with contextlib.closing(schreib_verbindung()) as c:
                jetzt = _jetzt()
                for schluessel in fehlend:
                    rueckfall, typ = _KONFIG_DEFAULTS[schluessel]
                    roh = json.dumps(rueckfall, ensure_ascii=False) if typ == "json" else str(rueckfall)
                    c.execute(
                        "INSERT OR IGNORE INTO star_konfig (schluessel, wert, typ, quelle, geaendert) "
                        "VALUES (?,?,?,?,?)",
                        (schluessel, roh, typ, "code-rueckfallwert", jetzt),
                    )
                c.commit()
        except Exception as e:
            _log(f"konfig nachtragen fehler: {e}")

    _KONFIG_CACHE = ergebnis
    return ergebnis


def _konfig_setzen(schluessel: str, wert, typ: str, quelle: str = "manuell") -> None:
    roh = json.dumps(wert, ensure_ascii=False) if typ == "json" else str(wert)
    with contextlib.closing(schreib_verbindung()) as c:
        c.execute(
            "INSERT INTO star_konfig (schluessel, wert, typ, quelle, geaendert) VALUES (?,?,?,?,?) "
            "ON CONFLICT(schluessel) DO UPDATE SET wert=excluded.wert, typ=excluded.typ, "
            "quelle=excluded.quelle, geaendert=excluded.geaendert",
            (schluessel, roh, typ, quelle, _jetzt()),
        )
        c.commit()
    global _KONFIG_CACHE
    _KONFIG_CACHE = None  # naechster _konfig()-Aufruf liest frisch

def validiere_verbindung(c: sqlite3.Connection) -> dict:
    """Prueft ob die Verbindung die kritischen Eigenschaften hat."""
    ergebnis = {"ok": True, "mode": "unbekannt", "query_only": None, "busy_timeout": None, "fehler": []}
    try:
        mode_row = c.execute("PRAGMA journal_mode").fetchone()
        ergebnis["mode"] = mode_row[0] if mode_row else "unbekannt"
        qo_row = c.execute("PRAGMA query_only").fetchone()
        ergebnis["query_only"] = qo_row[0] if qo_row else None
        bt_row = c.execute("PRAGMA busy_timeout").fetchone()
        ergebnis["busy_timeout"] = bt_row[0] if bt_row else None
        if ergebnis["query_only"] != 0:
            ergebnis["fehler"].append(f"query_only={ergebnis['query_only']} — erwartet 0 für Schreibzugriff")
            ergebnis["ok"] = False
        if not ergebnis["busy_timeout"] or ergebnis["busy_timeout"] < 5000:
            ergebnis["fehler"].append(f"busy_timeout={ergebnis['busy_timeout']}ms — erwartet >=5000ms")
            ergebnis["ok"] = False
        if ergebnis["mode"] not in ("wal", "memory"):
            ergebnis["fehler"].append(f"journal_mode={ergebnis['mode']} — erwartet 'wal' oder 'memory'")
            ergebnis["ok"] = False
    except sqlite3.Error as e:
        ergebnis["fehler"].append(f"Prüfung fehlgeschlagen: {e}")
        ergebnis["ok"] = False
    return ergebnis


def validiere_tabelle_existiert(c: sqlite3.Connection, tabellenname: str) -> bool:
    try:
        result = c.execute(
            "SELECT COUNT(*) FROM sqlite_master WHERE type='table' AND name=?", (tabellenname,)
        ).fetchone()
        return result[0] > 0 if result else False
    except sqlite3.Error:
        return False


def validiere_db_schema(pfad: str = _FEUERSCHNEISE_DB_PATH) -> dict:
    ergebnis = {"ok": True, "tabellen": {}, "fehler": []}
    kritische_tabellen = ["artikel", "kante", "artikel_sphaere", "wm_topologie", "wm_lesart", "kuration"]
    try:
        c = verbindung(pfad)
        for tabelle in kritische_tabellen:
            existiert = validiere_tabelle_existiert(c, tabelle)
            ergebnis["tabellen"][tabelle] = existiert
            if not existiert:
                ergebnis["fehler"].append(f"Tabelle fehlt: {tabelle}")
                ergebnis["ok"] = False
        c.close()
    except sqlite3.Error as e:
        ergebnis["fehler"].append(f"Schema-Prüfung fehlgeschlagen: {e}")
        ergebnis["ok"] = False
    return ergebnis


def pruefe_kondition(kondition: str, kontext: Optional[dict] = None) -> dict:
    """Akzeptierte Konditionen: verbindung_lesend / verbindung_schreibend / schema_komplett."""
    ergebnis = {"kondition": kondition, "ok": False, "details": {}, "fehler": []}
    try:
        if kondition == "verbindung_lesend":
            c = verbindung()
            ergebnis["details"] = validiere_verbindung(c)
            ergebnis["ok"] = ergebnis["details"]["ok"]
            c.close()
        elif kondition == "verbindung_schreibend":
            c = schreib_verbindung()
            ergebnis["details"] = validiere_verbindung(c)
            ergebnis["ok"] = ergebnis["details"]["ok"]
            c.close()
        elif kondition == "schema_komplett":
            ergebnis["details"] = validiere_db_schema()
            ergebnis["ok"] = ergebnis["details"]["ok"]
        else:
            ergebnis["fehler"].append(f"Unbekannte Kondition: {kondition}")
    except Exception as e:
        ergebnis["fehler"].append(f"Prüfung fehlgeschlagen: {e}")
    return ergebnis

_ZAHL_RE = re.compile(r"\d+(?:[.,]\d+)?")
_QLABEL_RE = re.compile(r"\bQ\d+\b")


def zahl_marken(text: str) -> list:
    return _ZAHL_RE.findall(_QLABEL_RE.sub(" ", text or ""))


def zahlen_schluessel(roh: str) -> set:
    k = {roh, roh.replace(",", ".")}
    nur = re.sub(r"[.,]", "", roh)
    if nur:
        k.add(nur)
        k.add(nur.lstrip("0") or "0")
    try:
        f = float(roh.replace(",", "."))
        k.add(("%d" % f) if f == int(f) else repr(f))
    except ValueError:
        pass
    return k


def zahlen(text: str) -> set:
    raus = set()
    for m in zahl_marken(text):
        raus |= zahlen_schluessel(m)
    return raus

def _mr_kurz(text, cap: int) -> str:
    t = " ".join(str(text or "").split())
    return t[:cap]


def _mr_beleg_texte(beleg) -> list:
    if isinstance(beleg, list):
        return [str(b.get("ref", "")) for b in beleg if isinstance(b, dict)]
    if isinstance(beleg, str) and beleg:
        return [beleg]
    return []


def _mr_aussage_text(e: dict) -> str:
    if e.get("aussage"):
        return str(e["aussage"])
    von, nach, art = e.get("von"), e.get("nach"), e.get("art")
    if von and nach and art:
        belege = _mr_beleg_texte(e.get("beleg"))
        kern = "%s %s %s" % (von, art, nach)
        return (kern + " -- " + " / ".join(belege)) if belege else kern
    return ""


def _mr_quelltext(eintraege: list) -> str:
    teile = []
    for e in eintraege:
        teile.append(_mr_aussage_text(e))
        teile.append(str(e.get("gilt_fuer", "")))
        teile.extend(_mr_beleg_texte(e.get("beleg")))
        if e.get("von"):
            teile.append(str(e.get("von")))
        if e.get("nach"):
            teile.append(str(e.get("nach")))
    return " ".join(teile)


def _mr_seite(eintraege: list, label: str, cap_seite: int, cap_zeichen: int) -> tuple:
    gesehen: set = set()
    out = []
    vermerke = []
    uebersprungen_leer = 0
    for e in eintraege:
        txt = _mr_kurz(_mr_aussage_text(e), cap_zeichen)
        if not txt:
            uebersprungen_leer += 1
            continue
        if txt in gesehen:
            continue
        gesehen.add(txt)
        out.append({"text": txt, "ref": e.get("ref"), "art": e.get("art"), "status": e.get("status")})
        if len(out) >= cap_seite:
            break
    if len(eintraege) > len(out):
        vermerke.append("%s: %d von %d verwendet (Cap %d, Dubletten oder ohne jede Kennung "
                         "%d)" % (label, len(out), len(eintraege), cap_seite, uebersprungen_leer))
    return out, vermerke


def _mr_erfindungsmelder(seite: list, quellzahlen: set) -> tuple:
    behalten, verworfen = [], []
    for s in seite:
        marken = zahl_marken(s["text"])
        erfunden = [m for m in marken if not (zahlen_schluessel(m) & quellzahlen)]
        if erfunden:
            verworfen.append("verworfen (Erfindungsmelder, Zahl ohne Quelle %r): %s"
                              % (erfunden, s["text"][:80]))
            continue
        behalten.append(s)
    return behalten, verworfen

@dataclass(frozen=True)
class NeedleDecision:
    route: str
    score: float
    context_budget: int
    collision_risk: bool
    unresolved_hyperposition: bool
    effect_candidate: bool
    reasons: tuple

    def as_dict(self) -> dict:
        return asdict(self)


class NeedleRouter:
    """Small deterministic routing hint; it does not mint permissions or truth.

    In the single-Claude ClayVerse target, Needle is not a model chooser. It estimates
    context/coordination pressure so the main reasoning loop can distinguish ordinary
    turns from collision-heavy, unresolved or likely-effectful work while all real
    authority remains in EffectGate/EntityOS.
    """

    EFFECT_WORDS = (
        "write", "change", "edit", "delete", "deploy", "install", "restart", "stop",
        "start", "send", "create", "commit", "push", "merge", "run", "execute",
        "ändern", "loeschen", "löschen", "bauen", "deployen", "installieren", "starten",
        "stoppen", "senden", "erstellen", "committen", "ausführen", "ausfuehren",
    )

    @classmethod
    def analyze(cls, prompt: str, world: dict, hyperpositions: list, resources: list) -> dict:
        collision_count = len(world.get("cross_session_collision", []))
        memory_count = len(world.get("durable_memory", []))
        tokenish = len(prompt.split())
        effect_candidate = any(word in prompt.lower() for word in cls.EFFECT_WORDS)
        unresolved = bool(hyperpositions)
        reasons: list = []
        score = 0.10
        score += min(0.25, tokenish / 800)
        score += min(0.20, len(resources) * 0.04)
        score += min(0.20, memory_count / 250)
        if collision_count:
            score += 0.25
            reasons.append("cross_session_collision")
        if unresolved:
            score += 0.20
            reasons.append("unresolved_hyperposition")
        if effect_candidate:
            score += 0.15
            reasons.append("effect_language")
        if len(resources) > 3:
            reasons.append("multi_resource")
        if tokenish > 300:
            reasons.append("long_input")
        score = min(1.0, score)

        if collision_count or unresolved:
            route = "deep_shared_state"
            budget = 24000
        elif effect_candidate:
            route = "effect_aware"
            budget = 16000
        elif score >= 0.45:
            route = "deep_local"
            budget = 16000
        else:
            route = "direct_local"
            budget = 8000
        return NeedleDecision(
            route=route,
            score=round(score, 4),
            context_budget=budget,
            collision_risk=collision_count > 0,
            unresolved_hyperposition=unresolved,
            effect_candidate=effect_candidate,
            reasons=tuple(reasons),
        ).as_dict()

def moeglichkeitsraum(text: str, *, budget: int = 8, wirt: Optional[str] = None,
                       conn=None, pkt: Optional[dict] = None,
                       session_id: Optional[str] = None) -> dict:
    """DIE Schnittstelle. Frage rein, ein MOEGLICHKEITSRAUM raus -- oder ein
    expliziter Befund, warum keiner gebaut werden konnte. NIE eine einseitige
    Behauptung als Raum ausgegeben (Invariante F24).

    P5d ERGAENZUNG: `session_id`, wenn gegeben, haengt an JEDEM Rueckgabepfad
    ein `sperrhinweise`-Feld an -- aktive Datei-Pachten ANDERER Sitzungen,
    rein informativ (siehe leases-Abschnitt oben). Kein Blockieren, keine
    Aenderung an raum/einseitig/befund. Backward-kompatibel: ohne
    session_id (alter Aufrufstil) bleibt sperrhinweise eine leere Liste.

    PHASE 4b/4-Toetungskriterium: zwischenzeitlich reichte diese Funktion
    `session_id` (musterkonform gemacht) auch als `sitzung` an
    _paket.paket() durch, damit wegehierarchie.ist_echte_sitzung() ueberhaupt
    feuern konnte -- das hat den Kategorienfehler aus Konzeptdokument §2.4
    Punkt 2 nachweisbar behoben (47 echte Tunnel ueber 5 Sitzungen gemessen).
    PHASE 4 hat danach trotzdem GLEICHVERTEILT gemessen und wegehierarchie
    ersatzlos entfernt (siehe Kommentarblock vor pruefen()) -- `sitzung=`
    wird deshalb NICHT mehr durchgereicht: paket.py's eigener F94-Aufruf
    faengt die fehlende wegehierarchie.py zwar fail-closed ab, wuerde bei
    JEDEM Aufruf aber eine stderr-Zeile schreiben, fuer einen Pfad, den es
    nicht mehr gibt."""
    hinweise = lease_hinweise(ausser_sitzung=session_id) if session_id else []
    if _paket is None:
        return {"raum": None, "einseitig": None, "befund": "paket-Modul nicht geladen",
                "sperrhinweise": hinweise}
    k = _konfig()
    cap_seite = k["raum_moeglichkeit.cap_seite"]
    cap_zeichen = k["raum_moeglichkeit.cap_zeichen_je_aussage"]
    if pkt is None:
        pkt = _paket.paket(text, budget=budget, wirt=wirt, conn=conn)
    t0 = time.perf_counter_ns()

    roh_hyp = pkt.get("hypothesen") or []
    roh_gegen = pkt.get("gegenseite") or []

    basis = {
        "paket_id": pkt.get("paket_id"),
        "paket_latenz_us": pkt.get("latenz_us"),
        "quelle_treffer": len(pkt.get("treffer") or []),
        "quelle_hypothesen": len(roh_hyp),
        "quelle_gegenseite": len(roh_gegen),
        "sperrhinweise": hinweise,
    }

    if not roh_hyp and not roh_gegen:
        return {
            "raum": None, "einseitig": None,
            "befund": "kein Material -- paket() lieferte weder Hypothese noch Gegenseite",
            **basis,
            "latenz_zusatz_us": int((time.perf_counter_ns() - t0) // 1000),
        }

    hyp_seite, hyp_vermerk = _mr_seite(roh_hyp, "hypothesenseite", cap_seite, cap_zeichen)
    gegen_seite, gegen_vermerk = _mr_seite(roh_gegen, "gegenseite", cap_seite, cap_zeichen)

    quelltext = _mr_quelltext(roh_hyp) + " " + _mr_quelltext(roh_gegen)
    quellzahlen = zahlen(quelltext)
    hyp_seite, hyp_raus = _mr_erfindungsmelder(hyp_seite, quellzahlen)
    gegen_seite, gegen_raus = _mr_erfindungsmelder(gegen_seite, quellzahlen)

    verworfen = hyp_vermerk + gegen_vermerk + hyp_raus + gegen_raus

    if not hyp_seite and not gegen_seite:
        return {
            "raum": None, "einseitig": None,
            "befund": "beide Seiten nach Pruefung leer (Dubletten/Erfindungsmelder) -- kein Raum",
            **basis, "verworfen": verworfen,
            "latenz_zusatz_us": int((time.perf_counter_ns() - t0) // 1000),
        }

    gegenseite_fehlt = len(gegen_seite) == 0
    hypothesenseite_fehlt = len(hyp_seite) == 0

    raum = {
        "feld": "moeglichkeitsraum",
        "plausibel_richtung": [s["text"] for s in hyp_seite],
        "plausibel_ref": [s["ref"] for s in hyp_seite if s.get("ref")],
        "spricht_dagegen": [s["text"] for s in gegen_seite],
        "spricht_dagegen_ref": [s["ref"] for s in gegen_seite if s.get("ref")],
        "hinweis": ("Hypothese UND Gegenhypothese zusammen -- keine Seite ist die Antwort, "
                    "keine Seite ist bevorzugt. Diskriminieren, nicht bestaetigen."),
    }
    if gegenseite_fehlt:
        raum["warnung"] = "Gegenseite fehlt -- nur Hypothesenseite vorhanden, KEIN vollstaendiger Raum"
    elif hypothesenseite_fehlt:
        raum["warnung"] = "Hypothesenseite fehlt -- nur Gegenseite vorhanden, KEIN vollstaendiger Raum"

    return {
        "raum": raum,
        "einseitig": bool(gegenseite_fehlt or hypothesenseite_fehlt),
        **basis,
        "verwendet_hypothesen": len(hyp_seite), "verwendet_gegenseite": len(gegen_seite),
        "verworfen": verworfen,
        "caps": {"seite": cap_seite, "zeichen_je_aussage": cap_zeichen},
        "latenz_zusatz_us": int((time.perf_counter_ns() - t0) // 1000),
    }

def _sicht_mvc0_vektor(im) -> list:
    """MVC0-Merkmalsvektor, reines Python + Pillow, KEIN numpy (§5bis.7).
    Bitgenau nachgebaut aus frank-api:/opt/retina-lokal/app.py:144-177 bzw.
    research_entity/harness/reference/digital_retina_runtime.py:_features
    (dieselbe Formel, wortgleiche Uebernahme dort). 88 Dimensionen:
    rgb_mean(3) + rgb_std(3) + hist(8) + quads(4) + edges(2) + centroid(2) +
    asym(2) + 8x8-Miniaturbild(64). Siehe Kennungs-Hinweis am Abschnittskopf
    zur Verfahrenskennung `retina-mvc0-pixel-purepy-v1`."""
    im = im.convert("RGB")
    small = im.resize((32, 32), Image.Resampling.BILINEAR)
    px = list(small.getdata())
    n = len(px)

    sums = [0.0, 0.0, 0.0]
    for r, g, b in px:
        sums[0] += r / 255.0
        sums[1] += g / 255.0
        sums[2] += b / 255.0
    means = [s / n for s in sums]
    sq = [0.0, 0.0, 0.0]
    for r, g, b in px:
        sq[0] += (r / 255.0 - means[0]) ** 2
        sq[1] += (g / 255.0 - means[1]) ** 2
        sq[2] += (b / 255.0 - means[2]) ** 2
    stds = [math.sqrt(s / n) for s in sq]
    rgb_mean = [m - 0.5 for m in means]
    rgb_std = stds

    gray = [0.0] * n
    for i, (r, g, b) in enumerate(px):
        gray[i] = 0.299 * (r / 255.0) + 0.587 * (g / 255.0) + 0.114 * (b / 255.0)

    def g_at(row, col):
        return gray[row * 32 + col]

    hist = [0] * 8
    for v in gray:
        idx = int(v * 8)
        idx = 7 if idx > 7 else (0 if idx < 0 else idx)
        hist[idx] += 1
    hist_feat = [h / n - (1.0 / 8.0) for h in hist]

    def quad_mean(r0, r1, c0, c1):
        s, cnt = 0.0, 0
        for r in range(r0, r1):
            for c in range(c0, c1):
                s += g_at(r, c)
                cnt += 1
        return s / cnt
    quads = [quad_mean(0, 16, 0, 16) - 0.5, quad_mean(0, 16, 16, 32) - 0.5,
             quad_mean(16, 32, 0, 16) - 0.5, quad_mean(16, 32, 16, 32) - 0.5]

    dx_sum = sum(abs(g_at(r, c + 1) - g_at(r, c)) for r in range(32) for c in range(31))
    dy_sum = sum(abs(g_at(r + 1, c) - g_at(r, c)) for r in range(31) for c in range(32))
    edges = [dx_sum / (32 * 31), dy_sum / (31 * 32)]

    total = sum_x = sum_y = 0.0
    for r in range(32):
        for c in range(32):
            w = g_at(r, c)
            if w < 1e-4:
                w = 1e-4
            total += w
            sum_x += c * w
            sum_y += r * w
    centroid = [(sum_x / total) / 31 - 0.5, (sum_y / total) / 31 - 0.5]

    asym_h = sum(abs(g_at(r, c) - g_at(r, 31 - c)) for r in range(32) for c in range(32))
    asym_v = sum(abs(g_at(r, c) - g_at(31 - r, c)) for r in range(32) for c in range(32))
    asym = [asym_h / n, asym_v / n]

    thumb = im.convert("L").resize((8, 8), Image.Resampling.BILINEAR)
    ds = [(v / 255.0) - 0.5 for v in list(thumb.getdata())]

    vec = rgb_mean + rgb_std + hist_feat + quads + edges + centroid + asym + ds
    norm = math.sqrt(sum(x * x for x in vec))
    if norm < 1e-12:
        return [0.0] * len(vec)
    return [x / norm for x in vec]

def _sicht_kosinus(a, b) -> float:
    if len(a) != len(b) or not a:
        return 0.0
    punkt = sum(x * y for x, y in zip(a, b))
    na = math.sqrt(sum(x * x for x in a))
    nb = math.sqrt(sum(y * y for y in b))
    if na < 1e-12 or nb < 1e-12:
        return 0.0
    return punkt / (na * nb)

# ── Der Abruf VOR der Analyse (Rueckwaertsrichtung, F27) ────────────────
def _sicht_abruf(c, quelle_id: str, frage: str = "", grenze: int = 12) -> dict:
    begriffe: list = []
    bekannt: list = []
    for zeile in c.execute(
            "SELECT o.objekt_id, o.semantische_identitaet, o.erst_gesehen_am, "
            "       s.sichtbarkeit, s.letzte_positive_evidenz_epoche, "
            "       s.stuetzende_hypothese_id "
            "FROM sicht_weltobjekt o "
            "LEFT JOIN sicht_weltobjekt_stand s "
            "  ON s.objekt_id=o.objekt_id AND s.gueltig_bis IS NULL "
            "WHERE o.quelle_id=? ORDER BY o.erst_gesehen_am DESC LIMIT ?",
            (quelle_id, grenze)).fetchall():
        eintrag = dict(zeile)
        bekannt.append(eintrag)
        name = (eintrag.get("semantische_identitaet") or "").strip()
        if name and name not in begriffe:
            begriffe.append(name)
    for wort in (frage or "").replace(",", " ").split():
        wort = wort.strip("?!.:;\"'()").lower()
        if len(wort) >= 4 and wort not in begriffe:
            begriffe.append(wort)
    return {"begriffe": begriffe[:grenze], "bekannt": bekannt, "treffer": len(bekannt)}

def _sicht_evidenz_buchen(c, *, aufnahme_id: str, quelle_id: str, quelle_epoche: int,
                           rohbezug_hash: Optional[str], saetze) -> list:
    jetzt = _jetzt()
    kennungen = []
    for satz in saetze:
        wert = json.dumps(satz["wert"], sort_keys=True, ensure_ascii=False)
        raum = json.dumps(satz["raum_bezug"], sort_keys=True) if satz.get("raum_bezug") else None
        kid = _sicht_evidenz_kennung(aufnahme_id, satz["transformation"],
                                      satz["beobachtungsart"], raum, wert)
        c.execute(
            "INSERT OR IGNORE INTO sicht_evidenz(evidenz_id,aufnahme_id,quelle_id,"
            "quelle_epoche,aufnahme_zeit,eingang_zeit,rohbezug_hash,raum_bezug,"
            "transformation,transformation_version,beobachtungsart,wert,"
            "bottom_up_praezision,kalibrierung_ref,autoritaet,erzeugt_am) "
            "VALUES(?,?,?,?,?,?,?,?,?,?,?,?,?,?,'NUR_BEOBACHTUNG',?)",
            (kid, aufnahme_id, quelle_id, quelle_epoche, satz.get("aufnahme_zeit"),
             jetzt, rohbezug_hash, raum, satz["transformation"],
             satz.get("transformation_version", "?"), satz["beobachtungsart"], wert,
             satz.get("bottom_up_praezision"), satz.get("kalibrierung_ref", "UNKALIBRIERT"), jetzt),

    return kennungen

# ── Die Wiedererkennung -- vier Ausgaenge (Abweichung 2 am Abschnittskopf) ──
def _sicht_zuordnen(c, *, aufnahme_id: str, quelle_id: str, anker: str,
                     vektor: Optional[list], verfahren: str,
                     verfahren_version: str = "") -> dict:
    k = _konfig()
    schwelle = float(k.get("sicht.wiedererkannt_schwelle", 0.96))
    mehrdeutig_abstand = float(k.get("sicht.mehrdeutig_abstand", 0.025))
    jetzt = _jetzt()
    zid = _sicht_kennung()

    def buchen(objekt_id, entscheidung, begruendung, aehnlich, abstand):
        c.execute(
            "INSERT INTO sicht_zuordnung(zuordnung_id,aufnahme_id,anker,objekt_id,"
            "entscheidung,begruendung,aehnlichkeit,abstand_zum_zweiten,verfahren,"
            "verfahren_version,erzeugt_am) VALUES(?,?,?,?,?,?,?,?,?,?,?)",
            (zid, aufnahme_id, anker, objekt_id, entscheidung, begruendung,
             aehnlich, abstand, verfahren, verfahren_version, jetzt),
        )
        return {"zuordnung_id": zid, "objekt_id": objekt_id, "entscheidung": entscheidung,
                "begruendung": begruendung, "aehnlichkeit": aehnlich,
                "abstand_zum_zweiten": abstand, "verfahren": verfahren}

    if not vektor:
        return buchen(None, "UNENTSCHIEDEN_NICHT_MESSBAR",
                       "Kein Merkmalsvektor vorhanden.", None, None)

    kandidaten = []
    for zeile in c.execute(
            "SELECT m.vektor, z.objekt_id FROM sicht_merkmal m "
            "JOIN sicht_zuordnung z ON z.aufnahme_id=m.aufnahme_id AND z.anker=m.anker "
            "JOIN sicht_weltobjekt o ON o.objekt_id=z.objekt_id "
            "WHERE m.verfahren=? AND o.quelle_id=? AND z.objekt_id IS NOT NULL "
            "AND m.aufnahme_id<>?", (verfahren, quelle_id, aufnahme_id)).fetchall():
        kandidaten.append((_sicht_kosinus(vektor, json.loads(zeile["vektor"])), zeile["objekt_id"]))

    if not kandidaten:
        return buchen(None, "UNENTSCHIEDEN_NICHT_MESSBAR",
                       "Kein Vergleichsbestand fuer dieses Verfahren an dieser Quelle -- "
                       "der erste Gegenstand ist nicht 'neu', er ist unvergleichbar.",
                       None, None)

    kandidaten.sort(key=lambda x: (-x[0], x[1]))
    beste, objekt_id = kandidaten[0]
    zweite = kandidaten[1][0] if len(kandidaten) > 1 else -1.0
    abstand = round(beste - zweite, 6)

    if beste >= schwelle and abstand < mehrdeutig_abstand:
        return buchen(None, "MEHRDEUTIG",
                       f"Zwei Gegenstaende liegen naeher als {mehrdeutig_abstand} "
                       f"beieinander ({beste:.4f} vs {zweite:.4f}). Schwellen: "
                       f"wiedererkannt_schwelle={schwelle}, mehrdeutig_abstand={mehrdeutig_abstand}.",
                       round(beste, 6), abstand)
    if beste >= schwelle:
        return buchen(objekt_id, "WIEDERERKANNT_ZU_BESTAETIGEN",
                       f"Aehnlichkeit {beste:.4f} ueber Schwelle {schwelle} -- "
                       f"wahrscheinlich derselbe, Bestaetigung durch Claude noch offen. "
                       f"Schwellen: wiedererkannt_schwelle={schwelle}, mehrdeutig_abstand={mehrdeutig_abstand}.",
                       round(beste, 6), abstand)
    return buchen(None, "NEU_ANGELEGT",
                  f"Beste Aehnlichkeit im Bestand nur {beste:.4f}, unter {schwelle}. "
                  f"Verglichen gegen {len(kandidaten)} bekannte Gegenstaende. "
                  f"Schwellen: wiedererkannt_schwelle={schwelle}, mehrdeutig_abstand={mehrdeutig_abstand}.",
                  round(beste, 6), abstand)

# ── Ebene 2 (Hypothesen) -- hypothese_bilden / verengen (F57) ───────────
def _sicht_hypothese_bilden(c, *, aufnahme_id: str, anker: str, art: str,
                             lesarten: list, stuetzende_evidenz: list,
                             bestand_stuetze: Optional[list] = None,
                             zustand: str = "VORLAEUFIG",
                             vorgaenger_id: Optional[str] = None) -> str:
    jetzt = _jetzt()
    hid = _sicht_kennung()
    gesamt = sum(max(0.0, float(l.get("p", 0.0))) for l in lesarten) or 1.0
    aus_bestand = sum(max(0.0, float(l.get("p", 0.0))) for l in lesarten
                       if l.get("herkunft") == "BESTAND")
    prior_dominanz = round(aus_bestand / gesamt, 6)
    beste = max((float(l.get("p", 0.0)) for l in lesarten), default=0.0)
    revision = 1
    if vorgaenger_id:
        alt = c.execute("SELECT revision FROM sicht_hypothese WHERE hypothese_id=?",
                         (vorgaenger_id,)).fetchone()
        revision = int(alt["revision"]) + 1 if alt else 1
        c.execute("UPDATE sicht_hypothese SET abgeloest_von=? WHERE hypothese_id=?",
                   (hid, vorgaenger_id))
    c.execute(
        "INSERT INTO sicht_hypothese(hypothese_id,aufnahme_id,anker,art,lesarten,"
        "lesarten_geschlossen,stuetzende_evidenz,widersprechende_evidenz,"
        "bestand_stuetze,prior_dominanz,zuversicht,zustand,revision,vorgaenger_id,"
        "abgeloest_von,erzeugt_am) VALUES(?,?,?,?,?,0,?,'[]',?,?,?,?,?,?,NULL,?)",
        (hid, aufnahme_id, anker, art, json.dumps(lesarten, ensure_ascii=False),
         json.dumps(stuetzende_evidenz), json.dumps(bestand_stuetze or []),
         prior_dominanz, round(beste, 6), zustand, revision, vorgaenger_id, jetzt),
    )
    return hid


def _sicht_verengen(c, hypothese_id: str, neue_gewichte: dict, *, stufe: str) -> str:
    """F57, mechanisch: eine spaetere Stufe darf Gewichte verschieben, nie eine
    Lesart entfernen (sicht.py:919-974, wortgleich portiert)."""
    zeile = c.execute("SELECT * FROM sicht_hypothese WHERE hypothese_id=?",
                       (hypothese_id,)).fetchone()
    if zeile is None:
        raise SichtFehler(f"Hypothese {hypothese_id} gibt es nicht")
    if zeile["abgeloest_von"]:
        raise SichtFehler("Hypothese bereits abgeloest -- verenge die juengste Revision")
    alt = json.loads(zeile["lesarten"])
    alte_menge = {l["lesart"] for l in alt}
    fehlend = alte_menge - set(neue_gewichte)
    if fehlend:
        raise SichtFehler(f"F57: Stufe {stufe} darf keine Lesart entfernen: {sorted(fehlend)}")
    neu = []
    for l in alt:
        kopie = dict(l)
        kopie["p"] = round(max(0.0, float(neue_gewichte[l["lesart"]])), 6)
        kopie["verengt_von"] = stufe
        neu.append(kopie)
    for name in sorted(set(neue_gewichte) - alte_menge):
        neu.append({"lesart": name, "p": round(float(neue_gewichte[name]), 6),
                    "aus": [], "herkunft": stufe})
    return _sicht_hypothese_bilden(
        c, aufnahme_id=zeile["aufnahme_id"], anker=zeile["anker"], art=zeile["art"],
        lesarten=neu, stuetzende_evidenz=json.loads(zeile["stuetzende_evidenz"]),
        bestand_stuetze=json.loads(zeile["bestand_stuetze"]), zustand=zeile["zustand"],
        vorgaenger_id=hypothese_id)

def _sicht_stand_setzen(c, *, objekt_id: str, sichtbarkeit: str, grund: str,
                         aufnahme_id: Optional[str] = None,
                         hypothese_id: Optional[str] = None,
                         epoche: Optional[int] = None,
                         ungewissheit: Optional[float] = None,
                         turn_id: Optional[str] = None,
                         causal_id: Optional[str] = None) -> str:
    zeile = c.execute(
        "SELECT sichtbarkeit,belegt_abwesenheit FROM sicht_sichtbarkeit WHERE sichtbarkeit=?",
        (sichtbarkeit,)).fetchone()
    if zeile is None:
        raise SichtFehler(f"Unbekannter Sichtbarkeitszustand: {sichtbarkeit}")
    if int(zeile["belegt_abwesenheit"]) == 1 and not (grund or "").strip():
        raise SichtFehler(
            f"{sichtbarkeit} verlangt einen Beleg (Grund) -- ein Nichtfund ist kein "
            "Abwesenheitsbeleg, er gehoert nach UNBEKANNT.")
    jetzt = _jetzt()

        "SELECT * FROM sicht_weltobjekt_stand WHERE objekt_id=? AND gueltig_bis IS NULL",

                   (jetzt, vorher["stand_id"]))
    letzte_positive = epoche if epoche is not None else (
        vorher["letzte_positive_evidenz_epoche"] if vorher is not None else None)
    sid = _sicht_kennung()
    c.execute(

        "VALUES(?,?,?,NULL,?,?,'[]','[]',?,?,?,?,?,?,?)",
        (sid, objekt_id, jetzt, sichtbarkeit, grund, letzte_positive, ungewissheit,
         hypothese_id, aufnahme_id, turn_id, causal_id, jetzt),
    )
    return sid

def _sicht_bucket(wert_0_255: int) -> str:
    v = wert_0_255 / 255.0
    if v < 0.33:
        return "dunkel"
    if v < 0.66:
        return "mittel"
    return "hell"


def _sicht_bildpfad_aufloesen(pfad: Optional[str], bild: Optional[str],
                               session_override: Optional[str]) -> Path:
    if pfad:
        p = Path(pfad).expanduser()
        if not p.is_file():
            raise SichtFehler(f"Datei nicht gefunden: {p}")
        return p
    if bild:
        session_id = session_override or os.environ.get("CLAUDE_CODE_SESSION_ID")
        if not session_id:
            raise SichtFehler(
                "--bild braucht eine Sitzung (CLAUDE_CODE_SESSION_ID nicht gesetzt) -- "
                "notfalls --session <id> mitgeben.")
        basis = os.environ.get("CLAUDE_IMAGE_CACHE") or str(Path.home() / ".claude" / "image-cache")
        p = Path(basis) / session_id / f"{bild}.png"
        if not p.is_file():
            raise SichtFehler(f"Bild #{bild} nicht im image-cache gefunden: {p}")
        return p
    raise SichtFehler("Weder <pfad> noch --bild angegeben -- nichts zum Ansehen.")


def sicht_vorbereiten(*, pfad: Optional[str] = None, bild: Optional[str] = None,
                       frage: str = "", quelle: Optional[str] = None,
                       session_override: Optional[str] = None) -> dict:
    """Teil A (§5bis.5): rein mechanisch, kein LLM, kein Netz."""
    if not _PIL_OK:
        raise SichtFehler("Pillow nicht ladbar -- SICHT-Abschnitt kann kein Bild dekodieren.")
    p = _sicht_bildpfad_aufloesen(pfad, bild, session_override)
    roh = p.read_bytes()
    art = _sicht_bildart(roh)
    if art is None:
        raise SichtFehler(
            f"{p} ist keine erkannte Bilddatei (PNG/JPEG/GIF/BMP/WEBP) -- "
            "'Evidenz ueber nichts' (sicht.py:1258-1264), wird nicht analysiert.")
    bild_hash = hashlib.sha256(roh).hexdigest()
    im = Image.open(io.BytesIO(roh)).convert("RGB")
    breite, hoehe = im.width, im.height
    rahmen_hash = hashlib.sha256(f"{breite}x{hoehe}:RGB:".encode() + im.tobytes()).hexdigest()
    vektor = _sicht_mvc0_vektor(im)
    quelle_id = quelle or "default"

    session_id = session_override or os.environ.get("CLAUDE_CODE_SESSION_ID")
    turn_id = causal_id = None
    with contextlib.closing(schreib_verbindung()) as c:
        c.row_factory = sqlite3.Row
        _sicht_stammdaten_sicherstellen(c)
        if session_id:
            aktiv = c.execute("SELECT turn_id, causal_id FROM active_turns WHERE session_id=?",
                               (session_id,)).fetchone()
            if aktiv:
                turn_id, causal_id = aktiv["turn_id"], aktiv["causal_id"]

        quelle_epoche = 1 + (c.execute(
            "SELECT COUNT(*) AS n FROM sicht_aufnahme WHERE quelle_id=?",
            (quelle_id,)).fetchone()["n"])

        abruf = _sicht_abruf(c, quelle_id, frage)

        aufnahme_id = _sicht_kennung()
        _sicht_merkmal_buchen(c, aufnahme_id=aufnahme_id, anker="ganzes_bild",
                               verfahren=SICHT_VERFAHREN, dimension=len(vektor),
                               vektor=vektor, rahmen_hash=rahmen_hash)
        zuordnung = _sicht_zuordnen(c, aufnahme_id=aufnahme_id, quelle_id=quelle_id,
                                     anker="ganzes_bild", vektor=vektor,
                                     verfahren=SICHT_VERFAHREN)

        # Blindheitsprobe (§5bis.7-Korrektur): zwei Positionen im 8x8-Raster,
        # deterministisch aus bild_hash, Wert wird NUR intern gespeichert.
        thumb = _sicht_thumbnail_8x8(im)
        hb = bytes.fromhex(bild_hash)
        pos1 = (hb[0] % 8, hb[1] % 8)
        pos2 = (hb[2] % 8, hb[3] % 8)
        if pos2 == pos1:
            pos2 = ((pos2[0] + 3) % 8, pos2[1])
        sichtprobe = {
            "positionen": [list(pos1), list(pos2)],
            "werte": [_sicht_bucket(thumb[pos1[1] * 8 + pos1[0]]),
                      _sicht_bucket(thumb[pos2[1] * 8 + pos2[0]])],
        }

        abruf_kontext = {
            "begriffe": abruf["begriffe"], "bekannt": abruf["bekannt"],
            "quelle_epoche": quelle_epoche, "sichtprobe": sichtprobe,
            "zuordnung_kandidat": zuordnung.get("objekt_id"),
        }
        jetzt = _jetzt()
        c.execute(
            "INSERT INTO sicht_aufnahme(aufnahme_id,quelle_id,quelle_klasse,bild_hash,"
            "breite,hoehe,user_id,turn_id,causal_id,session_id,stufe,abruf_kontext,"
            "abruf_treffer,ergebnis,begonnen_am) "
            "VALUES(?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)",
            (aufnahme_id, quelle_id, "bild", bild_hash, breite, hoehe, USER_ID,
             turn_id, causal_id, session_id, "CLAUDE_SICHT",
             json.dumps(abruf_kontext, ensure_ascii=False), abruf["treffer"],
             "LAEUFT", jetzt),
        )
        c.commit()

    zeilen = [
        f"SICHT vorbereitet -- aufnahme_id={aufnahme_id}",
        f"Pfad: {p}",
        f"Bild: {breite}x{hoehe}, bild_hash={bild_hash[:16]}…, rahmen_hash={rahmen_hash[:16]}…",
        f"Stufe 1 (MVC0_PIXEL) Duplikaturteil: {zuordnung['entscheidung']} "
        f"(aehnlichkeit={zuordnung['aehnlichkeit']}, abstand_zum_zweiten={zuordnung['abstand_zum_zweiten']})",
        f"  {zuordnung['begruendung']}",
    ]
    if abruf["bekannt"]:
        zeilen.append(f"Bekannte Gegenstaende an Quelle '{quelle_id}' (bis 5 von {abruf['treffer']}):")
        for e in abruf["bekannt"][:5]:
            datum = time.strftime("%Y-%m-%d", time.localtime(e["erst_gesehen_am"]))
            name = e.get("semantische_identitaet") or "(unbenannt)"
            stand = e.get("sichtbarkeit") or "UNBEKANNT"
            zeilen.append(f"  - {e['objekt_id'][:12]}… '{name}', zuerst {datum}, letzter Glaube: {stand}")
            hid = e.get("stuetzende_hypothese_id")
            if hid:
                with contextlib.closing(verbindung()) as c2:
                    c2.row_factory = sqlite3.Row
                    hyp = c2.execute(
                        "SELECT lesarten FROM sicht_hypothese WHERE hypothese_id=? "
                        "AND abgeloest_von IS NULL", (hid,)).fetchone()
                    if hyp:
                        zeilen.append(f"    offene Hypothese: {hyp['lesarten']}")
    else:
        zeilen.append(f"Keine bekannten Gegenstaende an Quelle '{quelle_id}' -- erster Blick hier.")
    if abruf["begriffe"]:
        zeilen.append(f"Messauftrag (begriffe): {', '.join(abruf['begriffe'])}")
    zeilen.append("")
    zeilen.append(_SICHT_GRENZAUSSAGE)
    zeilen.append("")
    zeilen.append(
        f"Blindheitsprobe: beschreibe, was an Rasterposition (x={pos1[0]},y={pos1[1]}) "
        f"und (x={pos2[0]},y={pos2[1]}) der 8x8-Miniaturansicht des Bildes zu sehen ist "
        "(je 'dunkel'/'mittel'/'hell'), als Teil von 'gesehen_beleg' beim Buchen -- "

    return {"stand": "OK", "aufnahme_id": aufnahme_id, "text": "\n".join(zeilen)}


# ── Teil C: stern sicht-buchen ───────────────────────────────────────────
def sicht_buchen(aufnahme_id: str, nutzlast: dict) -> dict:
    with contextlib.closing(schreib_verbindung()) as c:
        c.row_factory = sqlite3.Row
        aufnahme = c.execute("SELECT * FROM sicht_aufnahme WHERE aufnahme_id=?",
                              (aufnahme_id,)).fetchone()
        if aufnahme is None:
            return {"stand": "FEHLER", "fehler": f"aufnahme_id unbekannt: {aufnahme_id}"}
        if aufnahme["ergebnis"] != "LAEUFT":
            return {"stand": "FEHLER",
                    "fehler": f"aufnahme_id {aufnahme_id} ist bereits {aufnahme['ergebnis']}"}

        abruf_kontext = json.loads(aufnahme["abruf_kontext"] or "{}")
        sichtprobe = abruf_kontext.get("sichtprobe") or {}
        soll_positionen = sichtprobe.get("positionen") or []
        soll_werte = sichtprobe.get("werte") or []
        soll = {tuple(p): w for p, w in zip(soll_positionen, soll_werte)}

        # ── Blindheitstest (§5bis.7-Korrektur), rein mechanisch ──────────
        beleg = nutzlast.get("gesehen_beleg") or []
        ist = {}
        for eintrag in beleg:
            try:
                pos = tuple(eintrag["pos"])
                ist[pos] = str(eintrag["wert"]).strip().lower()
            except Exception:
                continue
        hingesehen = bool(soll) and all(ist.get(pos) == wert for pos, wert in soll.items())
        if not hingesehen:
            jetzt = _jetzt()
            c.execute(
                "UPDATE sicht_aufnahme SET ergebnis='NICHT_HINGESEHEN', "
                "fehlergrund=?, beendet_am=?, dauer_ms=? WHERE aufnahme_id=?",
                ("gesehen_beleg fehlt oder stimmt nicht mit der Blindheitsprobe ueberein",
                 jetzt, round((jetzt - aufnahme["begonnen_am"]) * 1000, 3), aufnahme_id),
            )
            c.commit()
            return {"stand": "NICHT_HINGESEHEN", "aufnahme_id": aufnahme_id,
                    "fehler": "gesehen_beleg fehlt oder stimmt nicht -- Buchung verworfen, "
                              "keine Evidenz gebucht."}

        # ── Identitaetssperre (harte Grenze §1.9.6) ───────────────────────
        erlaubte_ids = {e["objekt_id"] for e in abruf_kontext.get("bekannt", [])}
        kandidat = abruf_kontext.get("zuordnung_kandidat")
        if kandidat:
            erlaubte_ids.add(kandidat)
        genannte_ids = set()
        for eintrag in (nutzlast.get("zuordnung") and [nutzlast["zuordnung"]] or []):
            if eintrag.get("objekt_id"):
                genannte_ids.add(eintrag["objekt_id"])
        for eintrag in nutzlast.get("sichtbarkeit") or []:
            if eintrag.get("objekt_id"):
                genannte_ids.add(eintrag["objekt_id"])
        unerlaubt = genannte_ids - erlaubte_ids
        if unerlaubt:
            return {"stand": "FEHLER",
                    "fehler": f"Identitaetssperre (§1.9.6): objekt_id nicht in abruf_kontext "
                              f"oder Zuordnungskandidat dieser Aufnahme: {sorted(unerlaubt)}"}

        modell = str(nutzlast.get("modell") or "unbekannt")
        quelle_epoche = int(abruf_kontext.get("quelle_epoche") or 1)

        # ── Evidenz (unveraenderlich) ─────────────────────────────────────
        roh_beobachtungen = nutzlast.get("beobachtungen") or []
        saetze = [{
            "transformation": "claude-sicht-v1", "transformation_version": modell,
            "beobachtungsart": b.get("art", "beobachtung"),
            "wert": {"anker": b.get("anker"), "wert": b.get("wert")},
            "raum_bezug": b.get("raum_bezug"), "kalibrierung_ref": "UNKALIBRIERT",
        } for b in roh_beobachtungen]
        evidenz_ids = _sicht_evidenz_buchen(
            c, aufnahme_id=aufnahme_id, quelle_id=aufnahme["quelle_id"],
            quelle_epoche=quelle_epoche, rohbezug_hash=aufnahme["bild_hash"], saetze=saetze,
        ) if saetze else []
        frische_evidenz = set(evidenz_ids)

        # ── Hypothesen, gruppiert nach anker; prior_dominanz MECHANISCH ──
        je_anker: dict = {}
        for l in nutzlast.get("lesarten") or []:
            je_anker.setdefault(l.get("anker", "ganzes_bild"), []).append(l)
        hypothese_ids = []
        for anker, gruppe in je_anker.items():
            lesarten = []
            for l in gruppe:
                aus = l.get("aus") or []
                herkunft = "BEOBACHTUNG" if any(a in frische_evidenz for a in aus) else "BESTAND"
                lesarten.append({"lesart": l.get("lesart"), "p": l.get("p", 0.0),
                                  "aus": aus, "herkunft": herkunft})
            hid = _sicht_hypothese_bilden(
                c, aufnahme_id=aufnahme_id, anker=anker, art="claude-lesart",
                lesarten=lesarten, stuetzende_evidenz=list(frische_evidenz))
            hypothese_ids.append(hid)

        # ── Zuordnungsurteil: bestaetigt / verworfen / unentschieden ──────
        objekt_id_ergebnis = None
        zuordnung = nutzlast.get("zuordnung") or {}
        if zuordnung:
            urteil = zuordnung.get("urteil")
            oid = zuordnung.get("objekt_id")
            anker = zuordnung.get("anker", "ganzes_bild")
            if urteil == "BESTAETIGT" and oid:
                _sicht_stand_setzen(c, objekt_id=oid, sichtbarkeit="SICHTBAR",
                                     grund="Claude bestaetigt Stufe-1-Wiedererkennung",
                                     aufnahme_id=aufnahme_id,
                                     hypothese_id=hypothese_ids[0] if hypothese_ids else None,
                                     epoche=quelle_epoche, turn_id=aufnahme["turn_id"],
                                     causal_id=aufnahme["causal_id"])
                objekt_id_ergebnis = oid
            elif urteil == "VERWORFEN":
                neues_objekt = _sicht_objekt_anlegen(
                    c, quelle_id=aufnahme["quelle_id"], aufnahme_id=aufnahme_id,
                    semantische_identitaet=(roh_beobachtungen[0].get("wert")
                                             if roh_beobachtungen else None))
                _sicht_stand_setzen(c, objekt_id=neues_objekt, sichtbarkeit="SICHTBAR",
                                     grund="Claude verwirft Stufe-1-Kandidat, neuer Gegenstand",
                                     aufnahme_id=aufnahme_id,
                                     hypothese_id=hypothese_ids[0] if hypothese_ids else None,
                                     epoche=quelle_epoche, turn_id=aufnahme["turn_id"],
                                     causal_id=aufnahme["causal_id"])
                objekt_id_ergebnis = neues_objekt
            if urteil in ("BESTAETIGT", "VERWORFEN", "UNENTSCHIEDEN"):
                c.execute(
                    "INSERT INTO sicht_zuordnung(zuordnung_id,aufnahme_id,anker,objekt_id,"
                    "entscheidung,begruendung,verfahren,verfahren_version,erzeugt_am) "
                    "VALUES(?,?,?,?,?,?,?,?,?)",
                    (_sicht_kennung(), aufnahme_id, anker, objekt_id_ergebnis,
                     f"CLAUDE_{urteil}", "Unabhaengiges Urteil von Claude (Teil C)",
                     "claude-sicht-v1", modell, _jetzt()),
                )
        elif abruf_kontext.get("zuordnung_kandidat") is None and not abruf_kontext.get("bekannt"):
            # Erster Blick an dieser Quelle, kein Urteil mitgeliefert -- kein
            # stilles Erfinden einer Identitaet, siehe Abschnittskopf.
            pass

        # ── Sichtbarkeitsstaende ───────────────────────────────────────────
        for eintrag in nutzlast.get("sichtbarkeit") or []:
            _sicht_stand_setzen(
                c, objekt_id=eintrag["objekt_id"], sichtbarkeit=eintrag["zustand"],
                grund=eintrag.get("grund", ""), aufnahme_id=aufnahme_id,
                epoche=quelle_epoche, turn_id=aufnahme["turn_id"], causal_id=aufnahme["causal_id"],
            )

        jetzt = _jetzt()
        dauer_ms = round((jetzt - aufnahme["begonnen_am"]) * 1000, 3)  # [MODELLIERT]:
        # Laenge eines Gespraechszugs, keine Inferenzzeit -- wird NIE nach
        # sicht_kaskade_stufe.gemessen_ms geschrieben (sicht.py:162-164).
        c.execute(
            "UPDATE sicht_aufnahme SET ergebnis='ERFOLG', beendet_am=?, dauer_ms=? "
            "WHERE aufnahme_id=?", (jetzt, dauer_ms, aufnahme_id),
        )
        c.commit()

    return {"stand": "ERFOLG", "aufnahme_id": aufnahme_id, "evidenz_ids": evidenz_ids,
            "hypothese_ids": hypothese_ids, "objekt_id": objekt_id_ergebnis,
            "dauer_ms": dauer_ms}


def sicht_nutzung_mensch(objekt_id: str, urteil: str) -> dict:
    """`stern sicht-nuetzlich`/`sicht-nutzlos` -- menschliches Signal,
    Vorrang vor dem mechanischen (Teil D)."""
    art = "MENSCH_NUETZLICH" if urteil == "nuetzlich" else "MENSCH_NUTZLOS"
    with contextlib.closing(schreib_verbindung()) as c:
        vorhanden = c.execute("SELECT 1 FROM sicht_weltobjekt WHERE objekt_id=?",
                               (objekt_id,)).fetchone()
        if not vorhanden:
            return {"stand": "FEHLER", "fehler": f"objekt_id unbekannt: {objekt_id}"}
        nid = _sicht_kennung()
        c.execute(
            "INSERT INTO sicht_nutzung(nutzung_id,objekt_id,nutzungsart,beleg,"
            "festgestellt_von,erzeugt_am) VALUES(?,?,?,?,?,?)",
            (nid, objekt_id, art, f"Handbefehl stern sicht-{urteil}", "MENSCH", _jetzt()),
        )
        c.commit()
    return {"stand": "OK", "nutzung_id": nid, "objekt_id": objekt_id, "nutzungsart": art}


# ── Teil D: mechanisches Nutzungssignal, NUTZT den Stop-Hook-Abgleich aus
# P3 mit (kein zweiter Mechanismus, siehe §5bis.5 Teil D). ────────────────
def _sicht_nutzung_hook(session_id: str, transcript_path) -> None:
    """Stop -- fail-silent wie jeder andere Hook-Pfad. Sucht Aufnahmen dieser
    Sitzung, deren Runde noch nicht mechanisch geprueft wurde (kein
    sicht_nutzung-Eintrag zu ihrem turn_id), und prueft, ob die Benennung
    eines bekannten/neu bestaetigten Gegenstands im Antworttext auftaucht."""
    try:
        with contextlib.closing(schreib_verbindung()) as c:
            c.row_factory = sqlite3.Row
            turns = c.execute(
                "SELECT DISTINCT turn_id, quelle_id FROM sicht_aufnahme "
                "WHERE session_id=? AND ergebnis='ERFOLG' AND turn_id IS NOT NULL "
                "AND turn_id NOT IN (SELECT DISTINCT turn_id FROM sicht_nutzung "
                "WHERE turn_id IS NOT NULL)", (session_id,),
            ).fetchall()
            if not turns:
                return
            assistent_text = _transcript_assistenttext_lesen(transcript_path).lower()
            jetzt = _jetzt()
            for row in turns:
                turn_id, quelle_id = row["turn_id"], row["quelle_id"]
                objekte = c.execute(
                    "SELECT DISTINCT o.objekt_id, o.semantische_identitaet FROM "
                    "sicht_weltobjekt_stand s JOIN sicht_weltobjekt o ON o.objekt_id=s.objekt_id "
                    "WHERE s.turn_id=? AND o.quelle_id=?", (turn_id, quelle_id),
                ).fetchall()
                for obj in objekte:
                    name = (obj["semantische_identitaet"] or "").strip().lower()
                    if name and name in assistent_text:
                        art, beleg = "IN_ANTWORT_ZITIERT", f"Wortlaut '{name}' gefunden"
                    else:
                        art, beleg = "NICHT_NACHWEISBAR", "Benennung nicht im Wortlaut -- nicht messbar, nicht widerlegt"
                    c.execute(
                        "INSERT INTO sicht_nutzung(nutzung_id,objekt_id,turn_id,nutzungsart,"
                        "beleg,festgestellt_von,erzeugt_am) VALUES(?,?,?,?,?,?,?)",
                        (_sicht_kennung(), obj["objekt_id"], turn_id, art, beleg, "MECHANISCH", jetzt),
                    )
                if not objekte:
                    # Turn hatte eine Aufnahme, aber (noch) keinen bestaetigten
                    # Gegenstand -- trotzdem als geprueft markieren, sonst
                    # wird jede Folgerunde erneut (leer) abgefragt.
                    c.execute(
                        "INSERT INTO sicht_nutzung(nutzung_id,turn_id,nutzungsart,beleg,"
                        "festgestellt_von,erzeugt_am) VALUES(?,?,?,?,?,?)",
                        (_sicht_kennung(), turn_id, "KEIN_OBJEKT", "kein bestaetigter Gegenstand in dieser Runde",
                         "MECHANISCH", jetzt),
                    )
            c.commit()
    except Exception as e:
        _log(f"sicht_nutzung_hook fehler sid={session_id}: {e}")


# ── 14d -- P0-Instrument 7 "Sicht-Bilanz" ────────────────────────────────
def _instrument_sicht_bilanz() -> list:
    quelle = "unified.db sicht_* (PHASE 14)"
    try:
        with contextlib.closing(verbindung()) as c:
            c.row_factory = sqlite3.Row
            zeilen = []
            jetzt = _jetzt()

            nach_ergebnis: dict = {}
            abgebrochen = 0
            for row in c.execute("SELECT ergebnis, begonnen_am FROM sicht_aufnahme"):
                ergebnis = row["ergebnis"]
                if ergebnis == "LAEUFT" and (jetzt - row["begonnen_am"]) > 3600:
                    abgebrochen += 1
                    continue
                nach_ergebnis[ergebnis or "NULL"] = nach_ergebnis.get(ergebnis or "NULL", 0) + 1
            for ergebnis, n in nach_ergebnis.items():
                zeilen.append(_messung_zeile("sicht", quelle, f"aufnahme_ergebnis:{ergebnis}",
                                              float(n), "anzahl", {"ergebnis": ergebnis}))
            zeilen.append(_messung_zeile("sicht", quelle, "aufnahme_abgebrochen_ueber_1h",
                                          float(abgebrochen), "anzahl",
                                          {"hinweis": "LAEUFT und begonnen_am > 1h her -- gemeldet, nicht in der DB umgeschrieben"}))

            nach_entscheidung: dict = {}
            for row in c.execute("SELECT entscheidung, COUNT(*) AS n FROM sicht_zuordnung GROUP BY entscheidung"):
                nach_entscheidung[row["entscheidung"]] = row["n"]
            for entscheidung, n in nach_entscheidung.items():
                zeilen.append(_messung_zeile("sicht", quelle, f"zuordnung:{entscheidung}",
                                              float(n), "anzahl", {}))

            wiedererkannt = c.execute(
                "SELECT COUNT(*) AS n FROM (SELECT objekt_id FROM sicht_zuordnung "
                "WHERE objekt_id IS NOT NULL GROUP BY objekt_id "
                "HAVING COUNT(DISTINCT aufnahme_id) >= 2)"
            ).fetchone()["n"]
            zeilen.append(_messung_zeile("sicht", quelle, "weltobjekt_mit_2plus_aufnahmen",
                                          float(wiedererkannt), "anzahl",
                                          {"hinweis": "die eigentliche Wiedererkennungszahl"}))

            pdw = [row["prior_dominanz"] for row in c.execute(
                "SELECT prior_dominanz FROM sicht_hypothese WHERE prior_dominanz IS NOT NULL")
                if row["prior_dominanz"] is not None]
            if pdw:
                pdw.sort()
                median = pdw[len(pdw) // 2] if len(pdw) % 2 else (pdw[len(pdw)//2 - 1] + pdw[len(pdw)//2]) / 2
                zeilen.append(_messung_zeile("sicht", quelle, "prior_dominanz_median",
                                              float(median), "anteil",
                                              {"marke": "[MODELLIERT]", "n": len(pdw)}))
            else:
                zeilen.append(_messung_zeile("sicht", quelle, "prior_dominanz_median",
                                              0.0, "bool", _messung_unbekannt("keine sicht_hypothese-Zeilen")))
            return zeilen
    except sqlite3.OperationalError as e:
        return [_messung_zeile("sicht", quelle, "tabelle_fehlt", 0.0, "bool",
                                _messung_unbekannt(f"{type(e).__name__}: {e}"))]

def _naechste_ordinal(c, session_id: str) -> int:
    row = c.execute(
        "SELECT COALESCE(MAX(ordinal), -1) FROM turns WHERE session_id=?", (session_id,)
    ).fetchone()
    return (row[0] if row else -1) + 1


def zyklus_oeffnen(session_id: str, model: str) -> Optional[str]:
    """Eroeffnet den Sitzungszyklus. Gibt die episode_id zurueck (oder None).

    PHASE 2 (ZECP17): sessions.generation ist ab hier ECHT -- nicht mehr
    hartkodiert 1. zyklus_oeffnen wird laut cmd_hook() ausschliesslich bei
    SessionStart mit source='startup' aufgerufen (siehe dortige Verzweigung);
    das ON CONFLICT hier zaehlt daher genau bei jedem 'startup'-Ereignis
    hoch, wie im Konzeptdokument PHASE 2 gefordert ("hochgezaehlt bei jedem
    SessionStart mit source='startup'"). Eine brandneue session_id bleibt
    bei Generation 1 (INSERT-Zweig greift, kein Konflikt); ein wiederholter
    'startup' fuer dieselbe session_id (z.B. Terminal neu verbunden, gleiche
    ID wiederaufgenommen) zaehlt hoch -- exakt der Fall, den die neue
    PreToolUse-Zulassungspruefung unten als "veraltete Generation" erkennen
    soll, wenn eine Wirkung noch aus der alten Zaehlung angestossen wird."""
    try:
        with contextlib.closing(schreib_verbindung()) as c:
            _sicherstellen_nutzer(c)
            jetzt = _jetzt()
            c.execute(
                "INSERT INTO sessions (session_id, user_id, created_at, updated_at, "
                "generation, terminal_name) VALUES (?,?,?,?,?,?) "
                "ON CONFLICT(session_id) DO UPDATE SET "
                "generation = sessions.generation + 1, updated_at = excluded.updated_at",
                (session_id, USER_ID, jetzt, jetzt, 1, f"claude-code:{session_id}"),
            )
            generation = (
                c.execute("SELECT generation FROM sessions WHERE session_id=?", (session_id,)).fetchone()
                or (1,)
            )[0]

            turn_id = _uid("t-open")
            ordinal = _naechste_ordinal(c, session_id)
            c.execute(
                "INSERT INTO turns (turn_id, session_id, user_id, ordinal, ts, role, "
                "content, fidelity, compression_generation, resource_refs, causal_refs, "
                "provenance) VALUES (?,?,?,?,?,?,?,?,?,?,?,?)",
                (
                    turn_id, session_id, USER_ID, ordinal, jetzt, "system",
                    "[Sitzungsmarke: Zyklus eroeffnet ueber stern.py -- kein echtes "
                    "Transkript erfasst, nur Randmarke]",
                    "marker", 0, "[]", "[]",
                    json.dumps({"quelle": "stern.py", "model": model}, ensure_ascii=False),
                ),
            )
            episode_id = _uid("ep")
            c.execute(
                "INSERT INTO workspace_episodes (episode_id, session_id, ts, "
                "observation_turn_id, salience, alternatives, selected, state) "
                "VALUES (?,?,?,?,?,?,?,?)",
                (episode_id, session_id, jetzt, turn_id, 1.0, "[]", None, "offen"),
            )
            causal_id = _uid("c")
            c.execute(
                "INSERT OR REPLACE INTO active_turns (session_id, user_id, turn_id, "
                "episode_id, causal_id, generation, resource_refs, effect_id, outcome, "
                "workspace_selected, started_at) VALUES (?,?,?,?,?,?,?,?,?,?,?)",
                (session_id, USER_ID, turn_id, episode_id, causal_id, generation, "[]", None, None, 0, jetzt),
            )
            c.commit()
            return episode_id
    except Exception as e:
        _log(f"zyklus_oeffnen fehler sid={session_id}: {e}")
        return None


def zyklus_schliessen(session_id: str) -> None:
    try:
        with contextlib.closing(schreib_verbindung()) as c:
            row = c.execute(
                "SELECT episode_id FROM active_turns WHERE session_id=?", (session_id,)
            ).fetchone()
            jetzt = _jetzt()
            ordinal = _naechste_ordinal(c, session_id)
            turn_id = _uid("t-close")
            c.execute(
                "INSERT INTO turns (turn_id, session_id, user_id, ordinal, ts, role, "
                "content, fidelity, compression_generation, resource_refs, causal_refs, "
                "provenance) VALUES (?,?,?,?,?,?,?,?,?,?,?,?)",
                (
                    turn_id, session_id, USER_ID, ordinal, jetzt, "system",
                    "[Sitzungsmarke: Zyklus geschlossen ueber stern.py]",
                    "marker", 0, "[]", "[]",
                    json.dumps({"quelle": "stern.py"}, ensure_ascii=False),
                ),
            )
            if row and row[0]:
                c.execute(
                    "UPDATE workspace_episodes SET state='geschlossen' WHERE episode_id=?",
                    (row[0],),
                )
            c.execute(
                "UPDATE active_turns SET outcome='geschlossen' WHERE session_id=?",
                (session_id,),
            )
            c.execute("UPDATE sessions SET updated_at=? WHERE session_id=?", (jetzt, session_id))
            c.commit()
    except Exception as e:
        _log(f"zyklus_schliessen fehler sid={session_id}: {e}")
    _checkpoint("Stop/SessionEnd", session_id, "-", "schliessen", "")


def _checkpoint(event: str, session_id: str, model: str, phase: str, injiziert_text: str) -> str:
    digest_quelle = injiziert_text if injiziert_text else f"{phase}:{event}:{session_id}"
    state_hash = hashlib.sha256(digest_quelle.encode("utf-8")).hexdigest()[:16]
    note = json.dumps(
        {
            "phase": phase, "event": event, "session_id": session_id, "model": model,
            "kapsel_laenge": len(injiziert_text) if injiziert_text else 0,
            "kapsel_anfang": _kurz(injiziert_text, 160) if injiziert_text else None,
        },
        ensure_ascii=False, separators=(",", ":"),
    )
    cp_id = _uid(f"udb-{phase}")
    try:
        with contextlib.closing(schreib_verbindung()) as c:
            c.execute(
                "INSERT OR REPLACE INTO checkpoints (checkpoint_id, ts, state_hash, note) "
                "VALUES (?, ?, ?, ?)",
                (cp_id, _jetzt(), state_hash, note),
            )
            c.commit()
    except Exception as e:
        _log(f"checkpoint fehler phase={phase} sid={session_id}: {e}")
    return state_hash

SKILLS_ORDNER = Path.home() / ".claude" / "skills"


def _skills_lesen() -> list:
    ergebnis = []
    if not SKILLS_ORDNER.is_dir():
        return ergebnis
    for d in sorted(SKILLS_ORDNER.iterdir()):
        if not d.is_dir():
            continue
        f = d / "SKILL.md"
        if not f.is_file():
            continue
        try:
            text = f.read_text(encoding="utf-8")
        except Exception:
            continue
        name, beschreibung, inhalt = d.name, "", text
        if text.startswith("---"):
            teile = text.split("---", 2)
            if len(teile) >= 3:
                for zeile in teile[1].splitlines():
                    if zeile.startswith("name:"):
                        name = zeile.split(":", 1)[1].strip() or d.name
                    elif zeile.startswith("description:"):
                        beschreibung = zeile.split(":", 1)[1].strip().strip('"')
                inhalt = teile[2].strip()
        try:
            mtime = f.stat().st_mtime
        except Exception:
            mtime = _jetzt()
        ergebnis.append({
            "ordner": d.name, "name": name, "beschreibung": beschreibung,
            "inhalt": inhalt, "pfad": str(f), "mtime": mtime,
        })
    return ergebnis


def _graph_knoten_sichern(c, node_id: str, kind: str, label: str, source_table: str, source_id: str) -> None:
    c.execute(
        "INSERT OR IGNORE INTO graph_nodes (node_id, kind, label, source_table, source_id) "
        "VALUES (?,?,?,?,?)",
        (node_id, kind, label, source_table, source_id),
    )


def _graph_kanten_ziehen(c, node_id: str, text: str) -> None:
    tl = text.lower()
    for anker_id, muster in _konfig()["anker"].items():
        for m in muster:
            if m in tl:
                anker_node = f"anker:{anker_id}"
                edge_id = hashlib.sha256(f"{node_id}->{anker_node}".encode()).hexdigest()[:24]
                c.execute(
                    "INSERT OR IGNORE INTO graph_edges (edge_id, src, dst, kind, provenance) "
                    "VALUES (?,?,?,?,?)",
                    (edge_id, node_id, anker_node, "erwaehnt", f"keyword:{m}"),
                )
                break


def _skills_synchronisieren(c, skills: list) -> None:
    aktuelle_ids = set()
    for s in skills:
        aid = f"skill:{s['ordner']}"
        aktuelle_ids.add(aid)
        c.execute(
            "INSERT OR REPLACE INTO artikel (id, art, titel, quelle, datum, metadaten, "
            "belegt, erzeugt_am, herkunft) VALUES (?,?,?,?,?,?,?,?,?)",
            (
                aid, "skill", s["name"], s["pfad"],
                time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime(s["mtime"])),
                json.dumps({"ordner": s["ordner"]}, ensure_ascii=False), 1,
                time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime()), "nutzer",
            ),
        )
        c.execute(
            "INSERT OR REPLACE INTO artikel_suchtext (artikel_id, titel, text, blase, "
            "indiziert) VALUES (?,?,?,?,?)",
            (aid, s["name"], (s["beschreibung"] + "\n\n" + s["inhalt"]).strip(), None,
             time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())),
        )
        node_id = f"artikel:{aid}"
        _graph_knoten_sichern(c, node_id, "skill", s["name"], "artikel", aid)
        _graph_kanten_ziehen(c, node_id, s["name"] + " " + s["beschreibung"])
    vorhanden = [r[0] for r in c.execute("SELECT id FROM artikel WHERE art='skill'")]
    for aid in vorhanden:
        if aid not in aktuelle_ids:
            c.execute("DELETE FROM artikel WHERE id=?", (aid,))
            c.execute("DELETE FROM artikel_suchtext WHERE artikel_id=?", (aid,))
            c.execute("DELETE FROM graph_nodes WHERE node_id=?", (f"artikel:{aid}",))
            c.execute("DELETE FROM graph_edges WHERE src=?", (f"artikel:{aid}",))
    try:
        c.execute("INSERT INTO fts_index_artikel(fts_index_artikel) VALUES('rebuild')")
    except Exception:
        pass


def skills_aktualisieren() -> list:
    skills = _skills_lesen()
    try:
        with contextlib.closing(schreib_verbindung()) as c:
            _skills_synchronisieren(c, skills)
            c.commit()
    except Exception as e:
        _log(f"skills_aktualisieren fehler: {e}")
    return skills


def _funktionen_zeile(skills: list, schau: int = 3) -> str:
    if not skills:
        return "0 (keine Funktionen gefunden)"
    namen = [s["name"] for s in skills]
    n = len(namen)
    kurz = ", ".join(namen[:schau])
    rest = n - schau
    if rest > 0:
        return f"{n} ({kurz}, +{rest} weitere -- 'suche skills' fuer alle)"
    return f"{n} ({kurz})"

def _graph_bauen_anker_alle_artikel(c) -> int:
    """Erweitert graph_edges (bisher nur die 57 Skill-Artikel) auf ALLE
    Artikel -- dieselbe _graph_kanten_ziehen()/_graph_knoten_sichern()
    Funktion wie _skills_synchronisieren(), nur jetzt fuer jede Zeile in
    `artikel`. INSERT OR IGNORE (graph_edges.edge_id ist ein Hash aus
    node_id+anker_node) -- ein zweiter Lauf erzeugt keine Dubletten."""
    neu = 0
    for aid, art, titel, quelle, metadaten in c.execute(
        "SELECT id, art, titel, quelle, metadaten FROM artikel"
    ):
        node_id = f"artikel:{aid}"
        _graph_knoten_sichern(c, node_id, art, titel or aid, "artikel", aid)
        vorher = c.execute("SELECT COUNT(*) FROM graph_edges WHERE src=?", (node_id,)).fetchone()[0]
        text = " ".join(str(x or "") for x in (titel, quelle, metadaten))
        _graph_kanten_ziehen(c, node_id, text)
        nachher = c.execute("SELECT COUNT(*) FROM graph_edges WHERE src=?", (node_id,)).fetchone()[0]
        neu += max(0, nachher - vorher)
    return neu


def _graph_bauen_kante_artikel_artikel(c) -> int:
    """[4a-1] Zwei Artikel teilen einen Anker (aus graph_edges, jetzt fuer
    ALLE Artikel gebaut, siehe oben) -> kante.art='geteilter_anker', BEIDE
    Richtungen (Begruendung siehe Abschnittskopf)."""
    paare = c.execute(
        "SELECT e1.src AS a, e2.src AS b, e1.dst AS anker, "
        "e1.provenance AS beleg_a, e2.provenance AS beleg_b "
        "FROM graph_edges e1 JOIN graph_edges e2 "
        "ON e1.dst = e2.dst AND e1.src < e2.src "
        "WHERE e1.dst LIKE 'anker:%' AND e1.src LIKE 'artikel:%' AND e2.src LIKE 'artikel:%'"
    ).fetchall()
    jetzt = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
    neu = 0
    for a_node, b_node, anker, beleg_a, beleg_b in paare:
        a_id = a_node.split(":", 1)[1]
        b_id = b_node.split(":", 1)[1]
        beleg = f"{anker}: {a_id}({beleg_a}) <-> {b_id}({beleg_b})"
        for von, nach in ((a_id, b_id), (b_id, a_id)):
            cur = c.execute(
                "INSERT OR IGNORE INTO kante (von, nach, art, beleg, konfidenz, erzeugt_am, zeitquelle) "
                "VALUES (?,?,?,?,?,?,?)",
                (von, nach, "geteilter_anker", beleg, 0.6, jetzt, "graph-bauen"),
            )
            neu += cur.rowcount
    return neu


def _graph_bauen_kante_artikel_durable(c) -> int:
    """[4a-2] artikel<->durable_memory ueber geteilte vp_alias-Treffer.
    CANONICAL-Aliase sind artikel.id-Slugs (>=6 Zeichen, sehr eindeutig);
    DOCUMENT_DERIVED-Aliase sind Klartext-Titel (nur ab 10 Zeichen benutzt,
    damit ein kurzer generischer Titel nicht in halb durable_memory
    "trifft"). Deckel je Alias (15 Treffer) -- F45: zu viel ist
    unumkehrbar, kein Vollscan-Ertrag ohne Deckel."""
    dm = c.execute("SELECT memory_id, subject, value FROM durable_memory").fetchall()
    dm_text = [(r["memory_id"], f"{r['subject']} {r['value']}".lower()) for r in dm]
    aliase = c.execute(
        "SELECT target_id, alias_norm, alias_class FROM vp_alias WHERE target_kind='artikel' AND "
        "((alias_class='CANONICAL' AND length(alias_norm) >= 6) OR "
        " (alias_class='DOCUMENT_DERIVED' AND length(alias_norm) >= 10))"
    ).fetchall()
    jetzt = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
    neu = 0
    for target_id, alias_norm, alias_class in aliase:
        nadel = alias_norm.lower()
        treffer = 0
        for mem_id, text in dm_text:
            if nadel in text:
                cur = c.execute(
                    "INSERT OR IGNORE INTO kante (von, nach, art, beleg, konfidenz, erzeugt_am, zeitquelle) "
                    "VALUES (?,?,?,?,?,?,?)",
                    (target_id, mem_id, "alias_treffer",
                     f"alias:{alias_norm}|klasse:{alias_class}",
                     0.5 if alias_class == "CANONICAL" else 0.35, jetzt, "graph-bauen"),
                )
                neu += cur.rowcount
                treffer += 1
                if treffer >= 15:
                    break
    return neu


def _graph_bauen_kante_effekt(c) -> int:
    """[4a-3] effects.target (echte Dateipfade -- nur Write/Edit/Read/
    NotebookEdit, KEINE Bash-Kommandozeilen) <-> artikel/durable_memory,
    die denselben Pfad nennen. Ein Pfad wird EINMAL als kante-Knoten
    gefuehrt (`pfad:<sha256[:16]>`), nicht je effect_id -- sonst waeren
    Dutzende effects-Zeilen mit demselben Pfad Dutzende identischer Kanten."""
    pfade = [r[0] for r in c.execute(
        "SELECT DISTINCT target FROM effects WHERE capability IN "
        "('Write','Edit','Read','NotebookEdit') AND target LIKE '/%'"
    ).fetchall()]
    jetzt = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
    neu = 0
    for pfad in pfade:
        pfad_id = "pfad:" + hashlib.sha256(pfad.encode("utf-8")).hexdigest()[:16]
        n_effekte = c.execute(
            "SELECT COUNT(*) FROM effects WHERE target=? AND capability IN "
            "('Write','Edit','Read','NotebookEdit')", (pfad,),
        ).fetchone()[0]
        beleg = f"pfad={pfad}; effects_zeilen={n_effekte}"
        for (aid,) in c.execute("SELECT id FROM artikel WHERE quelle=?", (pfad,)).fetchall():
            cur = c.execute(
                "INSERT OR IGNORE INTO kante (von, nach, art, beleg, konfidenz, erzeugt_am, zeitquelle) "
                "VALUES (?,?,?,?,?,?,?)",
                (pfad_id, aid, "effects_pfad", beleg, 0.7, jetzt, "graph-bauen"),
            )
            neu += cur.rowcount
        like = f"%{pfad}%"
        for (mem_id,) in c.execute(
            "SELECT memory_id FROM durable_memory WHERE value LIKE ? LIMIT 25", (like,)
        ).fetchall():
            cur = c.execute(
                "INSERT OR IGNORE INTO kante (von, nach, art, beleg, konfidenz, erzeugt_am, zeitquelle) "
                "VALUES (?,?,?,?,?,?,?)",
                (pfad_id, mem_id, "effects_pfad", beleg, 0.5, jetzt, "graph-bauen"),
            )
            neu += cur.rowcount
    return neu


def graph_bauen() -> dict:
    """`stern graph-bauen` -- PHASE 4a, Handbefehl (kein Hook ruft das
    automatisch auf). Rein deterministisch, kein LLM, kein Netz. Siehe
    Abschnittskopf fuer die drei Kantenquellen und die Begruendung, warum
    graph_kante/graph_knoten bewusst leer bleiben."""
    t0 = time.monotonic()
    try:
        with contextlib.closing(schreib_verbindung()) as c:
            c.row_factory = sqlite3.Row
            graph_edges_neu = _graph_bauen_anker_alle_artikel(c)
            kante_aa = _graph_bauen_kante_artikel_artikel(c)
            kante_ad = _graph_bauen_kante_artikel_durable(c)
            kante_ae = _graph_bauen_kante_effekt(c)
            # paket.py's vp_plan_cache-Schluessel haengt (auch) an graph_epoch =
            # MAX(graph_kante.graph_epoch) -- und graph_kante bleibt hier bewusst
            # leer (siehe Abschnittskopf), also bewegt sich graph_epoch NIE.
            # Ohne diesen Schnitt wuerde jeder VOR diesem Lauf schon gecachte
            # Plan (gebaut, als `kante` noch 0 Zeilen hatte) unbegrenzt WARM_CACHE
            # bleiben und walk_gefunden=0 vortaeuschen, obwohl `kante` jetzt voll
            # ist -- ein reiner Cache-Leichnam, kein `paket.py`-Verhalten
            # (paket.py selbst wird nicht angefasst, vp_plan_cache ist seine
            # eigene, wiederherstellbare Ablage).
            cache_geloescht = c.execute("SELECT COUNT(*) FROM vp_plan_cache").fetchone()[0]
            c.execute("DELETE FROM vp_plan_cache")
            c.commit()
            gesamt = c.execute("SELECT COUNT(*) FROM kante").fetchone()[0]
            ohne_beleg = c.execute(
                "SELECT COUNT(*) FROM kante WHERE beleg IS NULL OR beleg=''"
            ).fetchone()[0]
            proben = [dict(r) for r in c.execute(
                "SELECT von, nach, art, beleg FROM kante ORDER BY RANDOM() LIMIT 8"
            ).fetchall()]
    except Exception as e:
        _log(f"graph_bauen fehler: {type(e).__name__}: {e}")
        return {"fehler": f"{type(e).__name__}: {e}"}
    ergebnis = {
        "graph_edges_neu_ueber_ankerkarte": graph_edges_neu,
        "kante_artikel_artikel_neu": kante_aa,
        "kante_artikel_durable_memory_neu": kante_ad,
        "kante_effekt_pfad_neu": kante_ae,
        "kante_gesamt": gesamt,
        "kante_ohne_beleg": ohne_beleg,
        "vp_plan_cache_geloescht": cache_geloescht,
        "proben_beleg": proben,
        "dauer_s": round(time.monotonic() - t0, 3),
    }
    _log(f"graph_bauen: kante_gesamt={gesamt} ohne_beleg={ohne_beleg} "
         f"aa={kante_aa} ad={kante_ad} ae={kante_ae} dauer_s={ergebnis['dauer_s']}")
    return ergebnis


def _turn_nutzung_lesen(transcript_path) -> Optional[dict]:
    if not transcript_path:
        return None
    try:
        p = Path(transcript_path)
        if not p.is_file():
            return None
        letzte = None
        with p.open(encoding="utf-8", errors="ignore") as f:
            for zeile in f:
                zeile = zeile.strip()
                if not zeile:
                    continue
                try:
                    d = json.loads(zeile)
                except Exception:
                    continue
                if d.get("type") == "assistant":
                    u = (d.get("message") or {}).get("usage")
                    if isinstance(u, dict) and (u.get("input_tokens") or u.get("output_tokens")):
                        letzte = u
        if not letzte:
            return None
        return {
            "input_tokens": int(letzte.get("input_tokens") or 0),
            "output_tokens": int(letzte.get("output_tokens") or 0),
            "cache_creation_input_tokens": int(letzte.get("cache_creation_input_tokens") or 0),
            "cache_read_input_tokens": int(letzte.get("cache_read_input_tokens") or 0),
        }
    except Exception:
        return None


def _kosten_zeilen(nutzung: Optional[dict]):
    if not nutzung:
        return (None, None)
    ip, op = nutzung["input_tokens"], nutzung["output_tokens"]
    tok_zeile = f"in {ip} / out {op}"
    if nutzung.get("cache_read_input_tokens"):
        tok_zeile += f" / cache-read {nutzung['cache_read_input_tokens']}"
    if nutzung.get("cache_creation_input_tokens"):
        tok_zeile += f" / cache-write {nutzung['cache_creation_input_tokens']}"
    saetze = _konfig()["geld_saetze"]
    opus = (ip / 1_000_000) * saetze["opus5"]["input"] + (op / 1_000_000) * saetze["opus5"]["output"]
    sonnet = (ip / 1_000_000) * saetze["sonnet5"]["input"] + (op / 1_000_000) * saetze["sonnet5"]["output"]
    geld_zeile = (
        f"Opus 5 pur (Schaetzung) ${opus:.4f} · Sonnet 5 + UnifiedDB (Turn, tatsaechlich) ${sonnet:.4f}"
    )
    return (tok_zeile, geld_zeile)


def _codebox(skills: list, transcript_path=None) -> str:
    nutzung = _turn_nutzung_lesen(transcript_path)
    tok_zeile, geld_zeile = _kosten_zeilen(nutzung)
    zeitstempel = time.strftime("%Y-%m-%dT%H:%M:%SZ", time.gmtime())
    zeilen = [
        "── ZEROv0.00001 ──",
        f"fn   {_funktionen_zeile(skills)}",
        f"t    {zeitstempel}",
    ]
    if tok_zeile is not None:
        zeilen.append(f"tok  {tok_zeile}")
    if geld_zeile is not None:
        zeilen.append(f"$    {geld_zeile}")
    return "\n".join(zeilen)

            ).fetchone()

                if ergebnis:

# ── Forschungs-Digest -- jetzt star_digest/star_digest_lauf statt JSON (1c) ──
def _forschung_digest_lesen() -> tuple:
    """Liest den letzten Digest-Lauf aus der DB -- instant, nie ein
    Netzwerkaufruf im Hook selbst (Netz passiert nur in cmd_digest(), das
    ausschliesslich der Timer aufruft). Gibt (items, generated_at) zurueck."""
    try:
        with contextlib.closing(verbindung()) as c:
            row = c.execute(
                "SELECT lauf_id, generated_at FROM star_digest_lauf ORDER BY generated_at DESC LIMIT 1"
            ).fetchone()
            if not row:
                return [], None
            lauf_id, generated_at = row
            items = [
                {"commit": r[0], "datum": r[1], "titel": r[2]}
                for r in c.execute(
                    'SELECT "commit", datum, titel FROM star_digest WHERE lauf_id=? ORDER BY rowid',
                    (lauf_id,),
                )
            ]
            return items, generated_at
    except Exception:
        return [], None


# ── SessionStart: reines Skript, KEIN Modellaufruf ───────────────────────
def _gw_zusammenfassung() -> Optional[str]:
    """PHASE 6/ZECP10, P6b Punkt 9 -- ein Satz im SessionStart, "wenn die
    Zahlen ihn tragen". Reine Tatsachendarstellung ueber den kumulierten
    Bestand (nicht sitzungsscharf -- SessionStart selbst hat fuer die NEUE

    Bewertung -- exakt das Format, das start_anzeige() schon fuer den

        with contextlib.closing(verbindung()) as c:
            n_kandidaten = c.execute(
                "SELECT COUNT(DISTINCT artikel_id) FROM gw_wettbewerb"
            ).fetchone()[0]

            m_ausgestrahlt = c.execute("SELECT COUNT(*) FROM gw_ausstrahlung").fetchone()[0]

def _ist_fehlerhaft(tool_response) -> bool:
    if isinstance(tool_response, dict):
        return bool(tool_response.get("is_error") or tool_response.get("error"))
    if isinstance(tool_response, str):
        return tool_response.strip().lower().startswith(("error", "fehler"))
    return False


def _argv_normalisiert(tool_input) -> str:
    """Kanonische Form fuer den Idempotenz-Schluessel -- sortierte Schluessel,
    kein Leerraum, damit dieselbe Eingabe immer denselben Hash ergibt,
    unabhaengig von Schluesselreihenfolge im eingehenden JSON."""
    try:
        return json.dumps(tool_input, ensure_ascii=False, sort_keys=True, separators=(",", ":"))
    except Exception:
        return _kurz(tool_input, 500)


def _idempotenz_schluessel(tool_name: str, target: str, tool_input) -> str:
    roh = f"{tool_name}|{target}|{_argv_normalisiert(tool_input)}"
    return hashlib.sha256(roh.encode("utf-8")).hexdigest()

def zulassen(payload: dict) -> None:
    """PreToolUse -- Zulassung. Budget 0.4s (star_konfig limit.PreToolUse),
    durchgesetzt von cmd_hook()'s SIGALRM, nicht hier selbst. Prueft in
    fester Reihenfolge: 1 Capability, 2 Generation, 3 Vorbedingung,
    4 Idempotenz -- und schreibt GENAU EINE effects-Zeile mit dem Befund.
    Die effect_id wird in active_turns.effect_id abgelegt (die bereits
    bestehende sitzungsgebundene Ablage, ein Feld je Sitzung) -- merken()
    (PostToolUse) liest sie von dort und aktualisiert dieselbe Zeile, statt
    eine zweite anzulegen."""
    session_id = str(payload.get("session_id") or "unknown")
    tool_name = payload.get("tool_name") or "?"
    tool_input = payload.get("tool_input") or {}
    if tool_name not in ZUSTANDSAENDERND:
        return  # nur zustandsaendernde Werkzeuge bekommen eine EffectGate-Zeile

    target = _effect_ziel(tool_input, tool_name)
    idem_schluessel = _idempotenz_schluessel(tool_name, target, tool_input)

    try:
        with contextlib.closing(schreib_verbindung()) as c:
            _sicherstellen_faehigkeit_tabelle(c)

            # 1) Capability -- Voreinstellung: alles zugelassen (Registrierung, kein Verbot)
            row = c.execute(
                "SELECT zugelassen FROM star_faehigkeit WHERE tool_name=?", (tool_name,)
            ).fetchone()
            if row is None:
                c.execute(
                    "INSERT OR IGNORE INTO star_faehigkeit (tool_name, zugelassen, grund, geaendert) "
                    "VALUES (?,1,?,?)",
                    (tool_name, "auto-registriert bei erster Zulassungspruefung", _jetzt()),
                )
                capability_ok = True
            else:
                capability_ok = bool(row[0])

            # 2) Generation -- Sitzungsgeneration (sessions.generation) vs. die
            # Generation, unter der die aktuelle Episode (active_turns) eroeffnet
            # wurde. Weichen sie ab, ist die Episode aus einer inzwischen per
            # neuem SessionStart(source=startup) ueberholten Zaehlung.
            srow = c.execute(
                "SELECT generation FROM sessions WHERE session_id=?", (session_id,)
            ).fetchone()
            aktuelle_generation = srow[0] if srow else 1
            arow = c.execute(
                "SELECT episode_id, generation FROM active_turns WHERE session_id=?", (session_id,)
            ).fetchone()
            episode_id = arow[0] if arow else None
            erwartete_generation = arow[1] if arow else aktuelle_generation
            generation_ok = (erwartete_generation == aktuelle_generation)

            # 3) Vorbedingung -- bei Edit/MultiEdit/NotebookEdit muss die Zieldatei
            # existieren, bei Write muss zumindest der Elternordner existieren.
            vorbedingung_ok, vorbedingung_grund = True, None
            if tool_name in ("Edit", "MultiEdit"):
                fp = tool_input.get("file_path") if isinstance(tool_input, dict) else None
                if not fp or not Path(fp).is_file():
                    vorbedingung_ok = False
                    vorbedingung_grund = f"Zieldatei existiert nicht: {fp}"
            elif tool_name == "NotebookEdit":
                np_ = tool_input.get("notebook_path") if isinstance(tool_input, dict) else None
                if not np_ or not Path(np_).is_file():
                    vorbedingung_ok = False
                    vorbedingung_grund = f"Notebook existiert nicht: {np_}"
            elif tool_name == "Write":
                fp = tool_input.get("file_path") if isinstance(tool_input, dict) else None
                if fp and not Path(fp).parent.is_dir():
                    vorbedingung_ok = False
                    vorbedingung_grund = f"Elternordner fehlt: {Path(fp).parent}"
            # Bash hat keine Datei-Vorbedingung -- bleibt ok.

            status, grund = "ZUGELASSEN", None
            if not capability_ok:
                status, grund = "ABGELEHNT", "capability: Werkzeug in star_faehigkeit gesperrt"
            elif not generation_ok:
                status, grund = (
                    "ABGELEHNT",
                    f"generation veraltet: Episode erwartet {erwartete_generation}, "
                    f"Sitzung aktuell {aktuelle_generation}",
                )
            elif not vorbedingung_ok:
                status, grund = "ABGELEHNT", f"vorbedingung: {vorbedingung_grund}"
            else:
                # 4) Idempotenz -- identischer, bereits VERIFIED verbuchter Effekt
                # in DERSELBEN Sitzung wird markiert, nicht blockiert.
                # Suchmuster bewusst NUR auf den gequoteten Hash-Wert, nicht auf
                # den umgebenden Schluessel/Doppelpunkt/Leerraum -- json.dumps()
                # ohne separators= setzt ein Leerzeichen nach dem Doppelpunkt
                # ("idempotenz_schluessel": "..."), ein Muster mit fest
                # eingebautem "):"  waere formatabhaengig und still falsch
                # (beim ersten Bau dieser Phase genau so gefunden: kein
                # Duplikat wurde je erkannt, weil das Muster nie traf). Der
                # Hash selbst ist 64 Hex-Zeichen -- als Teilstring praktisch
                # kollisionsfrei, ein Wrapper aus Anfuehrungszeichen genuegt.
                dup = c.execute(
                    "SELECT e.effect_id FROM effects e "
                    "JOIN workspace_episodes we ON we.episode_id = e.episode_id "
                    "WHERE we.session_id=? AND e.status='VERIFIED' "
                    "AND e.outcome LIKE ? LIMIT 1",
                    (session_id, f'%"{idem_schluessel}"%'),
                ).fetchone()
                if dup:
                    status = "WIEDERHOLUNG"
                    grund = f"identischer verifizierter Effekt bereits vorhanden: {dup[0]}"

            outcome = {
                "phase": "zulassung",
                "idempotenz_schluessel": idem_schluessel,
                "pruefungen": {
                    "capability": capability_ok, "generation": generation_ok,
                    "vorbedingung": vorbedingung_ok,
                },
            }
            if grund:
                outcome["grund"] = grund

            effect_id = _uid("eff")
            c.execute(
                "INSERT INTO effects (effect_id, episode_id, user_id, capability, "
                "target, argv, requested_generation, status, outcome, ts, verified_at) "
                "VALUES (?,?,?,?,?,?,?,?,?,?,?)",
                (
                    effect_id, episode_id, USER_ID, tool_name, _kurz_redigiert(target, 300),
                    _argv_normalisiert_redigiert(tool_input), aktuelle_generation, status,
                    json.dumps(outcome, ensure_ascii=False), _jetzt(), None,
                ),
            )
            if arow is not None:
                c.execute(
                    "UPDATE active_turns SET effect_id=? WHERE session_id=?",
                    (effect_id, session_id),
                )
            c.commit()
    except Exception as e:
        # Fail-silent gemaess harter Grenze 4 -- das Werkzeug laeuft so oder so
        # (cmd_hook() gibt fuer PreToolUse ohnehin keine Entscheidung zurueck).
        # Bestmoeglicher Nachtrag: eine UNGEPRUEFT-Zeile, wenn selbst das noch
        # in der Restzeit geht; misslingt auch das, bleibt die Sitzung einfach
        # ohne Zulassungszeile fuer diesen Aufruf -- merken() faengt das unten
        # als Fallback ab.
        _log(f"zulassen fehler sid={session_id} tool={tool_name}: {e}")
        try:
            with contextlib.closing(schreib_verbindung()) as c2:
                c2.execute(
                    "INSERT INTO effects (effect_id, episode_id, user_id, capability, "
                    "target, argv, requested_generation, status, outcome, ts, verified_at) "
                    "VALUES (?,?,?,?,?,?,?,?,?,?,?)",
                    (
                        _uid("eff"), None, USER_ID, tool_name, _kurz_redigiert(target, 300),
                        _argv_normalisiert_redigiert(tool_input), 1, "UNGEPRUEFT",
                        json.dumps({"grund": f"pruefung fehlgeschlagen: {e}"}, ensure_ascii=False),
                        _jetzt(), None,
                    ),
                )
                c2.commit()
        except Exception:
            pass  # selbst der Nachtrag darf nie eine Sitzung stoeren


def _wirkung_pruefen(tool_name: str, tool_input, tool_response) -> tuple:
    """PostToolUse -- das tatsaechliche Ergebnis, gehasht (Zeros Muster:
    {'ok':bool,'bytes':n,'sha256':...}) statt abgeschnittener Rohtext.
    Gibt (ok, outcome_objekt) zurueck."""
    fehlerhaft = _ist_fehlerhaft(tool_response)
    if tool_name == "Bash":
        exit_code = None
        text = ""
        if isinstance(tool_response, dict):
            if "exit_code" in tool_response:
                exit_code = tool_response.get("exit_code")
            text = str(tool_response.get("stdout") or "") + str(tool_response.get("stderr") or "")
        else:
            text = str(tool_response or "")
        roh = text.encode("utf-8", "replace")
        ok = (exit_code == 0) if exit_code is not None else (not fehlerhaft)
        return ok, {"ok": ok, "bytes": len(roh), "sha256": hashlib.sha256(roh).hexdigest(),
                     "exit_code": exit_code}

    if tool_name in ("Edit", "Write", "MultiEdit", "NotebookEdit"):
        pfad = None
        if isinstance(tool_input, dict):
            pfad = tool_input.get("file_path") or tool_input.get("notebook_path")
        if pfad and Path(pfad).is_file():
            try:
                roh = Path(pfad).read_bytes()
                ok = not fehlerhaft
                return ok, {"ok": ok, "bytes": len(roh), "sha256": hashlib.sha256(roh).hexdigest()}
            except Exception as e:
                return False, {"ok": False, "bytes": 0, "sha256": None, "fehler": str(e)}
        return (not fehlerhaft), {
            "ok": not fehlerhaft, "bytes": 0, "sha256": None,
            "hinweis": "Zieldatei nach Lauf nicht gefunden",
        }

    text = _kurz(tool_response, 2000).encode("utf-8", "replace")
    ok = not fehlerhaft
    return ok, {"ok": ok, "bytes": len(text), "sha256": hashlib.sha256(text).hexdigest()}


def merken(payload: dict) -> None:
    """PostToolUse -- Ereignisspur (unveraendert) + Wirkungsverifikation
    (PHASE 2, ZECP17). Aktualisiert dieselbe effects-Zeile, die zulassen()
    (PreToolUse) angelegt hat (gefunden ueber active_turns.effect_id); nur
    wenn dort status='ZUGELASSEN' stand, wird auf VERIFIED/FEHLGESCHLAGEN
    umgeschaltet -- ABGELEHNT/WIEDERHOLUNG/UNGEPRUEFT bleiben als Befund
    stehen (das IST die Information), bekommen aber trotzdem verified_at
    und einen Nachtrag mit dem tatsaechlichen Ergebnis. Fehlt eine
    PreToolUse-Zeile ganz (Timeout/Fehler/Fallback), legt merken() selbst
    eine an -- das Journal darf nie leer bleiben, nur weil die Zulassung es
    nicht rechtzeitig geschafft hat. Zusaetzlich: je verifiziertem Effekt
    eine causal_episodes-Zeile (hier zum ersten Mal ueberhaupt beschrieben,
    Kredit bleibt 0 -- das ist P3)."""
    session_id = str(payload.get("session_id") or "unknown")
    tool_name = payload.get("tool_name") or payload.get("hook_event_name") or "?"
    tool_input = payload.get("tool_input") or {}
    tool_response = payload.get("tool_response")

    try:
        with contextlib.closing(schreib_verbindung()) as c:
            mem_id = _uid(f"ereignis-{session_id}")
            wert = {"input": _kurz_redigiert(tool_input, 220), "response": _kurz_redigiert(tool_response, 220)}
            c.execute(
                "INSERT INTO durable_memory (memory_id, kind, subject, value, "
                "source_turn_id, user_id, ts, provenance) VALUES (?,?,?,?,?,?,?,?)",
                (
                    mem_id, "ereignis", f"tool:{tool_name}",
                    json.dumps(wert, ensure_ascii=False),
                    None, USER_ID, _jetzt(), f"hook:PostToolUse:session={session_id}",
                ),
            )

            # PHASE 6/ZECP10, P6a Punkt 2 -- gw_herkunft. Provenienzzeile je
            # NEU geschriebener durable_memory-Zeile. `merken()` ist [LIVE
            # geprueft, 2026-08-26] der EINZIGE Schreibpfad in diese Tabelle
            # in der gesamten Datei -- jede Zeile hier entsteht also
            # zwingend aus tool_input/tool_response, nie aus Gabriels
            # eigener Eingabe oder dem Korpus. fremdtext=1 ist deshalb hier
            # keine geratene Heuristik, sondern eine Tatsache ueber DIESEN
            # einen Schreibpfad -- ehrlich gemeldeter Befund, nicht "immer 1
            # weil bequem": sobald ein zweiter Schreibpfad (z.B. explizites
            # Merken von Gabriels eigenem Text) entsteht, muss der dortige
            # Aufrufer sein eigenes fremdtext setzen, nicht diese Konstante.
            gw_turn_row = c.execute(
                "SELECT turn_id FROM active_turns WHERE session_id=?", (session_id,)
            ).fetchone()
            c.execute(
                "INSERT INTO gw_herkunft(memory_id, turn_id, session_id, ts, "
                "fremdtext, werkzeuge) VALUES (?,?,?,?,?,?)",
                (
                    mem_id, gw_turn_row[0] if gw_turn_row else None, session_id,
                    _jetzt(), 1, json.dumps([tool_name], ensure_ascii=False),
                ),
            )

            if tool_name in ZUSTANDSAENDERND:
                ok, wirkung = _wirkung_pruefen(tool_name, tool_input, tool_response)

                arow = c.execute(
                    "SELECT episode_id, effect_id, turn_id FROM active_turns WHERE session_id=?",
                    (session_id,),
                ).fetchone()
                episode_id = arow[0] if arow else None
                effect_id = arow[1] if arow else None
                turn_id = arow[2] if arow else None

                zeile = c.execute(
                    "SELECT status, outcome FROM effects WHERE effect_id=?", (effect_id,)
                ).fetchone() if effect_id else None

                if zeile:
                    alter_status, alter_outcome_roh = zeile
                    try:
                        alter_outcome = json.loads(alter_outcome_roh) if alter_outcome_roh else {}
                    except Exception:
                        alter_outcome = {"unlesbar": alter_outcome_roh}
                    neuer_status = "VERIFIED" if ok else "FEHLGESCHLAGEN"
                    if alter_status not in ("ZUGELASSEN",):
                        neuer_status = alter_status  # ABGELEHNT/WIEDERHOLUNG/UNGEPRUEFT bleiben stehen
                    alter_outcome["nachtrag"] = wirkung
                    c.execute(
                        "UPDATE effects SET status=?, outcome=?, verified_at=? WHERE effect_id=?",
                        (neuer_status, json.dumps(alter_outcome, ensure_ascii=False), _jetzt(), effect_id),
                    )
                else:
                    # Fallback: keine PreToolUse-Zeile gefunden (Timeout, Fehler,
                    # oder Aufruf ausserhalb eines echten Sitzungszyklus).
                    srow = c.execute(
                        "SELECT generation FROM sessions WHERE session_id=?", (session_id,)
                    ).fetchone()
                    generation = srow[0] if srow else 1
                    effect_id = _uid("eff")
                    target = _effect_ziel(tool_input, tool_name)
                    c.execute(
                        "INSERT INTO effects (effect_id, episode_id, user_id, capability, "
                        "target, argv, requested_generation, status, outcome, ts, verified_at) "
                        "VALUES (?,?,?,?,?,?,?,?,?,?,?)",
                        (
                            effect_id, episode_id, USER_ID, tool_name, _kurz_redigiert(target, 300),
                            _argv_normalisiert_redigiert(tool_input), generation,
                            "VERIFIED" if ok else "FEHLGESCHLAGEN",
                            json.dumps(
                                {"nachtrag": wirkung,
                                 "hinweis": "keine PreToolUse-Zeile gefunden (Timeout/Fehler/Fallback)"},
                                ensure_ascii=False,
                            ),
                            _jetzt(), _jetzt(),
                        ),
                    )

                c.execute(
                    "INSERT INTO causal_episodes (causal_id, episode_id, effect_id, "
                    "observation_turn_id, outcome_hash, credit, reentered, ts) "
                    "VALUES (?,?,?,?,?,?,?,?)",
                    (_uid("c"), episode_id, effect_id, turn_id, wirkung.get("sha256"), 0, 0, _jetzt()),
                )
            c.commit()
    except Exception as e:
        _log(f"merken fehler sid={session_id} tool={tool_name}: {e}")
        return

    # P5d: Pacht auf die angefasste Datei erneuern -- NUR fuer die drei
    # echten Dateiwerkzeuge (Bash-Ziele sind keine verlaesslichen Dateipfade,
    # siehe target-Ableitung oben, die bei Bash den Befehlstext nimmt).
    # Fail-silent, ausserhalb der schreib_verbindung()-Transaktion oben, weil
    # lease_halten() seine eigene kurze Verbindung oeffnet -- eine
    # fehlschlagende Pacht darf die Ereignisspur nicht mit hinunterreissen.
    if tool_name in ("Write", "Edit", "MultiEdit", "NotebookEdit") and isinstance(tool_input, dict):
        pfad = tool_input.get("file_path") or tool_input.get("notebook_path")
        if pfad:
            try:
                with contextlib.closing(verbindung()) as c3:
                    genrow = c3.execute(
                        "SELECT generation FROM sessions WHERE session_id=?", (session_id,)
                    ).fetchone()
                generation = genrow[0] if genrow else 1
            except Exception:
                generation = 1
            lease_halten(pfad, session_id, generation)

def _gw_angekuendigte_werkzeuge(assistent_text: str) -> set:
    """Heuristik, ausdruecklich als solche dokumentiert (Konzeptdokument P6a
    Punkt 1, 'ehrliche Grenze'): Werkzeugnamen als eigenes Wort im
    Antworttext. Kein Modellurteil, keine Ausfuehrungssemantik."""
    if not assistent_text:
        return set()
    return {name for name, muster in _GW_WERKZEUG_MUSTER.items() if muster.search(assistent_text)}


def _gw_absicht_schreiben(session_id: str, transcript_path) -> None:
    """Stop-Hook, laeuft im 2.0s-Budget von limit.Stop mit. Schreibt GENAU
    EINE gw_absicht-Zeile je Runde -- rein mechanisch, drei Regeln, siehe
    Docstring der Verdachtslogik unten. Fail-silent (harte Grenze 4)."""
    t0 = time.monotonic()
    try:
        with contextlib.closing(schreib_verbindung()) as c:
            c.row_factory = sqlite3.Row
            aktiv = c.execute(
                "SELECT turn_id, episode_id, started_at FROM active_turns WHERE session_id=?",
                (session_id,),
            ).fetchone()
            if not aktiv:
                _log(f"gw_absicht uebersprungen sid={session_id}: keine active_turns-Zeile")
                return
            turn_id, episode_id, started_at = aktiv["turn_id"], aktiv["episode_id"], aktiv["started_at"]
            jetzt = _jetzt()

            frage = _transcript_nutzertext_lesen(transcript_path)
            assistent_text = _transcript_assistenttext_lesen(transcript_path)
            angekuendigt = _gw_angekuendigte_werkzeuge(assistent_text)

            # "werkzeuge" -- ALLE Werkzeuge, die in diesem Zeitfenster tatsaechlich
            # ueber PostToolUse gefeuert haben, aus den `ereignis`-Zeilen, die
            # merken() fuer JEDES Werkzeug schreibt (nicht nur ZUSTANDSAENDERND --
            # effects deckt nur die Teilmenge). Fenster = [started_at, jetzt],
            # dasselbe Muster wie kreditsignal_mechanisch() weiter unten benutzt.
            werkzeuge = set()
            for row in c.execute(
                "SELECT subject FROM durable_memory WHERE kind='ereignis' "
                "AND ts>=? AND ts<=? AND provenance=?",
                (started_at, jetzt, f"hook:PostToolUse:session={session_id}"),
            ):
                subj = str(row[0])
                if subj.startswith("tool:"):
                    werkzeuge.add(subj[len("tool:"):])

            # "aussenwirkung" -- die zustandsaendernde Teilmenge, VERIFIED aus
            # effects (P2/ZECP17), gleiches Fenster ueber episode_id.
            aussenwirkung = set()
            if episode_id:
                for row in c.execute(
                    "SELECT capability FROM effects WHERE episode_id=? AND status='VERIFIED' "
                    "AND ts>=? AND ts<=?",
                    (episode_id, started_at, jetzt),
                ):
                    aussenwirkung.add(str(row[0]))

            # Drei mechanische Regeln, EXAKT in dieser Prioritaet (Konzeptdokument
            # P6a Punkt 1) -- kein Modellurteil, ein `verdacht`-Feld je Zeile.
            nicht_ausgefuehrt = angekuendigt - werkzeuge
            wirkung_ohne_ankuendigung = aussenwirkung - angekuendigt
            nicht_angekuendigt = werkzeuge - angekuendigt
            if nicht_ausgefuehrt:
                verdacht = "ANGEKUENDIGT_NICHT_AUSGEFUEHRT"
                grund = f"angekuendigt, nie gefeuert: {sorted(nicht_ausgefuehrt)}"
            elif wirkung_ohne_ankuendigung:
                verdacht = "WIRKUNG_OHNE_ANKUENDIGUNG"
                grund = f"zustandsaendernd verifiziert, nirgends angekuendigt: {sorted(wirkung_ohne_ankuendigung)}"
            elif nicht_angekuendigt:
                verdacht = "AUSGEFUEHRT_NICHT_ANGEKUENDIGT"
                grund = f"gefeuert, nirgends angekuendigt: {sorted(nicht_angekuendigt)}"
            else:
                verdacht, grund = None, None

            k = _konfig()
            c.execute(
                "INSERT INTO gw_absicht(ts, turn_id, session_id, frage, werkzeuge, "
                "aussenwirkung, angekuendigt, verdacht, grund) VALUES (?,?,?,?,?,?,?,?,?)",
                (
                    jetzt, turn_id, session_id, _kurz(frage, int(k["gw.frage_kappung"])),
                    json.dumps(sorted(werkzeuge), ensure_ascii=False),
                    json.dumps(sorted(aussenwirkung), ensure_ascii=False),
                    json.dumps(sorted(angekuendigt), ensure_ascii=False),
                    verdacht, grund,
                ),
            )
            c.commit()
        _log(
            f"gw_absicht sid={session_id} dauer={round((time.monotonic()-t0)*1000)}ms "
            f"werkzeuge={sorted(werkzeuge)} angekuendigt={sorted(angekuendigt)} verdacht={verdacht}"
        )
    except Exception as e:
        _log(f"gw_absicht fehler sid={session_id}: {e}")

def suche(begriff: str) -> str:
    if begriff.strip().lower() == "skills":
        namen = [s["name"] for s in _skills_lesen()]
        if not namen:
            return "Keine Skills unter ~/.claude/skills/ gefunden."
        return f"{len(namen)} Skills: " + ", ".join(namen)

    treffer = []
    like = f"%{begriff}%"
    try:
        with contextlib.closing(verbindung()) as c:
            for mem_id, kind, subject, value, ts in c.execute(
                "SELECT memory_id, kind, subject, value, ts FROM durable_memory "
                "WHERE (subject LIKE ? OR value LIKE ? OR kind LIKE ?) AND kind != 'ereignis' "
                "ORDER BY ts DESC LIMIT 20",
                (like, like, like),
            ):
                treffer.append(f"[durable_memory:{kind}] {subject}: {_kurz(value, 200)}")
            for feld, wert in c.execute(
                "SELECT feld, wert FROM entityos_profil WHERE feld LIKE ? OR wert LIKE ?",
                (like, like),
            ):
                treffer.append(f"[entityos_profil] {feld}: {_kurz(wert, 200)}")
            try:
                for projekt, beschr in c.execute(
                    "SELECT projekt, beschreibung FROM entityos_projekte "
                    "WHERE projekt LIKE ? OR beschreibung LIKE ?", (like, like),
                ):
                    treffer.append(f"[entityos_projekte] {projekt}: {_kurz(beschr, 200)}")
            except Exception:
                pass
            try:
                for auftrag, womit, stand in c.execute(
                    "SELECT auftrag, womit, stand FROM entityos_arbeitspaket "
                    "WHERE auftrag LIKE ? OR womit LIKE ?", (like, like),
                ):
                    treffer.append(f"[entityos_arbeitspaket:{stand}] {_kurz(auftrag, 150)} ({womit})")
            except Exception:
                pass
            try:
                for titel, text in c.execute(
                    "SELECT titel, snippet(fts_index_artikel, 1, '[', ']', '…', 12) "
                    "FROM fts_index_artikel WHERE fts_index_artikel MATCH ? LIMIT 12",
                    (begriff,),
                ):
                    treffer.append(f"[artikel] {titel}: {text}")
            except Exception:
                try:
                    for aid, titel in c.execute(
                        "SELECT artikel_id, titel FROM artikel_suchtext "
                        "WHERE titel LIKE ? OR text LIKE ? LIMIT 12", (like, like),
                    ):
                        treffer.append(f"[artikel:{aid}] {titel}")
                except Exception:
                    pass
    except Exception as e:
        return f"Suche fehlgeschlagen: {e}"
    # 2026-08-27-Nachtrag: Themen-Register mitdurchsuchen (eigene Verbindung,
    # bewusst AUSSERHALB des obigen `with`-Blocks -- themen_status_suchen()
    # oeffnet/schliesst sich selbst, gleiches Fail-silent-Verhalten wie
    # jeder andere Block hier).
    for z in themen_status_suchen(begriff, limit=8):
        treffer.append(f"[thema:{z['stand']}] {z['thema']}: {_kurz(z['status_text'], 220)}")
    if not treffer:
        return f"Keine Treffer fuer '{begriff}'."
    return "\n".join(treffer)


# ── Automatischer Abruf (UserPromptSubmit): proaktives Nachschlagen ──────
def _stichwoerter(text: str) -> list:
    k = _konfig()
    stopwoerter = set(k["stopwoerter"])
    kurz_ausnahmen = set(k["kurz_ausnahmen"])
    roh = re.findall(r"[a-zA-ZäöüÄÖÜß0-9_-]{2,}", text.lower())
    gesehen, ergebnis = set(), []
    for w in roh:
        if w in gesehen:
            continue
        if w in stopwoerter:
            continue
        if len(w) < 4 and w not in kurz_ausnahmen:
            continue
        gesehen.add(w)
        ergebnis.append(w)
    return ergebnis[:12]


def _sicherstellen_gezeigt_tabelle(c) -> None:
    c.execute(
        "CREATE TABLE IF NOT EXISTS star_abruf_gezeigt("
        "session_id TEXT NOT NULL, item_id TEXT NOT NULL, ts REAL NOT NULL, "
        "PRIMARY KEY(session_id, item_id))"
    )


def _korpus_label(kind: str, value) -> str:
    """PHASE 13/§5.6 Punkt 2 -- kind='fehler'/'gelungen' (durable_memory)
    UND kind='lehre' (die gespiegelte lehre_eintragen()-Zeile) bekommen beim
    Ausliefern [don't]/[bewaehrt] statt des neutralen [kind]-Praefixes, damit
    die Zeile sofort als Verhaltenshinweis erkennbar ist. Fuer 'lehre'
    entscheidet das value.polaritaet-Feld (von lehre_eintragen() gesetzt);
    ohne lesbares Feld faellt es auf [don't], weil dieses Haus Fehler
    strukturell vor Erfolgen nennt (§5.6-Reihenfolge: "Fehler- und
    Erfolgskorpus", in dieser Reihenfolge)."""
    if kind == "fehler":
        return "[don't]"
    if kind == "gelungen":
        return "[bewährt]"
    if kind == "lehre":
        pol = "dont"
        try:
            d = value if isinstance(value, dict) else json.loads(value)
            pol = str(d.get("polaritaet") or "dont")
        except Exception:
            pass
        return "[bewährt]" if pol == "bewaehrt" else "[don't]"
    return f"[{kind}]"


def automatischer_abruf(session_id: str, nutzertext: str) -> Optional[str]:
    if not nutzertext or not nutzertext.strip():
        return None
    stichwoerter = _stichwoerter(nutzertext)
    schwelle = _konfig()["abruf.generisch_schwelle"]
    if not stichwoerter:
        return None

    kandidaten = []
    # paket-1788112504899 (Gabriels Befund: "Sachinfo war thematisch daneben").
    # DREI gemessene Ursachen, alle hier behoben:
    #   1) Sammelreihenfolge == Wortreihenfolge im Satz. Das erste, oft
    #      generische Wort fuellte alle 4 Slots, BEVOR das tragende Wort drankam.
    #   2) Innerhalb einer Quelle wurde NICHT nach Relevanz sortiert
    #      (durable_memory nach ts DESC = Frische, FTS in rowid-Reihenfolge =
    #      Zufall). Relevanz schlug nie Frische/Zufall.
    #   3) 60 der 290 Artikel im Korpus sind `skill:*` (Werkzeug-Doku, kein
    #      Wissen) und raeumten bei generischen Technikwoertern alles ab.
    # Antwort: JEDE Zeile traegt jetzt einen Score aus (a) Spezifitaet des
    # Stichworts in ihrer Quelle 1/log(2+n) -- seltenes Wort = hohes Signal,
    # (b) Quellgewicht (frischer themen_status > don't/bewaehrt-Korpus >
    # Wissen-Artikel > Profil/Projekt), (c) Bonus, wenn das Stichwort im TITEL/
    # SUBJECT steht statt nur im Fliesstext. Auswahl = Top-N nach Score, nicht
    # Sammelreihenfolge. Rein SQL, kein Modellaufruf (harte Grenze 2).
    # themen_status bleibt zusaetzlich vorrangestellt (siehe unten) -- frischer
    # Cross-Sitzungs-Stand ist das eine Signal, das im Score nicht steckt.
    kandidaten_themen = []
    gesperrte_praefixe = tuple(_konfig()["abruf.gesperrte_artikel_praefixe"] or [])
    gesperrte_paket_teile = [s.lower() for s in
                             (_konfig()["abruf.gesperrte_paket_teilstrings"] or [])]
    endgueltig = []
    try:
        with contextlib.closing(schreib_verbindung()) as c:
            _sicherstellen_gezeigt_tabelle(c)
            bereits_gezeigt = {
                r[0] for r in c.execute(
                    "SELECT item_id FROM star_abruf_gezeigt WHERE session_id=?",
                    (session_id,),
                )
            }

            def _spezifitaet(n: int) -> float:
                # n = Zahl der Dokumente, die dieses Stichwort in dieser Quelle
                # trifft. n=1 -> 0.91, n=8 -> 0.39, n=30 -> 0.24. Ein Wort, das
                # fast den ganzen Korpus trifft, traegt kaum Thema.
                return 1.0 / math.log(2.0 + max(1, n))

            for begriff in stichwoerter:
                like = f"%{begriff}%"

                # DER eigentliche Fix fuer "andere Sitzung fragt, diese
                # Sitzung weiss nichts davon" -- laeuft bei JEDER
                # Nutzernachricht ohnehin schon mit (kein zweiter Hook-
                # Pfad), sucht jetzt zusaetzlich gegen themen_status.
                # Reines SQL/LIKE, kein Modellaufruf -- siehe Docstring
                # des Themen-Register-Abschnitts oben
                # (_themen_status_schreiben() etc.).
                try:
                    anzahl_th = (c.execute(
                        "SELECT COUNT(*) FROM themen_status WHERE thema LIKE ? OR status_text LIKE ?",
                        (like, like),
                    ).fetchone() or (0,))[0]
                    if 0 < anzahl_th <= schwelle:
                        for thema_key, thema, status_text, stand, aktualisiert in c.execute(
                            "SELECT thema_key, thema, status_text, stand, aktualisiert FROM themen_status "
                            "WHERE (thema LIKE ? OR status_text LIKE ?) "
                            "AND NOT EXISTS (SELECT 1 FROM star_abruf_gezeigt g "
                            "  WHERE g.session_id=? AND g.item_id='themen_status:'||thema_key) "
                            "ORDER BY (thema LIKE ?) DESC, aktualisiert DESC LIMIT 6",
                            (like, like, session_id, like),
                        ):
                            item_id = f"themen_status:{thema_key}"
                            if item_id in bereits_gezeigt:
                                continue
                            alter_std = max(0, (_jetzt() - (aktualisiert or _jetzt())) / 3600)
                            titel_bonus = 0.5 if begriff in (thema or "").lower() else 0.0
                            score = (3.0 + titel_bonus) * _spezifitaet(anzahl_th)
                            kandidaten.append((score, item_id, f"[thema:{stand}] {thema} "
                                                               f"(vor {alter_std:.1f}h): "
                                                               f"{_kurz(status_text, 220)}"))
                except sqlite3.OperationalError:
                    pass  # Tabelle noch nicht angelegt (noch kein handoff/reconcile gelaufen)
                except Exception:
                    pass

                anzahl = (c.execute(
                    "SELECT COUNT(*) FROM durable_memory WHERE kind != 'ereignis' "
                    "AND (subject LIKE ? OR value LIKE ?)", (like, like),
                ).fetchone() or (0,))[0]
                if 0 < anzahl <= schwelle:
                    # Relevanz vor Frische: Subject-Treffer zuerst, dann ts DESC.
                    for mem_id, kind, subject, value in c.execute(
                        "SELECT memory_id, kind, subject, value FROM durable_memory "
                        "WHERE kind != 'ereignis' AND (subject LIKE ? OR value LIKE ?) "
                        "AND NOT EXISTS (SELECT 1 FROM star_abruf_gezeigt g "
                        "  WHERE g.session_id=? AND g.item_id='durable_memory:'||memory_id) "
                        "ORDER BY (subject LIKE ?) DESC, ts DESC LIMIT 6",
                        (like, like, session_id, like),
                    ):
                        item_id = f"durable_memory:{mem_id}"
                        if item_id in bereits_gezeigt:
                            continue
                        # Fehler-/Lehre-Zeilen sind Verhaltenshinweise ("mach das
                        # nicht so") -- die sind im Zweifel wichtiger als ein
                        # sachlicher Projekt-Eintrag, deshalb eigenes Gewicht.
                        quelle_gewicht = 2.6 if kind in ("fehler", "lehre") else 2.0
                        titel_bonus = 0.5 if begriff in (subject or "").lower() else 0.0
                        score = (quelle_gewicht + titel_bonus) * _spezifitaet(anzahl)
                        kandidaten.append((score, item_id,
                                           f"{_korpus_label(kind, value)} {subject}: {_kurz(value, 200)}"))

                try:
                    anzahl_a = (c.execute(
                        "SELECT COUNT(*) FROM fts_index_artikel WHERE fts_index_artikel MATCH ?",
                        (begriff,),
                    ).fetchone() or (0,))[0]
                except Exception:
                    anzahl_a = 0
                if 0 < anzahl_a <= schwelle:
                    try:
                        # bm25(): kleiner = besser. Vorher LIMIT 6 OHNE ORDER BY
                        # -- rowid-Reihenfolge, d.h. Zufall, welche 6 Wissen-
                        # Artikel ankamen.
                        for aid, titel, text in c.execute(
                            "SELECT s.artikel_id, f.titel, "
                            "snippet(fts_index_artikel, 1, '', '', '…', 12) "
                            "FROM fts_index_artikel f JOIN artikel_suchtext s ON s.id = f.rowid "
                            "WHERE f.fts_index_artikel MATCH ? "
                            "AND NOT EXISTS (SELECT 1 FROM star_abruf_gezeigt g "
                            "  WHERE g.session_id=? AND g.item_id='artikel:'||s.artikel_id) "
                            "ORDER BY bm25(fts_index_artikel) LIMIT 6",
                            (begriff, session_id),
                        ):
                            item_id = f"artikel:{aid}"
                            if item_id in bereits_gezeigt:
                                continue
                            if gesperrte_praefixe and aid.startswith(gesperrte_praefixe):
                                continue
                            titel_bonus = 0.5 if begriff in (titel or "").lower() else 0.0
                            score = (1.0 + titel_bonus) * _spezifitaet(anzahl_a)
                            kandidaten.append((score, item_id, f"[artikel] {titel}: {_kurz(text, 200)}"))
                    except Exception:
                        pass

                for feld, wert in c.execute(
                    "SELECT feld, wert FROM entityos_profil WHERE (feld LIKE ? OR wert LIKE ?) "
                    "AND NOT EXISTS (SELECT 1 FROM star_abruf_gezeigt g "
                    "  WHERE g.session_id=? AND g.item_id='entityos_profil:'||feld)",
                    (like, like, session_id),
                ):
                    item_id = f"entityos_profil:{feld}"
                    if item_id in bereits_gezeigt:
                        continue
                    titel_bonus = 0.5 if begriff in (feld or "").lower() else 0.0
                    score = (0.8 + titel_bonus) * _spezifitaet(max(1, schwelle))
                    kandidaten.append((score, item_id, f"[profil] {feld}: {_kurz(wert, 200)}"))

                # paket-1788112504899, Regression "grid10" (Koordinator-Messung):
                # das Wort stand ALS offenes Arbeitspaket in der DB und wurde
                # trotzdem nicht geliefert -- `entityos_arbeitspaket` war im
                # Abruf UEBERHAUPT nicht dran. Das ist die Tabelle, in der die
                # laufende Arbeit liegt; ohne sie ist der Abruf blind fuer das,
                # was gerade in Arbeit ist.
                try:
                    anzahl_p = (c.execute(
                        "SELECT COUNT(*) FROM entityos_arbeitspaket WHERE "
                        "auftrag LIKE ? OR womit LIKE ? OR warum LIKE ? OR ergebnis LIKE ?",
                        (like, like, like, like),
                    ).fetchone() or (0,))[0]
                    if 0 < anzahl_p <= schwelle:
                        # Offen/laufende Arbeit zuerst: das ist das, was ein
                        # Gespraech gerade braucht. Fertige/gescheiterte nur
                        # als Rueckfallebene.
                        for paket_id, stand, auftrag, womit in c.execute(
                            "SELECT paket_id, stand, auftrag, womit FROM entityos_arbeitspaket "
                            "WHERE (auftrag LIKE ? OR womit LIKE ? OR warum LIKE ? OR ergebnis LIKE ?) "
                            "AND NOT EXISTS (SELECT 1 FROM star_abruf_gezeigt g "
                            "  WHERE g.session_id=? AND g.item_id='entityos_arbeitspaket:'||paket_id) "
                            "ORDER BY CASE stand WHEN 'offen' THEN 0 WHEN 'laeuft' THEN 1 "
                            "WHEN 'fertig' THEN 2 ELSE 3 END, geaendert DESC LIMIT 6",
                            (like, like, like, like, session_id),
                        ):
                            item_id = f"entityos_arbeitspaket:{paket_id}"
                            if item_id in bereits_gezeigt:
                                continue
                            auftrag_klein = (auftrag or "").lower()
                            if any(s in auftrag_klein for s in gesperrte_paket_teile):
                                continue
                            # Paket-ids sind Lautsches-Muelltone; der Mensch
                            # braucht die Sache, nicht die id.
                            zeile = f"[paket:{stand}] {_kurz(auftrag, 180)}"
                            if womit and womit not in (auftrag or ""):
                                zeile += f" | {_kurz(womit, 60)}"
                            titel_bonus = 0.5 if begriff in (auftrag or "").lower() else 0.0
                            stand_gewicht = 2.4 if stand in ("offen", "laeuft") else 1.4
                            score = (stand_gewicht + titel_bonus) * _spezifitaet(anzahl_p)
                            kandidaten.append((score, item_id, zeile))
                except sqlite3.OperationalError:
                    pass  # Tabelle noch nicht angelegt
                except Exception:
                    pass

                try:
                    for projekt, beschr in c.execute(
                        "SELECT projekt, beschreibung FROM entityos_projekte "
                        "WHERE (projekt LIKE ? OR beschreibung LIKE ?) "
                        "AND NOT EXISTS (SELECT 1 FROM star_abruf_gezeigt g "
                        "  WHERE g.session_id=? AND g.item_id='entityos_projekte:'||projekt)",
                        (like, like, session_id),
                    ):
                        item_id = f"entityos_projekte:{projekt}"
                        if item_id in bereits_gezeigt:
                            continue
                        titel_bonus = 0.5 if begriff in (projekt or "").lower() else 0.0
                        score = (0.8 + titel_bonus) * _spezifitaet(max(1, schwelle))
                        kandidaten.append((score, item_id, f"[projekt] {projekt}: {_kurz(beschr, 200)}"))
                except Exception:
                    pass

            # Top-N nach Score, nicht Sammelreihenfolge. themen_status-Treffer
            # laufen NICHT mehr als separate Vorliste: sie sind jetzt im selben
            # Score-System (Quellgewicht 3.0) und gewinnen dort von selbst,
            # wenn sie spezifisch sind. Der alte Vorlist-Mechanismus hatte den
            # Fehler, dass sie IMMER gewannen -- auch wenn sie nur per
            # generischem Wort ("steht", "steht es um") getroffen hatten.
            gesehen = set()
            for _score, item_id, zeile in sorted(kandidaten, key=lambda x: x[0], reverse=True):
                if item_id in gesehen:
                    continue
                gesehen.add(item_id)
                endgueltig.append((item_id, zeile))
                if len(endgueltig) >= 4:
                    break

            if endgueltig:
                jetzt = _jetzt()
                c.executemany(
                    "INSERT OR IGNORE INTO star_abruf_gezeigt (session_id, item_id, ts) "
                    "VALUES (?,?,?)",
                    [(session_id, item_id, jetzt) for item_id, _ in endgueltig],
                )
                c.commit()
    except Exception as e:
        _log(f"automatischer_abruf fehler sid={session_id}: {e}")
        endgueltig = []

    raum_zeilen: list = []
    if _PAKET_OK and _paket is not None:
        try:
            with contextlib.closing(schreib_verbindung()) as c2:
                c2.row_factory = sqlite3.Row
                # PHASE 4b.1 -- Needle-Router bestimmt moeglichkeitsraum()s
                # eigene Groesse VOR dem Zusammenbau, nicht danach
                # zurechtgestutzt. Siehe Kommentarblock ueber NeedleRouter.
                t_needle0 = time.perf_counter_ns()
                offene_lesarten = [
                    row[0] for row in c2.execute(
                        "SELECT lesart_id FROM wm_lesart WHERE offen=1 LIMIT 10"
                    ).fetchall()
                ]
                needle_welt = {
                    "cross_session_collision": lease_hinweise(ausser_sitzung=session_id),
                    "durable_memory": [iid for _s, iid, _z in kandidaten
                                       if iid.startswith("durable_memory:")],
                }
                needle_resources = [iid for iid, _ in endgueltig]
                needle_entscheidung = NeedleRouter.analyze(
                    nutzertext, needle_welt, offene_lesarten, needle_resources,
                )
                needle_dauer_us = (time.perf_counter_ns() - t_needle0) // 1000
                mr_budget = _needle_budget_zu_item_budget(needle_entscheidung["context_budget"])
                _log(
                    f"needle sid={session_id} dauer_us={needle_dauer_us} "
                    f"route={needle_entscheidung['route']} "
                    f"context_budget={needle_entscheidung['context_budget']} "
                    f"mr_budget={mr_budget} score={needle_entscheidung['score']} "
                    f"reasons={','.join(needle_entscheidung['reasons'])}"
                )
                r = moeglichkeitsraum(nutzertext, budget=mr_budget, wirt="ai-core-node", conn=c2,
                                      session_id=session_id)
            raum = r.get("raum")
            if raum:
                if raum.get("plausibel_richtung"):
                    raum_zeilen.append("  Moeglich (aus dem Korpus, keine Antwort):")
                    for s in raum["plausibel_richtung"][:4]:
                        raum_zeilen.append(f"    + {s}")
                if raum.get("spricht_dagegen"):
                    raum_zeilen.append("  Spricht dagegen:")
                    for s in raum["spricht_dagegen"][:4]:
                        raum_zeilen.append(f"    - {s}")
                if raum.get("warnung"):
                    raum_zeilen.append(f"  ({raum['warnung']})")
            # P5d: Kollisionsmeldung, keine Sperre -- auch wenn kein Raum
            # gebaut werden konnte, koennen Pacht-Hinweise vorliegen.
            # paket-1788112504899, Nullfall-Regression "adolf hitler":
            # bisher lieferten die Pacht-Hinweise bei JEDER Nachricht bis zu
            # 4 Zeilen -- themafremd, rein weil gerade irgendeine Datei bei
            # irgendeiner Sitzung in Arbeit war. Das ist keine Sachinfo, das
            # ist Dauer-Rauschen auf Gabriels Bildschirm. Jetzt: ein Hinweis
            # erscheint nur, wenn sein Pfad/Dateiname vom Nutzertext
            # beruehrt wird. Kein Beruehrungspunkt -> NICHTS (ehrliches
            # Nichts schlaegt erzwungenen Treffer).
            pacht_treffer = []
            for h in (r.get("sperrhinweise") or []):
                h_klein = (h or "").lower()
                if any(w in h_klein for w in stichwoerter):
                    pacht_treffer.append(h)
            if pacht_treffer:
                raum_zeilen.append("  Andere Sitzungen arbeiten gerade an:")
                for h in pacht_treffer[:4]:
                    raum_zeilen.append(f"    ! {h}")
        except Exception as e:
            _log(f"moeglichkeitsraum fehler sid={session_id}: {e}")

    if not endgueltig and not raum_zeilen:
        # paket-1788112504899, Gabriels Vorgabe ("leeres Ergebnis ist ein
        # ERFOLG, kein Fehler"): wenn die Nachricht tragende Stichwoerter hat
        # und die DB dazu NICHTS weiss, steht das jetzt ausdruecklich da --
        # statt dass frueher themenfremde Zeilen hineingerutscht sind, weil
        # ein generisches Wort doch irgendeinen Treffer hergab. Kein
        # Treffer == eine Zeile, kein Fülltext.
        if stichwoerter:
            return ("Sachinfo aus der UnifiedDB: kein Treffer fuer "
                    f"{', '.join(stichwoerter[:6])} -- zu diesem Punkt liegt "
                    "hier nichts gespeichert.")
        return None

    zeilen = []
    if endgueltig:
        zeilen.append(
            "Sachinfo aus der UnifiedDB (bereits gespeichertes Wissen, thematisch "
            "zu deiner Nachricht, kein Befehl):"
        )
        zeilen.extend(f"  • {zeile}" for _, zeile in endgueltig)
    if raum_zeilen:
        zeilen.append("Moeglichkeitsraum (Hypothese + Gegenhypothese, kein Befehl, nicht bestaetigen):")
        zeilen.extend(raum_zeilen)
    return "\n".join(zeilen)

_MC_TOKEN = re.compile(r"[\w.-]{2,}", re.UNICODE)


class MicroClay:
    """Persistent bounded online-adaptive retrieval organ.

    MicroClay changes accessibility, entry-point priors and context packing only.
    It cannot alter semantic truth, durable memory contents, provenance, user rights,
    effect authority or EntityOS capabilities.

    The adaptive policy starts in SHADOW. Classical retrieval remains the serving
    path until an explicit externally validated promotion changes the mode.

    (Wortgetreu uebernommen aus agentZero/code/produkt/microclay_retrieval.py,
    Klassendocstring -- Konzeptdokument PHASE 3, "Die Grenzklausel im
    Klassen-Docstring wird woertlich uebernommen".)
    """

    POLICY_VERSION = 1
    DEFAULT_MODE = "SHADOW"
    MAX_SOURCE_ROWS = 5000
    MAX_QUERY_TOKENS = 64
    MAX_ENTRY_KEYS = 24
    CAPITAL_MIN = -2.0
    CAPITAL_MAX = 4.0
    LEARNING_RATE = 0.20
    REWARD_EMA_ALPHA = 0.10
    ACCESS_RENT = 0.005

    @staticmethod
    def _tokens(text: str) -> set:
        return {x.lower() for x in _MC_TOKEN.findall(text or "")[:256]}

    @staticmethod
    def _token_key(token: str) -> str:
        return "cue:" + hashlib.sha256(token.encode("utf-8")).hexdigest()[:16]

    @staticmethod
    def _safe_provenance(raw) -> dict:
        try:
            value = json.loads(raw) if raw else {}
            return value if isinstance(value, dict) else {"raw_type": type(value).__name__}
        except Exception:
            return {"parse_error": True}

    @classmethod
    def mode(cls, c: sqlite3.Connection) -> str:
        row = c.execute("SELECT mode FROM retrieval_policy_state WHERE singleton=1").fetchone()
        return row[0] if row else cls.DEFAULT_MODE

    @classmethod
    def _policy_capitals(cls, c: sqlite3.Connection) -> dict:
        return {
            str(row[0]): float(row[1])
            for row in c.execute("SELECT entry_key,capital FROM retrieval_entrypoint_policy")
        }

    @classmethod
    def _entry_keys(cls, kind: str, subject: str, query_tokens: set, value_tokens: set) -> tuple:
        keys = [f"kind:{kind}", f"subject:{subject}"]
        for token in sorted(query_tokens.intersection(value_tokens))[: cls.MAX_ENTRY_KEYS - 2]:
            keys.append(cls._token_key(token))
        return tuple(keys)

    @classmethod
    def _score_candidate(cls, row, query_tokens: set, now: float, capitals: dict) -> dict:
        value = str(row["value"])
        subject = str(row["subject"])
        kind = str(row["kind"])
        value_tokens = cls._tokens(value + " " + subject + " " + kind)
        if query_tokens:
            overlap = len(query_tokens.intersection(value_tokens))
            coverage = overlap / max(1, len(query_tokens))
            precision = overlap / max(1, len(value_tokens))
            lexical = min(1.0, 0.75 * coverage + 0.25 * precision)
        else:
            lexical = 0.0

        age_days = max(0.0, (now - float(row["ts"])) / 86400.0)
        recency = math.exp(-age_days / 30.0)
        structured = 1.0 if kind in {
            "fact", "decision", "task", "commitment", "preference", "open", "open_question"
        } else 0.4
        # Kein Cue-Treffer heisst UNKNOWN, nicht "vermutlich relevant weil
        # neu" -- woertlich aus dem Original uebernommen (Z.132-133 dort).
        baseline = (
            (0.82 * lexical) + (0.12 * recency) + (0.06 * structured)
            if lexical > 0.0 else 0.0
        )

        keys = cls._entry_keys(kind, subject, query_tokens, value_tokens)
        learned = sum(
            max(cls.CAPITAL_MIN, min(cls.CAPITAL_MAX, capitals.get(k, 0.0)))
            for k in keys
        )
        learned /= max(1, len(keys))
        adaptive = baseline + 0.08 * learned

        return {
            "memory_id": str(row["memory_id"]), "kind": kind, "subject": subject,
            "value": value, "ts": float(row["ts"]),
            "provenance": cls._safe_provenance(row["provenance"]),
            "baseline_score": baseline, "adaptive_score": adaptive, "entry_keys": keys,
            # PHASE 6/ZECP10, P6b Punkt 4 -- zusaetzlich zu den zwei
            # kombinierten Werten (baseline_score/adaptive_score, unveraendert)
            # auch die beiden Faktoren einzeln durchgereicht, NICHTS neu
            # berechnet: lexical->relevanz, recency->neuheit in gw_wettbewerb.
            "lexical": lexical, "recency": recency,
        }

    @classmethod
    def retrieve(
        cls, c: sqlite3.Connection, *, query: str, user_id: str, session_id: str,
        turn_id: str, episode_id: str, causal_id: str, generation: int,
        limit: int = 24, budget_chars: int = 12_000,
    ) -> dict:
        if not isinstance(query, str) or not query.strip():
            return {"items": [], "receipt": None, "grund": "leere_anfrage"}
        if limit < 1 or limit > 100 or budget_chars < 256 or budget_chars > 200_000:
            return {"items": [], "receipt": None, "grund": "ungueltige_parameter"}

        c.row_factory = sqlite3.Row
        query_hash = hashlib.sha256(query.encode("utf-8")).hexdigest()
        existing = c.execute(
            "SELECT * FROM retrieval_episodes WHERE causal_id=?", (causal_id,)
        ).fetchone()
        if existing:
            # Idempotenz wie im Original: derselbe causal_id-Aufruf ein zweites
            # Mal liefert die schon gebuchte Zeile zurueck statt neu zu
            # schreiben. In `_microclay_schatten()` unten bekommt jeder Aufruf
            # einen frischen causal_id (Star fuehrt anders als Zero noch
            # keinen eigenen Turn-scharfen causal_id -- Stars `active_turns`
            # ist eine Zeile je SITZUNG, nicht je Runde), dieser Zweig greift
            # also im Regelbetrieb nicht, bleibt aber als Schutz stehen.
            return {
                "items": [], "receipt": {
                    "retrieval_id": existing["retrieval_id"], "replayed": True,
                    "mode": existing["mode"], "status": existing["status"],
                    "candidate_count": None, "selected_count": None,
                },
            }

        query_tokens = set(sorted(cls._tokens(query))[: cls.MAX_QUERY_TOKENS])
        rows = c.execute(
            "SELECT * FROM durable_memory "
            "WHERE kind NOT IN ('source_archive','compression_receipt','compression_clearance') "
            "AND (user_id IS NULL OR user_id=?) "
            "AND (source_turn_id IS NULL OR source_turn_id<>?) "
            "ORDER BY ts DESC LIMIT ?",
            (user_id, turn_id, cls.MAX_SOURCE_ROWS),
        ).fetchall()
        now = _jetzt()
        capitals = cls._policy_capitals(c)
        candidates = [cls._score_candidate(r, query_tokens, now, capitals) for r in rows]
        baseline = sorted(
            candidates, key=lambda x: (x["baseline_score"], x["ts"], x["memory_id"]), reverse=True,
        )
        adaptive = sorted(
            candidates, key=lambda x: (x["adaptive_score"], x["ts"], x["memory_id"]), reverse=True,
        )

        # PHASE 6/ZECP10, P6b Punkt 6 -- Anti-Dauersieger-Malus. Liest den VOR
        # dieser Runde bestehenden Streak-Zaehler (gw_dauersieger, aus der
        # VORHERIGEN Runde fortgeschrieben, siehe Ende dieser Methode) und
        # zieht ihn -- falls die Schwelle erreicht ist -- von GENAU dem einen
        # Kandidaten ab, der ihn haelt. Wirkt NUR auf adaptive_score, NIE auf
        # baseline_score: die "serving"-Entscheidung unten bleibt in SHADOW
        # (der Vorgabemodus) ausschliesslich von baseline_score abhaengig, der
        # Malus kann also die tatsaechlich ausgelieferte Liste erst
        # beeinflussen, wenn/falls MicroClay separat auf ACTIVE befoerdert
        # wird (eigener, gegateter Schritt) -- bewusste, dokumentierte Folge
        # von §1.9/SHADOW-vor-ACTIVE, keine Abschwaechung des Auftrags.
        _sicherstellen_gw_dauersieger_tabelle(c)
        k_gw = _konfig()
        gw_schwelle = int(k_gw.get("gw.dauersieger_schwelle", 5))
        gw_malus = float(k_gw.get("gw.dauersieger_malus", 0.5))
        gw_vorheriger_sieger = c.execute(
            "SELECT artikel_id, serie FROM gw_dauersieger ORDER BY zuletzt_ts DESC LIMIT 1"
        ).fetchone()
        if gw_vorheriger_sieger and int(gw_vorheriger_sieger[1]) >= gw_schwelle:
            gw_ziel_id = str(gw_vorheriger_sieger[0])
            for item in adaptive:
                if item["memory_id"] == gw_ziel_id:
                    item["adaptive_score"] = item["adaptive_score"] - gw_malus
                    break
            adaptive = sorted(
                adaptive, key=lambda x: (x["adaptive_score"], x["ts"], x["memory_id"]), reverse=True,
            )

        mode = cls.mode(c)
        # Unveraendert aus dem Original: WELCHE Liste bedient wird, haengt
        # ausschliesslich vom Modus ab. In SHADOW (Vorgabe dieser Phase,
        # immer) ist das die BASELINE-Liste -- adaptive wird nur berechnet
        # und in shadow_memory_ids festgehalten, nie ausgeliefert.
        serving = adaptive if mode == "ACTIVE" else baseline
        selected: list = []
        chars = 0
        for item in serving:
            if item["baseline_score"] <= 0.0:
                continue
            item_chars = len(item["value"])
            if selected and chars + item_chars > budget_chars:
                continue
            selected.append(item)
            chars += item_chars
            if len(selected) >= limit:
                break

        retrieval_id = str(uuid.uuid4())
        token_hashes = [cls._token_key(x)[4:] for x in sorted(query_tokens)]
        selected_ids = [x["memory_id"] for x in selected]
        shadow_ids = [x["memory_id"] for x in adaptive if x["baseline_score"] > 0.0][:limit]
        entry_keys = sorted({k for item in selected for k in item["entry_keys"]})[:256]

        c.execute(
            "INSERT INTO retrieval_episodes("
            "retrieval_id,turn_id,episode_id,causal_id,session_id,user_id,generation,"
            "policy_version,mode,query_hash,query_token_hashes,selected_memory_ids,"
            "shadow_memory_ids,entry_keys,budget_chars,chars_selected,status,ts"
            ") VALUES(?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?,?)",
            (
                retrieval_id, turn_id, episode_id, causal_id, session_id, user_id, int(generation),
                cls.POLICY_VERSION, mode, query_hash, json.dumps(token_hashes),
                json.dumps(selected_ids), json.dumps(shadow_ids), json.dumps(entry_keys),
                int(budget_chars), int(chars),
                "PRESENT" if selected else "UNKNOWN", now,
            ),
        )

        # PHASE 6/ZECP10, P6b Punkte 4/5/6/7 -- gw_wettbewerb, gw_ausstrahlung,
        # Dauersieger-Fortschreibung, kausaler Wiedereintritt. Alle Werte kommen
        # UNVERAENDERT aus der oben bereits berechneten `adaptive`-Rangfolge
        # (inkl. des ggf. angewandten Dauersieger-Malus) -- hier wird nichts
        # neu bewertet, nur festgehalten (Konzeptdokument P6b Punkt 4).
        # `rang` zaehlt die VOLLE Kandidatenliste (das war "zur Wahl"),
        # persistiert wird nur eine begrenzte Teilmenge -- Budget-Grund:
        # UserPromptSubmit hat 1.8s (§1.9.3), Tausende Nullpunkt-Kandidaten
        # (baseline_score==0, MicroClays eigenes "UNKNOWN, kein Cue-Treffer")
        # je Abruf mitzuschreiben wuerde das reissen und ist auch inhaltlich
        # kein "Wettbewerb" -- nur zugelassene und/oder tatsaechlich
        # (schatten-)ausgestrahlte Kandidaten gelten als Teilnehmer.
        selected_id_set = set(selected_ids)
        shadow_id_set = set(shadow_ids)
        GW_WETTBEWERB_CAP = 60
        gw_geschrieben = 0
        for rang, item in enumerate(adaptive, start=1):
            zugelassen = 1 if item["baseline_score"] > 0.0 else 0
            gewaehlt = 1 if item["memory_id"] in selected_id_set else 0
            if not (zugelassen or item["memory_id"] in shadow_id_set):
                continue
            if gw_geschrieben >= GW_WETTBEWERB_CAP and not gewaehlt:
                continue
            c.execute(
                "INSERT INTO gw_wettbewerb(ts,turn_id,session_id,artikel_id,rang,"
                "salienz,relevanz,neuheit,wert,zugelassen,gewaehlt) VALUES(?,?,?,?,?,?,?,?,?,?,?)",
                (
                    now, turn_id, session_id, item["memory_id"], rang,
                    item["adaptive_score"], item["lexical"], item["recency"],
                    item["baseline_score"], zugelassen, gewaehlt,
                ),
            )
            gw_geschrieben += 1

        for mid in selected_ids:
            gw_item = next((x for x in adaptive if x["memory_id"] == mid), None)
            gw_wert = gw_item["baseline_score"] if gw_item else 0.0
            c.execute(
                "INSERT INTO gw_ausstrahlung(ts,turn_id,session_id,artikel_id,wert) "
                "VALUES(?,?,?,?,?)",
                (now, turn_id, session_id, mid, gw_wert),
            )
            # P6b Punkt 7 -- kausaler Wiedereintritt: dieselbe artikel_id war in
            # einer FRUEHEREN Runde DERSELBEN Sitzung schon einmal ausgestrahlt.
            # Neue causal_episodes-Zeile (effect_id=NULL -- Spalte ist nullable,
            # kein Effekt, sondern ein Wiedereintrittsereignis), reentered=1.
            gw_frueher = c.execute(
                "SELECT 1 FROM gw_ausstrahlung WHERE artikel_id=? AND session_id=? AND ts<? LIMIT 1",
                (mid, session_id, now),
            ).fetchone()
            if gw_frueher:
                c.execute(
                    "INSERT INTO causal_episodes(causal_id,episode_id,effect_id,"
                    "observation_turn_id,outcome_hash,credit,reentered,ts) "
                    "VALUES(?,?,?,?,?,?,?,?)",
                    (_uid("c-reentry"), episode_id, None, turn_id, None, 0, 1, now),
                )

        # Streak fuer die NAECHSTE Runde fortschreiben -- Rang-1 unter den
        # ZUGELASSENEN Kandidaten der (ggf. malus-korrigierten) adaptive-Liste.
        gw_sieger = next((x for x in adaptive if x["baseline_score"] > 0.0), None)
        if gw_sieger:
            if gw_vorheriger_sieger and str(gw_vorheriger_sieger[0]) == gw_sieger["memory_id"]:
                gw_neue_serie = int(gw_vorheriger_sieger[1]) + 1
            else:
                gw_neue_serie = 1
            c.execute(
                "INSERT INTO gw_dauersieger(artikel_id,serie,zuletzt_ts,zuletzt_session) "
                "VALUES(?,?,?,?) ON CONFLICT(artikel_id) DO UPDATE SET "
                "serie=excluded.serie, zuletzt_ts=excluded.zuletzt_ts, "
                "zuletzt_session=excluded.zuletzt_session",
                (gw_sieger["memory_id"], gw_neue_serie, now, session_id),
            )

        return {
            "items": [
                {"memory_id": x["memory_id"], "kind": x["kind"], "subject": x["subject"],
                 "value": x["value"], "ts": x["ts"], "provenance": x["provenance"],
                 "retrieval_score": x["baseline_score"]}
                for x in selected
            ],
            "receipt": {
                "retrieval_id": retrieval_id, "policy_version": cls.POLICY_VERSION, "mode": mode,
                "query_sha256": query_hash, "candidate_count": len(candidates),
                "selected_count": len(selected), "chars_selected": chars,
                "status": "PRESENT" if selected else "UNKNOWN",
                "shadow_differs": selected_ids != shadow_ids,
                "truth_authority": False, "effect_authority": False,
            },
        }

    @classmethod
    def _apply_credit(cls, c: sqlite3.Connection, row, *, credit: float, signal_class: str,
                       receipt_id: Optional[str] = None) -> dict:
        """Die Kapitalbuchung selbst -- LEARNING_RATE/REWARD_EMA_ALPHA/
        ACCESS_RENT unveraendert aus dem Original, nur auf sqlite3.Connection
        umgestellt. Kein Generationszaun hier: gebucht werden ausschliesslich
        `retrieval_entrypoint_policy` und `retrieval_feedback`, beide
        generationsfrei -- es gibt hier keinen Weltzustand, in den eine
        veraltete Annahme hineinschreiben koennte (Original-Begruendung,
        woertlich uebernommen)."""
        value = max(-1.0, min(1.0, float(credit)))
        generation = int(row["generation"])
        causal_id = str(row["causal_id"])

        rid = receipt_id or hashlib.sha256(
            f"{row['retrieval_id']}\x1f{causal_id}\x1f{generation}\x1f{signal_class}\x1f{value:.9f}".encode()
        ).hexdigest()
        exists = c.execute("SELECT 1 FROM retrieval_feedback WHERE receipt_id=?", (rid,)).fetchone()
        if exists:
            return {"applied": False, "reason": "duplicate_receipt", "receipt_id": rid}

        keys = json.loads(row["entry_keys"]) if row["entry_keys"] else []
        now = _jetzt()
        for key in keys:
            current = c.execute(
                "SELECT capital,reward_ema,pulls FROM retrieval_entrypoint_policy WHERE entry_key=?",
                (key,),
            ).fetchone()
            if current:
                old_capital, old_ema, pulls = float(current[0]), float(current[1]), int(current[2])
            else:
                old_capital, old_ema, pulls = 0.0, 0.0, 0
            new_ema = (1.0 - cls.REWARD_EMA_ALPHA) * old_ema + cls.REWARD_EMA_ALPHA * value
            delta = cls.LEARNING_RATE * (value - old_ema) - cls.ACCESS_RENT
            new_capital = max(cls.CAPITAL_MIN, min(cls.CAPITAL_MAX, old_capital + delta))
            c.execute(
                "INSERT INTO retrieval_entrypoint_policy(entry_key,capital,reward_ema,pulls,updated_at) "
                "VALUES(?,?,?,?,?) ON CONFLICT(entry_key) DO UPDATE SET "
                "capital=excluded.capital,reward_ema=excluded.reward_ema,"
                "pulls=excluded.pulls,updated_at=excluded.updated_at",
                (key, new_capital, new_ema, pulls + 1, now),
            )

        c.execute(
            "INSERT INTO retrieval_feedback(receipt_id,retrieval_id,causal_id,generation,"
            "credit,signal_class,ts) VALUES(?,?,?,?,?,?,?)",
            (rid, row["retrieval_id"], causal_id, generation, value, str(signal_class)[:80], now),
        )
        return {
            "applied": True, "receipt_id": rid, "policy_version": cls.POLICY_VERSION,
            "mode": row["mode"], "updated_entrypoints": len(keys),
            "truth_authority": False, "effect_authority": False,
        }

    # ── Das Kreditsignal (Konzeptdokument PHASE 3) ────────────────────────
    #
    # Zero bekommt Kredit aus verifizierten Effekten und menschlicher
    # Rueckmeldung (`learn_from_verified_outcomes`/`menschliche_rueckmeldung`,
    # beide NICHT portiert, siehe Abschnittskopf oben). Star braucht ein
    # eigenes, MECHANISCHES Signal:
    #
    #   PRIMAER (mechanisch, Stop-Hook): ein in dieser Runde von MicroClay
    #   ausgewaehlter Treffer gilt als "hat getragen", wenn seine memory_id
    #   ODER ein charakteristisches Token seines subject in einem Wirkungs-
    #   ziel dieser Runde (effects.target/argv) ODER im Assistententext
    #   dieser Runde (aus transcript_path) vorkommt.
    #
    #   ⚠️ DAS IST EINE VERMUTUNG UEBER NUETZLICHKEIT, KEINE MESSUNG VON
    #   NUETZLICHKEIT. "Erwaehnt" heisst nicht "hat geholfen" -- ein Treffer
    #   kann zufaellig erwaehnt und trotzdem irrelevant gewesen sein, und ein
    #   Treffer kann getragen haben, ohne dass sein Wortlaut je woertlich
    #   auftaucht. Deshalb zaehlt dieses Signal NICHT allein fuer eine
    #   Befoerderung (siehe `befoerderung_pruefen()` unten).
    #
    #   SEKUNDAER (explizit, Vorrang): `stern nuetzlich <id>` / `stern
    #   nutzlos <id>` (siehe `mensch_urteil()`) schreibt ein
    #   signal_class='mensch:...'-Urteil zu GENAU dieser retrieval_id.
    @classmethod
    def kreditsignal_mechanisch(cls, c: sqlite3.Connection, session_id: str, transcript_path) -> dict:
        c.row_factory = sqlite3.Row
        offen = c.execute(
            "SELECT * FROM retrieval_episodes WHERE session_id=? "
            "AND retrieval_id NOT IN (SELECT retrieval_id FROM retrieval_feedback) "
            "ORDER BY ts DESC LIMIT 20",
            (session_id,),
        ).fetchall()
        if not offen:
            return {"geprueft": 0, "getragen": 0}

        assistent_text = _transcript_assistenttext_lesen(transcript_path).lower()
        aktiv = c.execute(
            "SELECT episode_id FROM active_turns WHERE session_id=?", (session_id,)
        ).fetchone()
        episode_id = aktiv["episode_id"] if aktiv else None
        jetzt = _jetzt()

        geprueft = 0
        getragen = 0
        for r in offen:
            geprueft += 1
            effekte_text = ""
            if episode_id:
                for erow in c.execute(
                    "SELECT target, argv FROM effects WHERE episode_id=? AND ts>=? AND ts<=?",
                    (episode_id, r["ts"], jetzt),
                ):
                    effekte_text += f" {erow[0]} {erow[1]}"
            haystack = (effekte_text + " " + assistent_text).lower()

            selected_ids = json.loads(r["selected_memory_ids"] or "[]")
            trug = False
            for mid in selected_ids:
                if mid.lower() in haystack:
                    trug = True
                    break
                mrow = c.execute(
                    "SELECT subject FROM durable_memory WHERE memory_id=?", (mid,)
                ).fetchone()
                subject = mrow[0] if mrow else ""
                if any(tok in haystack for tok in _stichwoerter(subject)):
                    trug = True
                    break

            credit = 1.0 if trug else 0.0
            signal_class = "mechanisch:erwaehnt" if trug else "mechanisch:ungenutzt"
            ergebnis = cls._apply_credit(c, r, credit=credit, signal_class=signal_class)
            if trug and ergebnis.get("applied"):
                getragen += 1

            # PHASE 6/ZECP10, P6b Punkt 8 -- gw_lernen, REIN BEOBACHTEND.
            # Schreibt boost/n/belohnt je (merkmal,artikel_id); niemand liest
            # das zurueck in eine Bewertung -- _policy_capitals() (oben) bleibt
            # ausschliesslich an retrieval_entrypoint_policy gebunden. Kein
            # exaktes Pro-Kandidat-Mapping verfuegbar hier (retrieval_episodes
            # speichert nur die RUNDEN-weite Vereinigung der entry_keys, nicht
            # je Kandidat) -- deshalb bewusst vereinfacht: Kreuzprodukt aus
            # (gekappten) entry_keys x selected_ids DIESER Runde. Das ordnet
            # einem Merkmal auch Artikel zu, die es nicht exakt ausgeloest
            # haben -- fuer ein Beobachtungssignal, das nie in eine
            # Entscheidung zurueckfliesst, hingenommen (offen gelegt, nicht
            # verschwiegen). Gekappt (6x6=36 je Runde) wegen des 2.0s-
            # Stop-Budgets bei bis zu 20 offenen Runden je Aufruf.
            if _konfig().get("gw.lernen_aktiv", 1):
                gw_keys = (json.loads(r["entry_keys"]) if r["entry_keys"] else [])[:6]
                gw_mids = selected_ids[:6]
                gw_delta = 0.05 if trug else -0.02
                for gw_merkmal in gw_keys:
                    for gw_mid in gw_mids:
                        gw_row = c.execute(
                            "SELECT boost, n FROM gw_lernen WHERE merkmal=? AND artikel_id=?",
                            (gw_merkmal, gw_mid),
                        ).fetchone()
                        gw_alt_boost = float(gw_row[0]) if gw_row else 0.0
                        gw_alt_n = int(gw_row[1]) if gw_row else 0
                        c.execute(
                            "INSERT INTO gw_lernen(merkmal,artikel_id,boost,n,belohnt,ts) "
                            "VALUES(?,?,?,?,?,?) ON CONFLICT(merkmal,artikel_id) DO UPDATE SET "
                            "boost=excluded.boost, n=excluded.n, belohnt=excluded.belohnt, ts=excluded.ts",
                            (
                                gw_merkmal, gw_mid, gw_alt_boost + gw_delta, gw_alt_n + 1,
                                1 if trug else 0, jetzt,
                            ),
                        )
        return {"geprueft": geprueft, "getragen": getragen}

    @classmethod
    def mensch_urteil(cls, c: sqlite3.Connection, retrieval_id: str, urteil: str) -> dict:
        """`stern nuetzlich <id>` / `stern nutzlos <id>` -- SEKUNDAERES,
        VORRANGIGES Kreditsignal. Schreibt eine NEUE Quittung mit
        signal_class='mensch:...' zu genau dieser retrieval_id; ersetzt keine
        Zeile, sondern kommt hinzu -- `befoerderung_pruefen()` zaehlt
        DISTINCT-Quittungen mit `signal_class LIKE 'mensch:%'`, unabhaengig
        davon, ob vorher schon ein mechanisches Urteil zu derselben
        retrieval_id vorlag. Das mechanische Urteil wird dadurch faktisch
        ueberstimmt, sobald irgendein Auswerter die JUENGSTE Quittung nimmt."""
        gewichte = {"nuetzlich": 1.0, "nutzlos": -1.0}
        if urteil not in gewichte:
            return {"applied": False, "reason": f"unbekanntes urteil {urteil!r}, erlaubt: nuetzlich/nutzlos"}
        c.row_factory = sqlite3.Row
        row = c.execute(
            "SELECT * FROM retrieval_episodes WHERE retrieval_id=?", (retrieval_id,)
        ).fetchone()
        if not row:
            return {"applied": False, "reason": "retrieval_id_unbekannt"}
        return cls._apply_credit(c, row, credit=gewichte[urteil], signal_class=f"mensch:{urteil}")

    # ── Befoerderung: SHADOW -> ACTIVE ─────────────────────────────────────
    #
    # ⭐ Bis hierher schrieb NICHTS ausser der Tabellenerzeugung selbst in
    # `retrieval_policy_state` -- der Modus blieb fuer immer SHADOW, egal wie
    # viel geprueftes Lernsignal in `retrieval_entrypoint_policy` ankam. Die
    # Klassendoku sagt "bis eine explizit extern validierte Befoerderung den
    # Modus aendert" -- diese Befoerderung existierte nirgends im Code. Reine
    # Beobachtung ohne einen Uebergang ist keine SHADOW-Politik, sondern eine
    # Sackgasse, die wie eine Politik aussieht.
    # (Woertlich aus dem Original uebernommen, agentZero/.../
    # microclay_retrieval.py Z.287-293 -- das ist Zeros eigener
    # Korrekturkommentar zu einem frueheren Bug BEI ZERO, kein Star-Befund.)
    #
    # ⭐⭐⭐ WELCHE QUITTUNGEN NICHT ZAEHLEN.
    # Gemessen 2026-08-21 [FUND, bei Zero]: `retrieval_policy_state` stand auf
    # **ACTIVE**, befoerdert von **20 Zeilen, die ausnahmslos
    # `bootstrap:geholfen` und `bootstrap:daneben` waren** -- Setzlinge aus
    # dem Bau, kein einziges unabhaengiges Signal. Die Zaehlung darueber
    # versprach im Kommentar ausdruecklich "validiertes Lernsignal" und
    # zaehlte in Wahrheit alles.
    #
    # Das ist genau der Kurzschluss, gegen den der SHADOW-Modus gebaut wurde:
    # ein System, das sich selbst befoerdert, weil es sich selbst beim Lernen
    # zugesehen hat. (Woertlich aus dem Original uebernommen, Z.312-324.)
    #
    # PHASE-3-ANPASSUNG (Konzeptdokument, Abschnitt "Das Kreditsignal"): Star
    # hat kein `effect_outcome:*`-Signal wie Zero -- stattdessen das oben
    # gebaute MECHANISCHE Signal (`mechanisch:erwaehnt`/`mechanisch:
    # ungenutzt`). Der Auftrag ist ausdruecklich STRENGER als Zeros Original:
    # nicht "alles ausser Bootstrap/Selftest/Probe/Seed zaehlt", sondern NUR
    # `signal_class LIKE 'mensch:%'` zaehlt fuer die Befoerderungsschwelle.
    # Das mechanische Signal ist eine VERMUTUNG ueber Nuetzlichkeit, keine
    # Messung -- genau die Selbstbestaetigungsschleife, vor der Zeros eigener
    # Korrekturkommentar oben warnt, waere hier: der Retriever saehe seiner
    # eigenen Auswahl beim "erwaehnt werden" zu und befoerderte sich selbst
    # darauf. OHNE MINDESTENS EIN `mensch:`-SIGNAL BEFOERDERT DIESE FUNKTION
    # NIE -- das ist keine Verschaerfung "zur Sicherheit", das ist die
    # woertliche Vorgabe aus PHASE 3.
    MIN_SCHLUESSEL_FUER_AKTIV = 3
    MIN_QUITTUNGEN_FUER_AKTIV = 2

    @classmethod
    def befoerderung_pruefen(cls, c: sqlite3.Connection) -> dict:
        """Prueft das validierte Lernsignal und hebt SHADOW auf ACTIVE, wenn
        genug da ist. Rein mechanisch, kein Modellaufruf, keine Behauptung
        ohne Beleg -- die Rueckgabe traegt immer die Rohzahlen.

        PHASE 3 ruft diese Funktion NIE automatisch aus einem Hook auf
        (Nicht-Ziel: "Keine Befoerderung auf ACTIVE in dieser Phase. Gar
        keine."). Erreichbar ausschliesslich ueber `stern
        befoerderung-pruefen`, ein Handbefehl fuer Gabriel/die Abnahme --
        wie `stern messen`."""
        vorher = cls.mode(c)
        schluessel = int(c.execute(
            "SELECT COUNT(*) FROM retrieval_entrypoint_policy WHERE pulls>=1"
        ).fetchone()[0])
        quittungen = int(c.execute(
            "SELECT COUNT(DISTINCT receipt_id) FROM retrieval_feedback "
            "WHERE signal_class LIKE 'mensch:%'"
        ).fetchone()[0])
        alle_quittungen = int(c.execute(
            "SELECT COUNT(DISTINCT receipt_id) FROM retrieval_feedback"
        ).fetchone()[0])
        ergebnis: dict = {
            "modus_vorher": vorher,
            "gelernte_schluessel": schluessel,
            "schwelle_schluessel": cls.MIN_SCHLUESSEL_FUER_AKTIV,
            "unabhaengige_mensch_quittungen": quittungen,
            "schwelle_quittungen": cls.MIN_QUITTUNGEN_FUER_AKTIV,
            "quittungen_gesamt": alle_quittungen,
        }
        if vorher == "ACTIVE":
            if quittungen < cls.MIN_QUITTUNGEN_FUER_AKTIV:
                c.execute(
                    "UPDATE retrieval_policy_state SET mode='SHADOW', updated_at=? WHERE singleton=1",
                    (_jetzt(),),
                )
                ergebnis.update(
                    befoerdert=False, zurueckgenommen=True, modus_nachher="SHADOW",
                    grund=(f"zurueckgenommen: {quittungen} unabhaengige mensch:-Quittungen, "
                           f"Schwelle {cls.MIN_QUITTUNGEN_FUER_AKTIV}"),
                )
                return ergebnis
            ergebnis.update(befoerdert=False, modus_nachher="ACTIVE",
                             grund="bereits ACTIVE, unabhaengiges mensch:-Signal genuegt")
            return ergebnis
        if schluessel < cls.MIN_SCHLUESSEL_FUER_AKTIV or quittungen < cls.MIN_QUITTUNGEN_FUER_AKTIV:
            ergebnis.update(
                befoerdert=False, modus_nachher="SHADOW",
                grund=("noch nicht genug validiertes Lernsignal: "
                       f"{schluessel}/{cls.MIN_SCHLUESSEL_FUER_AKTIV} Schluessel, "
                       f"{quittungen}/{cls.MIN_QUITTUNGEN_FUER_AKTIV} unabhaengige mensch:-Quittungen"),
            )
            return ergebnis
        now = _jetzt()
        c.execute(
            "UPDATE retrieval_policy_state SET mode='ACTIVE', "
            "policy_version=policy_version+1, updated_at=? WHERE singleton=1",
            (now,),
        )
        ergebnis.update(
            befoerdert=True, modus_nachher="ACTIVE",
            grund="Schwellen erreicht: gelernte Eintrittspunkte + unabhaengige mensch:-Quittungen",
        )
        return ergebnis


def _transcript_assistenttext_lesen(transcript_path) -> str:
    """Wie `_turn_nutzung_lesen()` oben, aber liefert den TEXT statt der
    usage-Zeile -- fuer das mechanische Kreditsignal (PHASE 3): das braucht
    den tatsaechlichen Antworttext dieser Runde, nicht Tokenzahlen."""
    if not transcript_path:
        return ""
    try:
        p = Path(transcript_path)
        if not p.is_file():
            return ""
        teile = []
        with p.open(encoding="utf-8", errors="ignore") as f:
            for zeile in f:
                zeile = zeile.strip()
                if not zeile:
                    continue
                try:
                    d = json.loads(zeile)
                except Exception:
                    continue
                if d.get("type") != "assistant":
                    continue
                inhalt = (d.get("message") or {}).get("content")
                if isinstance(inhalt, list):
                    for block in inhalt:
                        if isinstance(block, dict) and block.get("type") == "text":
                            teile.append(str(block.get("text") or ""))
                elif isinstance(inhalt, str):
                    teile.append(inhalt)
        # nur die letzten ~20k Zeichen -- Stop-Hook-Budget (2.0s), kein
        # Grund, eine lange Sitzung komplett zu durchsuchen.
        return " ".join(teile)[-20000:]
    except Exception:
        return ""


def _microclay_schatten(session_id: str, nutzertext: str) -> None:
    """UserPromptSubmit -- PARALLEL zum ausliefernden `automatischer_abruf()`
    oben. Rechnet MicroClays Baseline/Adaptive-Abruf, schreibt NUR nach
    `retrieval_episodes` (SHADOW), gibt NICHTS an den Hook-Ausgang zurueck.
    Fail-silent wie jeder Hook-Pfad -- ein Fehler hier darf die Sitzung nie
    beruehren."""
    if not nutzertext or not nutzertext.strip():
        return
    t0 = time.monotonic()
    try:
        with contextlib.closing(schreib_verbindung()) as c:
            c.row_factory = sqlite3.Row
            # Singleton-Zeile sicherstellen, damit `retrieval_policy_state`
            # sichtbar auf SHADOW steht statt nur per Code-Rueckfall zu gelten
            # (Testschritt 1: "retrieval_policy_state steht auf SHADOW").
            c.execute(
                "INSERT OR IGNORE INTO retrieval_policy_state"
                "(singleton,mode,policy_version,updated_at) VALUES(1,?,?,?)",
                (MicroClay.DEFAULT_MODE, MicroClay.POLICY_VERSION, _jetzt()),
            )
            aktiv = c.execute(
                "SELECT generation FROM active_turns WHERE session_id=?",
                (session_id,),
            ).fetchone()
            generation = int(aktiv["generation"]) if aktiv else 1
            # Frischer causal_id je Aufruf, ABSICHTLICH nicht aus active_turns
            # -- Stars active_turns ist eine Zeile je SITZUNG (von
            # zyklus_oeffnen() einmal bei SessionStart angelegt), nicht je
            # Runde. Wuerde man diesen einen causal_id fuer jede Nachricht
            # der Sitzung wiederverwenden, griffe MicroClays eigener
            # Idempotenz-Zweig (oben in retrieve()) ab der ZWEITEN Nachricht
            # und es entstuende nie wieder eine neue retrieval_episodes-Zeile
            # -- das haette die gesamte Divergenzmessung (Testschritt 2)
            # unmoeglich gemacht.
            causal_id = str(uuid.uuid4())
            #
            # PHASE 6/ZECP10, BUGFIX WAEHREND DES BAUS (nicht Teil des
            # Phase-6-Auftrags selbst, aber Voraussetzung dafuer -- offen
            # gelegt): dieselbe Logik, die oben schon fuer causal_id gilt,
            # war fuer turn_id/episode_id NICHT angewandt -- beide kamen bis
            # hierher aus active_turns (eine Zeile je SITZUNG). retrieval_
            # episodes.turn_id UND .episode_id sind aber beide UNIQUE (+FK auf
            # turns/workspace_episodes). Befund [LIVE, vor diesem Fix
            # geprueft]: JEDE der 106 vorhandenen Sitzungen in unified.db hat
            # GENAU EINE retrieval_episodes-Zeile -- jeder zweite und
            # weitere UserPromptSubmit derselben Sitzung schlug seit
            # Einfuehrung von P3 mit "UNIQUE constraint failed:
            # retrieval_episodes.episode_id" fehl und wurde vom fail-silent
            # Try/Except dieser Funktion stillschweigend verschluckt (nur
            # "microclay schatten fehler" in hook.log, kein Absturz). MicroClay
            # lief damit in JEDER Sitzung nach der ersten Nachricht faktisch
            # tot -- ein reiner Erstrunden-Mechanismus, keine laufende
            # Buchhaltung. Fuer PHASE 6/P6b ist das kein Randfall: Dauersieger
            # (Test 3, "fuenfmal dieselbe Frage") *verlangt* mehrere Runden je
            # Sitzung -- ohne diesen Fix waere P6b nie ueberpruefbar UND in
            # der Praxis nur fuer die allererste Nachricht jeder Sitzung
            # wirksam. Fix: eigene, RUNDENSCHARFE turns-/workspace_episodes-
            # Zeilen je Aufruf (exakt das gleiche Muster wie zyklus_oeffnen()
            # fuer die SITZUNG einmalig anlegt) -- beruehrt active_turns/
            # EffectGate (P2) nicht, die bleiben sitzungsscharf wie zuvor.
            turn_id = _uid("t-mc")
            episode_id = _uid("ep-mc")
            mc_jetzt = _jetzt()
            mc_ordinal = _naechste_ordinal(c, session_id)
            c.execute(
                "INSERT INTO turns (turn_id, session_id, user_id, ordinal, ts, role, "
                "content, fidelity, compression_generation, resource_refs, causal_refs, "
                "provenance) VALUES (?,?,?,?,?,?,?,?,?,?,?,?)",
                (
                    turn_id, session_id, USER_ID, mc_ordinal, mc_jetzt, "system",
                    "[MicroClay-Schattenrunde -- kein echtes Transkript, nur "
                    "Randmarke fuer retrieval_episodes/gw_*]",
                    "marker", 0, "[]", "[]",
                    json.dumps({"quelle": "_microclay_schatten"}, ensure_ascii=False),
                ),
            )
            c.execute(
                "INSERT INTO workspace_episodes (episode_id, session_id, ts, "
                "observation_turn_id, salience, alternatives, selected, state) "
                "VALUES (?,?,?,?,?,?,?,?)",
                (episode_id, session_id, mc_jetzt, turn_id, 1.0, "[]", None, "offen"),
            )
            ergebnis = MicroClay.retrieve(
                c, query=nutzertext, user_id=USER_ID, session_id=session_id,
                turn_id=turn_id, episode_id=episode_id, causal_id=causal_id,
                generation=generation,
            )
            c.commit()
        dauer_ms = round((time.monotonic() - t0) * 1000)
        beleg = ergebnis.get("receipt") or {}
        _log(
            f"microclay schatten sid={session_id} dauer={dauer_ms}ms "
            f"retrieval_id={beleg.get('retrieval_id')} modus={beleg.get('mode')} "
