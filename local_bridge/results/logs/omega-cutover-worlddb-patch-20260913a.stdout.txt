+def discover_semantic_dbs(layout: Layout) -> list[Path]:
+    """Discover every SQLite-named store inside the protected volatile trees.
+
+    Directory/file symlinks fail closed: cutover evidence must describe the real
+    disposable tree, never an escape or alias outside it.
+    """
+    found: list[Path] = []
+    for rel in PROTECTED_ROOTS:
+        root = layout.rootfs_run / rel
+        if not root.exists():
+            continue
+        if root.is_symlink() or not root.is_dir():
+            fail(f"protected semantic root is not a real directory: {root}")
+        for dirpath, dirnames, filenames in os.walk(root, followlinks=False):
+            here = Path(dirpath)
+            for dirname in dirnames:
+                child = here / dirname
+                if child.is_symlink():
+                    fail(f"symlink inside protected semantic tree: {child}")
+            for filename in filenames:
+                if not filename.endswith(".db"):
+                    continue
+                candidate = here / filename
+                if candidate.is_symlink() or not candidate.is_file():
+                    fail(f"semantic DB path is not a regular file: {candidate}")
+                # Opening/fingerprinting here makes a malformed/unreadable .db
+                # a hard blocker instead of silently omitting it.
+                logical_fingerprint(candidate)
+                found.append(candidate)
+    return sorted(set(found), key=lambda p: str(p))
