function kopfRollDatei(w){
  return `/kopf_rgba/roll/gier_+00_nick_+00_roll_${w < 0 ? "-" : "+"}${String(Math.abs(w)).padStart(2,"0")}.png`;
}
let kopfRollWert = 0;        // Grad des aktuell angezeigten Roll-Bildes
let kopfRollSoll = 0;        // Grad des angepeilten Ziels (Feder-Futter)

let kopfRollFertigFn = null; // einmaliger Callback nach dem letzten Schritt
function kopfRollSequenzZu(ziel, fertig){
  const schritte = [];
  const richtung = ziel > kopfRollWert ? 5 : -5;
  for(let w = kopfRollWert + richtung; ; w += richtung){
    schritte.push({w, src: kopfRollDatei(w)});
    if(w === ziel) break;
  }
  kopfRollSchritte = schritte; kopfRollIdx = 0; kopfRollLetzterT = 0;
  kopfRollFertigFn = fertig || null;
}
function kopfRollTick(jetzt){
  if(!kopfRollSchritte) return;
  if(jetzt - kopfRollLetzterT < KOPF_ROLL_MS) return;
  const schritt = kopfRollSchritte[kopfRollIdx];
  const eintrag = bildLaden(schritt.src);
  if(!eintrag.ok) return; // Frame noch nicht dekodiert -- warten statt halbes Bild zeigen
  kopfRollLetzterT = jetzt;
  const ein = kopfRollAktiv ? "kopf-schicht" : "kopf-schicht2";
  const aus = kopfRollAktiv ? "kopf-schicht2" : "kopf-schicht";
  $(ein).src = schritt.src;
  // ⭐⭐ PERF-FIX 2026-09-11: atmen() aktualisiert seit diesem Fix nur noch
  // die aktuell sichtbare Kopf-Ebene (siehe atmen() oben) -- die Ebene, die
  // hier gerade sichtbar wird, muss deshalb JETZT einmalig den brandaktuellen
  // Atem-Wert bekommen, sonst zeigt sie beim ersten sichtbaren Frame noch den
  // Wert von ihrem letzten Auftritt (Pop). _atmenTransform/_atmenFilter sind
  // von atmen() diesen Frame schon gesetzt (schleife() ruft atmen() vor
  // kopfRollTick() auf).
  $(ein).style.transform = _atmenTransform;
  $(ein).style.filter = _atmenFilter;
  $(ein).style.opacity = "1";
  $(aus).style.opacity = "0";
  kopfRollAktiv = !kopfRollAktiv;
  kopfRollWert = schritt.w;
  kopfRollIdx++;
  if(kopfRollIdx >= kopfRollSchritte.length){
    const f = kopfRollFertigFn;
    kopfRollSchritte = null; kopfRollFertigFn = null;
    if(f) f();
  }
}

--- function kopfKoerperDrehSchritt ---
function kopfKoerperDrehSchritt(){
  // Federziel aus dem ECHTEN Rollwert (12% davon, auf ±3° begrenzt) -- die
  // Bildpose macht die Neigung, die Feder nur das kleine Nachschwingen.
  const zielRot = Math.max(-3, Math.min(3, kopfRollSoll * 0.12));
  [kDrehRot, kDrehRotV] = federSchritt(kDrehRot, kDrehRotV, zielRot, 0.020, 0.20);
  [kDrehTy,  kDrehTyV ] = federSchritt(kDrehTy,  kDrehTyV,  0,      0.020, 0.20);
  // 34% des AKTUELLEN (nicht Ziel-)Kopfwerts + eigene weichere Feder =
  // Koerper folgt spaeter und schwaecher, ohne starren festen Versatz.
  [bDrehRot, bDrehRotV] = federSchritt(bDrehRot, bDrehRotV, kDrehRot*0.34, 0.010, 0.26);
  [bDrehTy,  bDrehTyV ] = federSchritt(bDrehTy,  bDrehTyV,  kDrehTy*0.34,  0.010, 0.26);
  $("kopf-dreh").style.transform = `rotate(${kDrehRot.toFixed(2)}deg) translateY(${kDrehTy.toFixed(1)}px)`;
  $("koerper-dreh").style.transform = `rotate(${bDrehRot.toFixed(2)}deg) translateY(${bDrehTy.toFixed(1)}px)`;
  // Rest-Ausgleich fuer #pupillen-glow/#augen-blink -- siehe Kommentar oben
  // (STIRN-LUECKE-FIX): Ziel ist jetzt kopfRollWert selbst (kDrehRot kuerzt
  // sich per CSS-Vererbung von selbst weg, siehe Kommentar), UND geglaettet
  // per Lerp statt hartem Sprung -- entkoppelt den Flicken vom 5°-Sprung
  // des Fotowechsels, ohne ihn wahrnehmbar hinter dem echten Winkel
  // zurueckfallen zu lassen (Faktor 0.4, faengt sich in 2-3 Frames).
  // Geometrische Registrierung ist atomar mit dem bereits in kopfRollTick()
  // sichtbaren Foto-Schritt. Nur der gemeinsame #kopf-dreh/#koerper-dreh-Wrapper
  // darf federn. Eine lokale Lerp-Verzoegerung erzeugt sonst 2-3 Frames lang
  // absichtliche Fehlregistrierung zwischen Foto und Augen-/Blink-Overlay.
  kopfRollAusgleichDeg = kopfRollWert;
  const zielOffset = KOPF_ROLL_AUGEN_OFFSET[String(kopfRollWert)] || {dx:0, dy:0};
  augenOffsetDx = zielOffset.dx;
  augenOffsetDy = zielOffset.dy;
  const augenDxCss = augenOffsetDx * _kopfCanvasScale;
  const augenDyCss = augenOffsetDy * _kopfCanvasScale;
  $("augen-blink").style.transform =
    `translate(${augenDxCss.toFixed(2)}px, ${augenDyCss.toFixed(2)}px) rotate(${kopfRollAusgleichDeg.toFixed(2)}deg)`;
}

function sonderausdruckZeigen(register, ausdruck, haltenMs){

  haltenMs = haltenMs || 1400;
  _sonderLaeuft = true;
  $("mund-effekt").style.opacity = "0";
  const stufeSrc = n => `/ausdruck_erweitert/${register}/${ausdruck}/frontal/stufe_0${n}.png`;
  const wechsel = (neu, mitSchwung) => {
    const sichtbar = ausdruckAktiv === 1 ? $("ausdruck-schicht") : $("ausdruck-schicht2");
    const versteckt = ausdruckAktiv === 1 ? $("ausdruck-schicht2") : $("ausdruck-schicht");

    // Ebene braucht den brandaktuellen Wert deshalb hier einmalig.

    versteckt.style.filter = _atmenFilter;
    versteckt.style.opacity = "1";
    sichtbar.style.opacity = "0";
    ausdruckAktiv = ausdruckAktiv === 1 ? 2 : 1;
    if(mitSchwung){
      const rahmen = $("gesicht-rahmen");
      rahmen.classList.remove("pose-wechsel");
      requestAnimationFrame(() => rahmen.classList.add("pose-wechsel"));
      setTimeout(() => rahmen.classList.remove("pose-wechsel"), 420);
    }
  };
  const sauber = () => {
    $("ausdruck-schicht").style.opacity = "0";
    $("ausdruck-schicht2").style.opacity = "0";
    $("mund-effekt").style.opacity = "1";
    if(!mundAktivFrames) $("mund-effekt").src = `/mund_keyframes/${MUND_RUHE}.png`;
    _sonderLaeuft = false;
    gesichtNeuZeichnen(true);
  };

    stufe++;

  const runter = () => {

--- function gazeSchritt ---
function gazeSchritt(){
  const ziel = gazeZielBestimmen();
  zielGazeX = ziel.x; zielGazeY = ziel.y;
  aktGazeX += (zielGazeX - aktGazeX) * 0.07;
  aktGazeY += (zielGazeY - aktGazeY) * 0.07;
  const maxPx = 8.5; // Bewegungsspielraum bewusst klein -- ein Glimmer, kein Springen
  const dx = (aktGazeX - 0.5) * 2 * maxPx + mikroSakX;
  const dy = (aktGazeY - 0.5) * 2 * maxPx + BLICK_ABWAERTS_BIAS_PX + mikroSakY;
  // ⭐⭐ 2026-09-11 NACHTRAG 4: rotate() dranhaengen (Rest-Ausgleich zur echten
  // Rollpose, siehe kopfKoerperDrehSchritt()) -- sonst bleibt der Glimmer beim
  // Kopfkippen an der falschen Stelle stehen (Geisterkopf-Kante).
  // ⭐⭐⭐ 2026-09-12: dieselbe exakte Roll-Translation wie #augen-blink (siehe
  // KOPF_ROLL_AUGEN_OFFSET/kopfKoerperDrehSchritt oben) draufaddiert -- sonst
  // driftet der Glimmer beim Rollen vom Blinzel-Flicken weg, obwohl beide
  // denselben Augen-Anker verfolgen sollen.
  const dxGesamt = dx + augenOffsetDx * _kopfCanvasScale;
  const dyGesamt = dy + augenOffsetDy * _kopfCanvasScale;
  $("pupillen-glow").style.transform = `translate(${dxGesamt.toFixed(2)}px, ${dyGesamt.toFixed(2)}px) rotate(${kopfRollAusgleichDeg.toFixed(2)}deg)`;
  // ⭐ 2026-09-11: waehrend eines Sonderausdrucks aus -- dessen Frames tragen
  // ihre eigenen Augen, sonst laege der Erkennungs-Glimmer auf einem fremden
  // Gesicht (zweites Augenpaar statt zweiter Kopf).
  gazeSichtbar = ziel.opazitaet > 0; // Rueckwaertskompatibilitaet: anderswo evtl. gelesen
  $("pupillen-glow").style.opacity = _sonderLaeuft ? "0" : String(ziel.opazitaet);
}
