function gazeZielBestimmen(){
  if(!gazeBasisSichtbar()) return {x:0.5, y:0.5, opazitaet:0};
  const wegAktiv = blickPauseAktiv || gespraechsFokus === "thema";
  if(wegAktiv) return {x:_wegZiel.x, y:_wegZiel.y, opazitaet:0.4};
  return {x:serverBlick.cx, y:serverBlick.cy, opazitaet:1};
}
function gazeSchritt(){
  const ziel = gazeZielBestimmen();
  zielGazeX = ziel.x; zielGazeY = ziel.y;
  aktGazeX += (zielGazeX - aktGazeX) * 0.07;
  aktGazeY += (zielGazeY - aktGazeY) * 0.07;
  const maxPx = 8.5; // Bewegungsspielraum bewusst klein -- ein Glimmer, kein Springen
  const dx = (aktGazeX - 0.5) * 2 * maxPx + mikroSakX;
  const dy = (aktGazeY - 0.5) * 2 * maxPx + BLICK_ABWAERTS_BIAS_PX + mikroSakY;
  // ⭐⭐ 2026-09-11 NACHTRAG 4: rotate() dranhaengen (Rest-Ausgleich zur echten
  // Rollpose, siehe kopfKoerperDrehSchritt()) -- sonst bleibt der Glimmer beim
  // Kopfkippen an der falschen Stelle stehen (Geisterkopf-Kante).
  // ⭐⭐⭐ 2026-09-12: dieselbe exakte Roll-Translation wie #augen-blink (siehe
  // KOPF_ROLL_AUGEN_OFFSET/kopfKoerperDrehSchritt oben) draufaddiert -- sonst
  // driftet der Glimmer beim Rollen vom Blinzel-Flicken weg, obwohl beide
  // denselben Augen-Anker verfolgen sollen.
  const dxGesamt = dx + augenOffsetDx * _kopfCanvasScale;
  const dyGesamt = dy + augenOffsetDy * _kopfCanvasScale;
  $("pupillen-glow").style.transform = `translate(${dxGesamt.toFixed(2)}px, ${dyGesamt.toFixed(2)}px) rotate(${kopfRollAusgleichDeg.toFixed(2)}deg)`;
  // ⭐ 2026-09-11: waehrend eines Sonderausdrucks aus -- dessen Frames tragen
  // ihre eigenen Augen, sonst laege der Erkennungs-Glimmer auf einem fremden
  // Gesicht (zweites Augenpaar statt zweiter Kopf).
  gazeSichtbar = ziel.opazitaet > 0; // Rueckwaertskompatibilitaet: anderswo evtl. gelesen
  $("pupillen-glow").style.opacity = _sonderLaeuft ? "0" : String(ziel.opazitaet);
}

  if(neuerAusdruck && neuerAusdruck !== aktuellerAusdruck){
    if(_ausdruckTimer) clearTimeout(_ausdruckTimer);

      aktuellerAusdruck = neuerAusdruck;
