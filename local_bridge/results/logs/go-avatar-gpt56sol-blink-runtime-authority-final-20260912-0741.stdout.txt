def _vielleicht_ai(pfad: str) -> str:
    """Gibt die `_ai`-Variante zurueck, wenn sie existiert und der Schalter an
    ist. Faellt sonst still auf das Original zurueck -- ein fehlender
    geschaerfter Ordner darf die Buehne NIE schwarz machen."""
    if RELIEF_SATZ:
        r = pfad + "_relief2"
        if os.path.isdir(r):
            return r
    if not AI_BILDSATZ:
        return pfad
    kandidat = pfad + "_ai"
    return kandidat if os.path.isdir(kandidat) else pfad
