+def _werkzeug_treffer(c, stichwoerter: list, session_id: str, bereits_gezeigt: set) -> list:
+    """Wortschnittmenge Stichwoerter <-> kuratierte Tags (siehe Kommentar

+    treffer = []
+    try:
+        for tool_id, name, beschreibung, einstiegspunkt, tags_json in c.execute(
+            "SELECT tool_id, name, beschreibung, einstiegspunkt, tags "
+            "FROM star_werkzeugregister WHERE aktiv=1"
+        ):
+            item_id = f"werkzeug:{tool_id}"
+            if item_id in bereits_gezeigt:
+                continue
+            try:
+                tags = json.loads(tags_json)
+            except Exception:
+                tags = []
+            treffer_worte = set()
+            for begriff in stichwoerter:
+                for tag in tags:
+                    if begriff == tag or (
+                        len(begriff) >= 4 and len(tag) >= 4
