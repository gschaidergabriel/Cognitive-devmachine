def extract_foreground(path: Path, roi: tuple[int, int, int, int]) -> Image.Image:
    im = Image.open(path).convert('RGB')
    a = np.asarray(im).astype(np.int16)
    color = a.max(axis=2) - a.min(axis=2)
    lum = a.mean(axis=2)
    mask = (color > 22) | (lum < 115)
    gate = np.zeros(mask.shape, bool)
    x0, y0, x1, y1 = roi
    gate[y0:y1, x0:x1] = True
    mask &= gate
    mask = ndimage.binary_closing(mask, iterations=3)
    mask = ndimage.binary_dilation(mask, iterations=1)
    labels, count = ndimage.label(mask)
    if count:
        sizes = ndimage.sum(mask, labels, range(1, count + 1))
        keep = np.zeros_like(mask)
        for idx in np.argsort(sizes)[-8:]:
            if sizes[idx] > 80:
                keep |= labels == (idx + 1)
        mask = keep
    mask = ndimage.binary_fill_holes(mask)
    # Pull the cut inward to remove the generated grey checkerboard fringe.
    mask = ndimage.binary_erosion(mask, iterations=3)
    rgba = np.dstack([a.astype(np.uint8), (mask * 255).astype(np.uint8)])
    return Image.fromarray(rgba, 'RGBA')


def normalize_layer(im: Image.Image, target: tuple[int, int], anchor: tuple[int, int], anchor_mode: str) -> tuple[Image.Image, tuple[int, int, int, int], float]:
    bbox = im.getchannel('A').getbbox()
    if bbox is None:
        raise RuntimeError('empty alpha mask')
    crop = im.crop(bbox)
    tw, th = target
    scale = min(tw / crop.width, th / crop.height)
    nw, nh = max(1, round(crop.width * scale)), max(1, round(crop.height * scale))
    crop = crop.resize((nw, nh), Image.Resampling.LANCZOS)
    canvas = Image.new('RGBA', (W, H), (0, 0, 0, 0))
    if anchor_mode == 'bottom_center':
        pos = (round(anchor[0] - nw / 2), round(anchor[1] - nh))
    elif anchor_mode == 'top_center':
        pos = (round(anchor[0] - nw / 2), round(anchor[1]))
    else:
        raise ValueError(anchor_mode)
    canvas.alpha_composite(crop, pos)
    return canvas, bbox, scale


def save_asset(im: Image.Image, rel: str, **meta):
    path = OUT / rel
    path.parent.mkdir(parents=True, exist_ok=True)
    temp = path.with_suffix(path.suffix + '.part')
    im.save(temp, format='PNG', optimize=False)
    # Force a complete decode before making the file visible as a final asset.
    with Image.open(temp) as verify:
        verify.load()
    temp.replace(path)
    manifest['assets'].append({'path': rel, **meta})


def transform_crop_about_anchor(layer: Image.Image, angle=0.0, sx=1.0, sy=1.0, anchor=(768, 600), target_anchor=None) -> Image.Image:
    bbox = layer.getchannel('A').getbbox()
    crop = layer.crop(bbox)
    crop = crop.resize((max(1, round(crop.width * sx)), max(1, round(crop.height * sy))), Image.Resampling.BICUBIC)
    if angle:
        crop = crop.rotate(angle, resample=Image.Resampling.BICUBIC, expand=True)
    if target_anchor is None:
        target_anchor = anchor
    canvas = Image.new('RGBA', (W, H), (0, 0, 0, 0))
    x = round(target_anchor[0] - crop.width / 2)
    y = round(target_anchor[1] - crop.height)
    canvas.alpha_composite(crop, (x, y))
    return canvas


def transform_body_from_neck(layer: Image.Image, angle=0.0, sx=1.0, sy=1.0, target_anchor=(768, 505)) -> Image.Image:
    bbox = layer.getchannel('A').getbbox()

def rounded_alpha(size, radius=45, feather=3):
    m = Image.new('L', size, 0)
    d = ImageDraw.Draw(m)
    d.rounded_rectangle((feather, feather, size[0] - feather - 1, size[1] - feather - 1), radius=radius, fill=255)
    if feather:
        m = m.filter(__import__('PIL.ImageFilter', fromlist=['GaussianBlur']).GaussianBlur(feather))
    return m


def ellipse_alpha(size, feather=3):
    m = Image.new('L', size, 0)
    d = ImageDraw.Draw(m)
    inset_x = max(5, round(size[0] * .035))
    inset_y = max(4, round(size[1] * .035))
    d.ellipse((inset_x, inset_y, size[0]-inset_x-1, size[1]-inset_y-1), fill=255)
    if feather:
        m = m.filter(__import__('PIL.ImageFilter', fromlist=['GaussianBlur']).GaussianBlur(feather))
    return m


def suppress_red(im: Image.Image) -> Image.Image:
    a = np.array(im)
    rgb = a[:, :, :3].astype(np.int16)
    red = (rgb[:, :, 0] > 105) & (rgb[:, :, 0] > rgb[:, :, 1] * 1.35) & (rgb[:, :, 0] > rgb[:, :, 2] * 1.15)
    rgb[red, 0] = (rgb[red, 1] * 0.35).astype(np.int16)
    rgb[red, 1] = np.maximum(rgb[red, 1], 35)
    rgb[red, 2] = np.maximum(rgb[red, 2], 55)
    a[:, :, :3] = np.clip(rgb, 0, 255).astype(np.uint8)
    return Image.fromarray(a, 'RGBA')

def crop_scaled_patch(path: Path, raw_box, head_bbox, head_scale, alpha_shape='rounded'):
    im = Image.open(path).convert('RGBA').crop(raw_box)
    size = (max(1, round(im.width*head_scale)), max(1, round(im.height*head_scale)))
    im = im.resize(size, Image.Resampling.LANCZOS)
    if alpha_shape == 'rounded':
        im.putalpha(rounded_alpha(size, radius=max(20, size[1]//4), feather=3))
    elif alpha_shape == 'ellipse':
        alpha = np.array(ellipse_alpha(size, feather=3))
        arr = np.array(im)
        rgb = arr[:, :, :3].astype(np.int16)
        grey_background = ((rgb.max(axis=2)-rgb.min(axis=2)) < 16) & (rgb.mean(axis=2) > 118)
        labels, _ = ndimage.label(grey_background)
        edge_labels = np.unique(np.concatenate([labels[0], labels[-1], labels[:,0], labels[:,-1]]))
        edge_background = np.isin(labels, edge_labels[edge_labels != 0])
        edge_background = ndimage.binary_dilation(edge_background, iterations=1)
        alpha[edge_background] = 0
        im.putalpha(Image.fromarray(alpha.astype(np.uint8), 'L'))
    return im

def mood_transform(arr, mood, phase):
    x = arr.astype(np.float32)
    if mood == 'ruhig':
        x *= np.array([0.72, 0.88, 1.10]); gain=.92
    elif mood == 'wach':
        x *= np.array([0.72, 1.16, 1.20]); gain=1.06
    elif mood == 'angespannt':
        r, g, b = x[:,:,0], x[:,:,1], x[:,:,2]
        x = np.stack([np.maximum(r*1.15, b*.65), g*.67, b*.92], axis=2); gain=1.00
    elif mood == 'bedrohlich':
        r, g, b = x[:,:,0], x[:,:,1], x[:,:,2]
        x = np.stack([np.maximum(r*1.35, b*.98), g*.36, b*.38], axis=2); gain=.98
    else:
        r, g, b = x[:,:,0], x[:,:,1], x[:,:,2]
        x = np.stack([np.maximum(r*1.05, b*.34), g*1.16, b*1.22], axis=2); gain=1.14
    gain *= .97 + .03*math.sin(phase)
    return np.clip(x*gain,0,255).astype(np.uint8)


def build_backgrounds():
    base = np.array(Image.open(BG_RAW).convert('RGB'))
    yy = np.arange(H)[:,None]
    xx = np.arange(W)[None,:]
    for mood in ('ruhig','wach','angespannt','bedrohlich','heiter'):
        folder = OUT / 'hintergrund' / mood
        folder.mkdir(parents=True, exist_ok=True)
        for i in range(60):
            phase = 2*math.pi*i/60
            # Integer full-height cycles make frame 59 -> frame 00 periodic.
            cycles = {'ruhig':1,'wach':2,'angespannt':2,'bedrohlich':3,'heiter':1}[mood]
            yoff = round((i/60)*H*cycles) % H
            moved = np.roll(base, yoff, axis=0)
            shifts = np.rint((3 if mood=='ruhig' else 7)*np.sin(yy[:,0]/115 + phase)).astype(int)
            src_x = (xx - shifts[:,None]) % W
            moved = moved[np.arange(H)[:,None], src_x]
            moved = mood_transform(moved, mood, phase)
            im = Image.fromarray(moved,'RGB').quantize(colors=128, method=Image.Quantize.MEDIANCUT)
            rel=f'hintergrund/{mood}/{mood}_{i:03d}.png'
            save_asset(im, rel, layer='background', rgba=False, mood=mood, frame=i, loop_frames=60, quality='rig_ready_seamless_periodic')

def qa():
    rows=['path,width,height,mode,alpha_min,alpha_max,transparent_pixels']
    for p in sorted(OUT.rglob('*.png')):
        im=Image.open(p)
        if im.mode=='RGBA':
            aa=np.array(im.getchannel('A'))
            rows.append(f'{p.relative_to(OUT)},{im.width},{im.height},{im.mode},{aa.min()},{aa.max()},{int((aa<255).sum())}')
        else:
            rows.append(f'{p.relative_to(OUT)},{im.width},{im.height},{im.mode},NA,NA,0')
    (OUT/'qa/image_inventory.csv').write_text('\n'.join(rows)+'\n',encoding='utf-8')
