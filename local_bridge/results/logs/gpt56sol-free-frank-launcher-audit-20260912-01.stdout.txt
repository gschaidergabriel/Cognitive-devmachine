#!/usr/bin/env bash
# packen.sh — Schlussbericht bauen, PDF erzeugen, alles auf den Desktop legen.
#
# Reihenfolge ist wichtig: erst nachmessen, dann auswerten, dann setzen.
# Ein PDF, das vor der Nachmessung gebaut wird, zeigt die Baseline zweimal.
set -uo pipefail
L="$HOME/xeno-lauf"
D="$HOME/Desktop"
cd "$L"

echo "1/7  Global Workspace nachmessen"
python3 messstand.py --n 300 --marke "NACHHER — nach dem Umbau durch XO1+XS1" \
        --raus berichte/nachher.json 2>&1 | sed 's/^/     /'

echo "2/7  Tokenbilanz"
python3 tokenbilanz.py > /dev/null 2>&1 && echo "     berichte/tokenbilanz.json"
python3 wirkung.py > /dev/null 2>&1 && echo "     berichte/wirkung.json"
python3 vergleich.py > berichte/vergleich.txt 2>&1 && echo "     berichte/vergleich.txt"
python3 agentizitaet.py > berichte/agentizitaet.txt 2>&1 && echo "     berichte/agentizitaet.txt"

echo "3/7  Auswertung zusammenstellen"
python3 auswertung.py 2>&1 | sed 's/^/     /'

echo "4/7  Gesamtbericht bauen"
STAMP="$(date '+%Y-%m-%d %H:%M')"
{
  cat berichte/ERGEBNIS.md
  echo; echo "---"; echo
  cat berichte/EMERGENZ.md
  echo; echo "---"; echo
  cat berichte/ARCHITEKTUR-FRANK.md
  echo; echo "---"; echo
  cat berichte/AUSWERTUNG.md
  echo; echo '---'; echo
  cat berichte/HUERDEN.md
  echo; echo '---'; echo
  cat berichte/MASCHINENFORMAT.md
  echo; echo "---"; echo
  cat berichte/ABLATION.md
  echo; echo "---"; echo
  echo "# Anhang — Selektive Agentizitaet (Rohausgabe)"; echo
  printf '%s\n' '~~~'; cat berichte/agentizitaet.txt; printf '%s\n' '~~~'
  echo; echo "---"; echo
  cat berichte/ABSCHLUSS-KOORD.md
  for f in berichte/ABSCHLUSS-XO1.md berichte/ABSCHLUSS-XS1.md; do [ -f "$f" ] && { echo; echo "---"; echo; cat "$f"; }; done
  echo; echo "---"; echo
  echo "# Anhang — Wirkungsmetrik (Rohwerte)"; echo
  printf '%s\n' '~~~json'; cat berichte/wirkung.json; printf '%s\n' '~~~'
  echo; echo '---'; echo
  echo '# Anhang — Tokenbilanz (Rohwerte)'; echo
  printf '%s\n' '~~~json'; cat berichte/tokenbilanz.json; printf '%s\n' '~~~'
} > berichte/GESAMT.md
echo "     berichte/GESAMT.md ($(wc -c < berichte/GESAMT.md) Bytes)"

echo "5/7  PDF"
pandoc berichte/GESAMT.md -f gfm -t html5 --standalone --toc --toc-depth=2 \
  --metadata title="Xeno-Lauf — eine Stunde in einer fremden Sprache" \
  --metadata date="$STAMP" \
  -c /dev/null -o /tmp/xeno-gesamt.html 2>/dev/null
python3 - <<'PY'
from pathlib import Path
h = Path("/tmp/xeno-gesamt.html").read_text(encoding="utf-8")
css = """<style>
@page { size: A4; margin: 20mm 18mm; @bottom-center { content: counter(page); font: 9pt sans-serif; color:#888; } }
body { font: 10.5pt/1.5 "DejaVu Sans", sans-serif; color:#1a1a1a; }
h1 { font-size:19pt; border-bottom:2px solid #333; padding-bottom:4px; page-break-before:always; }
h1:first-of-type { page-break-before: avoid; }
h2 { font-size:14pt; margin-top:1.4em; color:#222; }
h3 { font-size:11.5pt; color:#444; }
code, pre { font-family:"DejaVu Sans Mono", monospace; font-size:8.5pt; }
pre { background:#f5f5f5; padding:7px 9px; border-left:3px solid #bbb; white-space:pre-wrap; word-wrap:break-word; }
table { border-collapse:collapse; width:100%; font-size:9pt; margin:0.8em 0; }
th,td { border:1px solid #ccc; padding:4px 6px; text-align:left; vertical-align:top; }
th { background:#eee; }
blockquote { border-left:3px solid #999; margin-left:0; padding-left:10px; color:#444; font-style:italic; }
#TOC { page-break-after: always; }
</style>"""
Path("/tmp/xeno-gesamt.html").write_text(h.replace("</head>", css + "</head>"), encoding="utf-8")

weasyprint /tmp/xeno-gesamt.html "$D/Xeno-Lauf-Auswertung.pdf" 2>/dev/null \
  && echo "     $D/Xeno-Lauf-Auswertung.pdf ($(du -h "$D/Xeno-Lauf-Auswertung.pdf" | cut -f1))" \
  || echo "     PDF FEHLGESCHLAGEN"

echo "6/7  Wissensdatenbank"
python3 einspeisen.py 2>&1 | sed 's/^/     /'

echo "7/7  Datenpaket"
Z="$D/Xeno-Lauf-Datenpaket.zip"
rm -f "$Z"
zip -qr "$Z" \
  berichte/ analysen/ db/verkehr.db gw-vorher/ \
  SPRACHE.txt BRIEF.md CLAUDE.md \
  messstand.py wache.py sprachwacht.py xsag auswertung.py tokenbilanz.py \
  interview.py vergleich.py einspeisen.py wirkung.py agentizitaet.py packen.sh \
  wache.log sprachwacht.log \
  -x '*/__pycache__/*' '*.pyc'
echo "     $Z ($(du -h "$Z" | cut -f1))"
echo
echo "fertig $(date '+%H:%M:%S')"
