function mundAnimation(text){
  const grobeSilben = Math.max(3, Math.round((text || "").length / 2.6));
  const dauerMs = Math.min(9000, grobeSilben * 220);
  let letzte = -1;
  const iv = setInterval(() => {
    let i = Math.floor(Math.random() * MUND_SPRECH_SET.length);
    if(i === letzte) i = (i + 1) % MUND_SPRECH_SET.length; // nie zweimal dieselbe Stellung hintereinander

    mundZielSetzen(MUND_SPRECH_SET[i]);
  }, 130);
  setTimeout(() => { clearInterval(iv); mundZielSetzen(MUND_RUHE); }, dauerMs);
}
